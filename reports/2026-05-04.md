```python
    def __init__(self, world_model, policy_model, inverse_dynamics):
        self.wm = world_model
        self.policy = policy_model
        self.idm = inverse_dynamics

    def decoupled_idm_inference(self, obs_t, instruction):
        # Paradigm 1: Predict first, then act (Decoupled)
        # 1. World model predicts future observations
        future_obs = self.wm.predict_future(obs_t, instruction)
        # 2. Inverse dynamics extracts action from current and future obs
        action_t = self.idm.predict_action(obs_t, future_obs)
        return action_t

    def unified_backbone_inference(self, obs_t, instruction):
        # Paradigm 2: Joint modeling in a single backbone
        # The model internally processes future prediction and action generation
        # Future visual branch might be marginalized or truncated during inference
        joint_representation = self.policy.encode(obs_t, instruction)
        action_t, _ = self.policy.decode_jointly(joint_representation)
        return action_t

    def simulator_rl_training_loop(self, initial_obs, instruction, steps):
        # Paradigm 3: World Model as a Learned Simulator for RL
        obs_t = initial_obs
        trajectory = []
        
        for t in range(steps):
            # Policy proposes an action
            action_t = self.policy.select_action(obs_t, instruction)
            
            # World model simulates the environment transition and reward
            next_obs_imagined, reward_imagined = self.wm.step(obs_t, action_t)
            
            trajectory.append((obs_t, action_t, reward_imagined, next_obs_imagined))
            obs_t = next_obs_imagined
            
        # Update policy using imagined trajectory (e.g., via GRPO or PPO)
        self.policy.update_from_imagination(trajectory)
```

### 第二模块：核心架构解释

Being-H0.7 的核心架构基于混合专家 Transformer（MoT），其本质是一个在单次前向传播中同时处理先验（部署）和后验（训练）计算流的双分支网络。

**1. 输入与上下文编码**
模型首先接收多模态上下文，包括语言指令、历史观测图像序列和机器人本体状态。这些输入被编码为共享的上下文 Token 序列。

**2. 隐式推理空间构建**
在共享上下文和动作 Token 之间，模型插入了固定数量的隐式查询（Latent Queries）。
- **先验分支（Prior Branch）**：使用一组可学习的参数化 Queries。
- **后验分支（Posterior Branch）**：使用经过冻结视觉编码器（V-JEPA）提取的未来观测图像特征（Future Embeddings）。

**3. 双分支注意力掩码（Dual-Branch Attention Mask）**
为了在同一个 Transformer 序列中并行训练两个分支且防止信息穿越，模型设计了特定的注意力掩码：共享上下文对所有 Token 可见；先验分支的 Token 与后验分支的 Token 相互不可见。

**4. 联合对齐与正则化（Joint Alignment & Regularization）**
在 Transformer 的深层，提取先验分支和后验分支对应位置的隐状态，计算 Frobenius 范数损失进行强制对齐。为防止隐空间特征坍塌，引入了基于特征幅值的范数正则化（Norm Preservation）和基于特征值分布的秩正则化（Spectral Diversity）。

**5. 动作解码（Action Decoding）**
两个分支的隐状态最终分别输入到基于 Flow-matching 的动作解码器中，预测目标动作轨迹。推理时，仅执行先验分支。

```python
import torch
import torch.nn as nn
import torch.nn.functional as F

class BeingH07(nn.Module):
    def __init__(self, context_encoder, future_encoder, transformer_backbone, action_head):
        super().__init__()
        self.context_encoder = context_encoder
        self.future_encoder = future_encoder # Frozen V-JEPA
        self.transformer = transformer_backbone
        self.action_head = action_head
        
        self.num_queries = 16
        self.hidden_dim = 1024
        # Learnable latent queries for the prior branch
        self.prior_queries = nn.Parameter(torch.randn(1, self.num_queries, self.hidden_dim))
        
    def forward(self, instruction, obs_history, state, future_obs, target_action):
        # 1. Encode shared context
        shared_context = self.context_encoder(instruction, obs_history, state)
        
        # 2. Prepare branch-specific tokens
        batch_size = shared_context.size(0)
        prior_q = self.prior_queries.expand(batch_size, -1, -1)
        
        with torch.no_grad():
            # Extract future embeddings for the posterior branch
            posterior_q = self.future_encoder(future_obs) 
            
        # 3. Pack sequence: [Context, Prior_Q, Posterior_Q]
        packed_sequence = torch.cat([shared_context, prior_q, posterior_q], dim=1)
        
        # 4. Create Dual-Branch Attention Mask
        # Context is visible to all. Prior and Posterior cannot see each other.
        attn_mask = self.create_dual_branch_mask(shared_context.size(1), self.num_queries)
        
        # 5. Transformer Forward Pass (extracting hidden states at aligned layers)
        hidden_states = self.transformer(packed_sequence, mask=attn_mask, return_intermediates=True)
        
        # Extract aligned layer outputs
        prior_hiddens = hidden_states['prior']       # Shape: [L, B, num_queries, D]
        posterior_hiddens = hidden_states['posterior'] # Shape: [L, B, num_queries, D]
        
        # 6. Compute Alignment Loss (Frobenius norm)
        loss_align = F.mse_loss(prior_hiddens, posterior_hiddens)
        
        # 7. Compute Anti-collapse Regularization (Norm & Rank)
        loss_reg = self.compute_regularization(prior_hiddens, posterior_hiddens)
        
        # 8. Action Decoding (Flow-matching)
        prior_action_pred = self.action_head(prior_hiddens[-1])
        posterior_action_pred = self.action_head(posterior_hiddens[-1])
        
        loss_fm_prior = self.flow_matching_loss(prior_action_pred, target_action)
        loss_fm_post = self.flow_matching_loss(posterior_action_pred, target_action)
        
        total_loss = loss_fm_prior + loss_fm_post + 1e-3 * loss_align + 1e-4 * loss_reg
        return total_loss

    def inference(self, instruction, obs_history, state):
        # Only execute the prior branch during deployment
        shared_context = self.context_encoder(instruction, obs_history, state)
        batch_size = shared_context.size(0)
        prior_q = self.prior_queries.expand(batch_size, -1, -1)
        
        packed_sequence = torch.cat([shared_context, prior_q], dim=1)
        hidden_states = self.transformer(packed_sequence)
        
        action_pred = self.action_head(hidden_states['prior'][-1])
        return action_pred
```

### 第二模块：核心架构解释

本文并非提出单一模型，而是对现有世界模型与机器人策略的耦合方式进行了高维度的架构解构。其核心逻辑可以分为三种主要范式：

1. **逆动力学解耦范式 (IDM-style)**：世界模型与策略模型物理分离。世界模型先根据当前观测和指令生成未来观测（如视频），策略模型（逆动力学模块）再根据当前和未来观测推导具体动作。
2. **统一骨干网络范式 (Unified/Single-backbone)**：观测、动作和未来预测在同一个多模态生成模型（如 Diffusion 或 Transformer）中联合建模。未来状态预测作为内部表征或辅助目标，直接指导动作生成。
3. **模拟器范式 (Simulator)**：世界模型作为独立的交互环境。策略在其中进行闭环的动作展开（Rollout），世界模型返回下一状态和奖励信号，用于强化学习（RL）或测试时搜索评估（MPC）。

以下是这三种核心范式的底层逻辑伪代码实现：

```python
class WorldModelSurveyParadigms:
    def __init__(self, world_model, policy_model, inverse_dynamics):
        self.wm = world_model
        self.policy = policy_model
        self.idm = inverse_dynamics

    def decoupled_idm_inference(self, obs_t, instruction):
        # Paradigm 1: Predict first, then act (Decoupled)
        # 1. World model predicts future observations
        future_obs = self.wm.predict_future(obs_t, instruction)
        # 2. Inverse dynamics extracts action from current and future obs
        action_t = self.idm.predict_action(obs_t, future_obs)
        return action_t

    def unified_backbone_inference(self, obs_t, instruction):
        # Paradigm 2: Joint modeling in a single backbone
        # The model internally processes future prediction and action generation
        # Future visual branch might be marginalized or truncated during inference
        joint_representation = self.policy.encode(obs_t, instruction)
        action_t, _ = self.policy.decode_jointly(joint_representation)
        return action_t

    def simulator_rl_training_loop(self, initial_obs, instruction, steps):
        # Paradigm 3: World Model as a Learned Simulator for RL
        obs_t = initial_obs
        trajectory = []
        
        for t in range(steps):
            # Policy proposes an action
            action_t = self.policy.select_action(obs_t, instruction)
            
            # World model simulates the environment transition and reward
            next_obs_imagined, reward_imagined = self.wm.step(obs_t, action_t)
            
            trajectory.append((obs_t, action_t, reward_imagined, next_obs_imagined))
            obs_t = next_obs_imagined
            
        # Update policy using imagined trajectory (e.g., via GRPO or PPO)
        self.policy.update_from_imagination(trajectory)
```

### 第二模块：核心架构解释

Being-H0.7 的核心架构基于混合专家 Transformer（MoT），其本质是一个在单次前向传播中同时处理先验（部署）和后验（训练）计算流的双分支网络。

**1. 输入与上下文编码**
模型首先接收多模态上下文，包括语言指令、历史观测图像序列和机器人本体状态。这些输入被编码为共享的上下文 Token 序列。

**2. 隐式推理空间构建**
在共享上下文和动作 Token 之间，模型插入了固定数量的隐式查询（Latent Queries）。
- **先验分支（Prior Branch）**：使用一组可学习的参数化 Queries。
- **后验分支（Posterior Branch）**：使用经过冻结视觉编码器（V-JEPA）提取的未来观测图像特征（Future Embeddings）。

**3. 双分支注意力掩码（Dual-Branch Attention Mask）**
为了在同一个 Transformer 序列中并行训练两个分支且防止信息穿越，模型设计了特定的注意力掩码：共享上下文对所有 Token 可见；先验分支的 Token 与后验分支的 Token 相互不可见。

**4. 联合对齐与正则化（Joint Alignment & Regularization）**
在 Transformer 的深层，提取先验分支和后验分支对应位置的隐状态，计算 Frobenius 范数损失进行强制对齐。为防止隐空间特征坍塌，引入了基于特征幅值的范数正则化（Norm Preservation）和基于特征值分布的秩正则化（Spectral Diversity）。

**5. 动作解码（Action Decoding）**
两个分支的隐状态最终分别输入到基于 Flow-matching 的动作解码器中，预测目标动作轨迹。推理时，仅执行先验分支。

```python
import torch
import torch.nn as nn
import torch.nn.functional as F

class BeingH07(nn.Module):
    def __init__(self, context_encoder, future_encoder, transformer_backbone, action_head):
        super().__init__()
        self.context_encoder = context_encoder
        self.future_encoder = future_encoder # Frozen V-JEPA
        self.transformer = transformer_backbone
        self.action_head = action_head
        
        self.num_queries = 16
        self.hidden_dim = 1024
        # Learnable latent queries for the prior branch
        self.prior_queries = nn.Parameter(torch.randn(1, self.num_queries, self.hidden_dim))
        
    def forward(self, instruction, obs_history, state, future_obs, target_action):
        # 1. Encode shared context
        shared_context = self.context_encoder(instruction, obs_history, state)
        
        # 2. Prepare branch-specific tokens
        batch_size = shared_context.size(0)
        prior_q = self.prior_queries.expand(batch_size, -1, -1)
        
        with torch.no_grad():
            # Extract future embeddings for the posterior branch
            posterior_q = self.future_encoder(future_obs) 
            
        # 3. Pack sequence: [Context, Prior_Q, Posterior_Q]
        packed_sequence = torch.cat([shared_context, prior_q, posterior_q], dim=1)
        
        # 4. Create Dual-Branch Attention Mask
        # Context is visible to all. Prior and Posterior cannot see each other.
        attn_mask = self.create_dual_branch_mask(shared_context.size(1), self.num_queries)
        
        # 5. Transformer Forward Pass (extracting hidden states at aligned layers)
        hidden_states = self.transformer(packed_sequence, mask=attn_mask, return_intermediates=True)
        
        # Extract aligned layer outputs
        prior_hiddens = hidden_states['prior']       # Shape: [L, B, num_queries, D]
        posterior_hiddens = hidden_states['posterior'] # Shape: [L, B, num_queries, D]
        
        # 6. Compute Alignment Loss (Frobenius norm)
        loss_align = F.mse_loss(prior_hiddens, posterior_hiddens)
        
        # 7. Compute Anti-collapse Regularization (Norm & Rank)
        loss_reg = self.compute_regularization(prior_hiddens, posterior_hiddens)
        
        # 8. Action Decoding (Flow-matching)
        prior_action_pred = self.action_head(prior_hiddens[-1])
        posterior_action_pred = self.action_head(posterior_hiddens[-1])
        
        loss_fm_prior = self.flow_matching_loss(prior_action_pred, target_action)
        loss_fm_post = self.flow_matching_loss(posterior_action_pred, target_action)
        
        total_loss = loss_fm_prior + loss_fm_post + 1e-3 * loss_align + 1e-4 * loss_reg
        return total_loss

    def inference(self, instruction, obs_history, state):
        # Only execute the prior branch during deployment
        shared_context = self.context_encoder(instruction, obs_history, state)
        batch_size = shared_context.size(0)
        prior_q = self.prior_queries.expand(batch_size, -1, -1)
        
        packed_sequence = torch.cat([shared_context, prior_q], dim=1)
        hidden_states = self.transformer(packed_sequence)
        
        action_pred = self.action_head(hidden_states['prior'][-1])
        return action_pred
```

### 第二模块：核心架构解释

Hamiltonian World Models (HWM) 的核心思想是将物理演化与视觉渲染解耦，用物理方程作为生成过程的骨架。其完整架构包含四个阶段：

1. **感知与编码 (Perception and Encoding)**：将历史观测 $o_{\le t}$ 映射为结构化的实体级相空间隐状态 $z_t = \{(q_t^i, p_t^i)\}_{i=1}^N$，其中 $q$ 为广义坐标，$p$ 为广义动量。
2. **哈密顿动力学演化 (Hamiltonian Dynamics)**：这是模型的核心。定义一个参数化的哈密顿能量函数 $H_\phi(z_t)$，包含动能、势能和交互能。状态的演化不再是黑盒映射，而是遵循扩展的哈密顿正则方程：包含保守的哈密顿流、耗散项（摩擦/阻尼）、动作控制项以及残差项（处理接触突变等未建模动态）。
3. **视频生成 (Video Generation)**：将预测出的未来隐状态轨迹 $\hat{z}_{t+1:t+H}$ 输入生成式解码器，渲染出未来的视觉观测 $\hat{o}_{t+1:t+H}$。
4. **规划与决策 (Planning and Decision)**：在物理可靠的隐空间中展开多条候选动作序列，通过效用函数评估并选择最优动作。

```python
import torch
import torch.nn as nn

class HamiltonianWorldModel(nn.Module):
    def __init__(self):
        super().__init__()
        self.encoder = PhaseSpaceEncoder() # 编码器：提取广义坐标 q 和动量 p
        self.decoder = VideoDecoder()      # 解码器：从相空间渲染像素
        
        # 能量函数网络
        self.kinetic_net = KineticEnergyNet()
        self.potential_net = PotentialEnergyNet()
        self.interaction_net = InteractionEnergyNet()
        
        # 真实物理扩展网络
        self.dissipation_net = DissipationMatrixNet() # R_psi: 耗散矩阵 (半正定)
        self.control_net = ControlMatrixNet()         # G_psi: 动作控制矩阵
        self.residual_net = ResidualDynamicsNet()     # epsilon_psi: 残差动态
        
    def hamiltonian(self, q, p):
        # H = K(p) + U(q) + V(q, q)
        K = self.kinetic_net(p)
        U = self.potential_net(q)
        V = self.interaction_net(q)
        return K + U + V
        
    def latent_dynamics_step(self, z, action):
        q, p = z.chunk(2, dim=-1)
        
        # 开启梯度追踪以计算哈密顿方程的偏导数
        q.requires_grad_(True)
        p.requires_grad_(True)
        
        H = self.hamiltonian(q, p)
        
        # 计算梯度: dH/dq, dH/dp
        dH_dq = torch.autograd.grad(H, q, create_graph=True)[0]
        dH_dp = torch.autograd.grad(H, p, create_graph=True)[0]
        grad_H = torch.cat([dH_dq, dH_dp], dim=-1)
        
        # 辛矩阵 J (Symplectic matrix)
        J = self.get_symplectic_matrix()
        
        # 扩展的受控-耗散哈密顿演化方程 (Eq. 19)
        # dz/dt = (J - R) * grad_H + G * action + epsilon
        R = self.dissipation_net(z)
        G = self.control_net(z)
        epsilon = self.residual_net(z, action)
        
        dz_dt = torch.matmul(J - R, grad_H) + torch.matmul(G, action) + epsilon
        
        # 使用数值积分器（如辛欧拉法或RK4）更新状态
        z_next = z + dz_dt * self.dt 
        return z_next

    def rollout(self, obs_history, actions):
        # 1. 感知
        z_t = self.encoder(obs_history)
        
        # 2. 动力学演化
        z_traj = [z_t]
        for a in actions:
            z_next = self.latent_dynamics_step(z_traj[-1], a)
            z_traj.append(z_next)
            
        # 3. 渲染
        pred_obs = self.decoder(torch.stack(z_traj))
        return pred_obs
```

2. 设计了适用于具身智能的受控-耗散哈密顿隐空间动力学方程，使纯理论的保守物理系统能够兼容动作输入、摩擦耗散及残差物理现象。
3. 明确界定并批判了现有三条世界模型技术路线（2D视频、3D场景、JEPA隐变量）在具身决策上的局限性，指明了向“物理原生（Physically Native）”演进的下一代发展方向。

Q5: 这项工作在哪些相关研究的基础上做了推进？（列举1-2个最核心的前置工作即可）

A5:
1. **Hamiltonian Neural Networks (HNN) / Hamiltonian Generative Networks (HGN)** (Greydanus et al., 2019; Toth et al., 2020)：证明了可以从数据或高维像素中学习哈密顿能量守恒动力学，本文将其扩展到了世界模型的宏观架构中。
2. **Symplectic ODE-Net** (Zhong et al., 2020)：为哈密顿动力学引入了控制输入，直接支撑了本文受控动力学（Control Matrix）的数学设计。

Q6: 客观评价：这篇论文的局限性、漏洞或"未讲明的故事"是什么？对后续科研有何启发？

A6: 局限性在于极度的“纸上谈兵”。从高维非结构化像素中端到端地学习并解耦出具有真实物理意义的广义坐标和动量 $(q, p)$ 极其困难，极易退化为普通的隐变量；此外，真实世界中频繁的非弹性碰撞、接触切换、软体形变等非平滑离散事件，很难用连续的哈密顿流方程来稳定建模，论文中的残差项 $\epsilon$ 在实际训练中极易“喧宾夺主”掩盖物理主干。
启发：未来的具身世界模型不应仅仅是“视频生成器”，而必须是“物理模拟器”。在现有的 Diffusion 或 Transformer 架构中，放弃纯数据驱动的暴力美学，转而注入物理微分方程级别的硬约束（Hard Constraints），将是解决长程预测漂移（Compounding Errors）极具潜力的破局点。
