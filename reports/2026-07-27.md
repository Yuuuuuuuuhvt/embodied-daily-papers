```python
class ControlledJEPA(nn.Module):

        super().__init__()
        # 非线性编码器：将高维观测映射到潜在空间

            nn.Linear(obs_dim, 256),
            nn.ReLU(),
            nn.Linear(256, latent_dim)
        )
        # 动作条件预测器：受控转移模型
        self.predictor = nn.Sequential(
            nn.Linear(latent_dim + action_dim, 256),
            nn.ReLU(),
            nn.Linear(256, latent_dim)
        )

    def forward(self, obs_t, action_t, obs_t_plus_1):

        z_t = self.encoder(obs_t)

        with torch.no_grad():

        return loss

    # 理论指出：如果动作变化非退化，预测器能准确捕捉反事实动作对状态的影响

    pred_states = predictor(torch.cat([learned_states, counterfactual_actions], dim=-1))

    aligned_true_next = true_next_states @ transformation_matrix

    counterfactual_error = compute_error(pred_states, aligned_true_next)

    return rep_error, counterfactual_error
```

### 第二模块：核心架构解释

Pigey 的核心架构是一个纯推理期的闭环控制流，完全不需要收集新的机器人数据或对现有模型进行微调。其运行流程可以拆解为以下几个关键模块：

1. 观察与基础定位 (Perceive & Grounding)：编排器首先调用感知工具，获取腕部相机图像、机器人状态（末端执行器位姿、夹爪开合度）以及开放词汇目标检测器返回的物体标签。这些标签构成了编排器后续操作的严格词汇表。
2. 规划与路由 (Plan & Route)：编排器将自然语言指令拆解为具体的子目标，并根据物体属性选择底层执行器。对于刚性、放置在桌面上的物体，路由给 TAMP（任务与运动规划）后端以获取几何精度；对于可变形物体、线缆或容器内的物体，直接路由给 $\pi_{0.5}$ VLA 后端。
3. 闭环验证 (Verification)：这是区别于开环 VLA 的核心。执行器完成动作后，编排器通过确定性信号（夹爪宽度传感器判断是否抓取成功）和视觉信号（腕部相机确认目标是否在夹爪中）进行双重验证。只有验证成功的动作才会推进到下一步（如放置）。
4. 错误恢复与升级 (Recovery & Escalation)：如果验证失败，编排器会触发重试机制（重新感知并规划）。如果 TAMP 连续失败，系统会触发双向降级机制，将任务升级交由 VLA 处理，反之亦然。

```python
import torch.nn as nn

class ControlledJEPA(nn.Module):
    def __init__(self, obs_dim, action_dim, latent_dim):
        super().__init__()
        # 非线性编码器：将高维观测映射到潜在空间
        self.encoder = nn.Sequential(
            nn.Linear(obs_dim, 256),
            nn.ReLU(),
            nn.Linear(256, latent_dim)
        )
        # 动作条件预测器：受控转移模型
        self.predictor = nn.Sequential(
            nn.Linear(latent_dim + action_dim, 256),
            nn.ReLU(),
            nn.Linear(256, latent_dim)
        )

    def forward(self, obs_t, action_t, obs_t_plus_1):
        # 1. 编码当前状态
        z_t = self.encoder(obs_t)
        
        # 2. 编码下一时刻目标状态 (实际中常使用 Stop-gradient 或 EMA 动量编码器)
        with torch.no_grad():
            z_t_plus_1_target = self.encoder(obs_t_plus_1)

        # 3. 动作条件预测
        z_t_plus_1_pred = self.predictor(torch.cat([z_t, action_t], dim=-1))

        # 4. 计算 JEPA 预测损失
        loss = nn.functional.mse_loss(z_t_plus_1_pred, z_t_plus_1_target)
        return loss

def evaluate_identifiability(encoder, predictor, true_states, true_dynamics, actions):
    """
    验证理论中的可识别性指标
    """
    # 提取学到的表征
    learned_states = encoder(true_states)

    # 检查表征可识别性 (Representation Identifiability)
    # 理论指出：如果满足谱分离，learned_states 应该是 true_states 的正交变换

    rep_error = compute_alignment_error(learned_states, true_states @ transformation_matrix)

    # 检查转移可识别性 (Transition Identifiability)
    # 理论指出：如果动作变化非退化，预测器能准确捕捉反事实动作对状态的影响
    counterfactual_actions = generate_weakly_excited_actions(actions)
    pred_states = predictor(torch.cat([learned_states, counterfactual_actions], dim=-1))
    
    true_next_states = true_dynamics(true_states, counterfactual_actions)
    aligned_true_next = true_next_states @ transformation_matrix

    counterfactual_error = compute_error(pred_states, aligned_true_next)

    return rep_error, counterfactual_error
```

```python
        self.tac_dec = tactile_decoder

        pred_vis = self.vis_dec(z_future)
```

# 第二模块：核心架构解释

本文的核心架构分为三个关键部分：状态发现、会话管理与缓存驱逐。

**1. 状态发现机制 (PCS Discovery via Fingerprint)**
系统不需要开发者手动 hardcode 需要保存哪些张量，而是通过一个自动化探针程序（Fingerprint）来测量。它通过“移除某部分状态并观察生成结果是否退化”来判断该状态是否是“不可重计算的必需品”。最终将模型状态划分为：必须保存的 PCS（如 RNG、Memory Bank）和可丢弃的 Derivable 状态（如可通过重新推理得到的激活值）。

**2. 会话中心化运行时 (Session-Centric Runtime)**
摒弃了 LLM 中“请求结束即销毁状态”的做法。GPU 采用时间复用（Time-multiplexed）机制，当前活跃的会话独占 GPU，而其他挂起会话的 PCS 被极速拷贝并持久化到 Host（CPU）内存中。由于 PCS 体积极小（如 Cosmos3 仅 1.38MB，WorldMem 仅 185KB），Host 可以轻松容纳海量会话。

**3. 相关性驱逐策略 (Relevance-keyed Eviction)**
当 Host 内存面临压力时，必须丢弃部分记忆。LLM 普遍采用基于近期（Recency-keyed，如 LRU）的驱逐策略，但世界模型需要回到“锚点（Anchor）”视角。因此，系统采用基于相关性（Relevance-keyed）的策略，优先保留与初始/关键视角相关的帧，丢弃中间游荡过程的帧。

```python
class GenerativeWorldModelRuntime:
    def __init__(self, model, tolerance=1e-5):
        self.model = model
        self.tolerance = tolerance
        # 1. 自动发现最小持久化状态 (PCS)
        self.pcs_components = self.fingerprint_model()
        self.sessions = {} # 存储在 Host 内存中的会话状态

    def fingerprint_model(self):
        """核心创新：通过测量而非假设来发现 PCS"""
        full_state = self.model.get_all_runtime_buffers()
        oracle_output = self.model.run_leave_and_return_probe(full_state)
        
        necessary_state = set()
        # 逐个剔除组件，测试是否导致结果退化
        for component in full_state:
            ablated_state = full_state - {component}
            test_output = self.model.run_leave_and_return_probe(ablated_state)
            
            # 如果剔除后结果偏离 Oracle，说明该组件不可重计算，必须保留
            if distance(test_output, oracle_output) > self.tolerance:
                necessary_state.add(component)
                
        return necessary_state

    def checkpoint(self, session_id):
        """极低开销的状态快照 (0.012 ms)"""
        current_state = self.model.get_buffers()
        # 仅提取 PCS，丢弃可重计算的冗余状态
        pcs_only = {k: v for k, v in current_state.items() if k in self.pcs_components}
        self.sessions[session_id] = copy_to_host(pcs_only)

    def restore(self, session_id):
        """字节级完美恢复"""
        pcs_state = copy_to_device(self.sessions[session_id])
        self.model.load_buffers(pcs_state)
        # 恢复后，模型可以无缝继续生成，仿佛从未中断

    def evict_memory_under_pressure(self, session_id, budget):
        """反 LLM 直觉的驱逐策略：保留相关性(Anchor)，丢弃近期(Recency)"""
        bank = self.sessions[session_id]['memory_bank']
        if size(bank) > budget:
            # 错误做法 (LLM LRU): bank = bank[-budget:] 
            # 正确做法 (World Model): 保留初始锚点帧，丢弃中间帧
            self.sessions[session_id]['memory_bank'] = keep_anchor_frames(bank, budget)
```

### 第二模块：核心架构解释

本文并非提出新模型，而是基于标准的 GRU 残差世界模型（类似 RSSM 的确定性部分）进行机制诊断。其核心流程分为训练期的**一致性约束施加**与评估期的**几何扩张率测量**。

1. **基础转移架构**: 采用卷积编码器提取潜在状态 $z_t$，通过 GRU 残差网络进行状态转移预测 $\hat{z}_{t+1} = z_t + \alpha \tanh(\Delta z_t)$。对于控制任务，转移过程额外接入动作 $a_t$。
2. **多步一致性损失**: 在标准的重建损失和单步预测损失之外，引入多步潜在一致性损失 $\mathcal{L}_{cons}$。模型在训练时展开 $K$ 步（如1, 3, 5, 10, 15, 20），计算自回归预测的潜在状态 $\hat{z}_{t+k}$ 与 Teacher-forced（由真实观测编码得到）的潜在状态 $z^{tf}_{t+k}$ 之间的 L2 距离，并由超参数 $\lambda$ 控制权重。
3. **局部扩张代理指标 ($L_{20, q95}$)**: 冻结训练好的模型，在验证集上采样潜在状态对 $(z_i, z_j)$。让这两个状态在模型中自由 rollout 20 步，计算它们在第 20 步的距离与初始距离的比值。取所有采样对中该比值的 95% 分位数作为 $L_{20, q95}$。若该值小于1，说明模型在局部流形上呈现收缩特性。
4. **随机强迫注入 (Stochastic Forcing)**: 为了探究为何控制任务无法收缩，在被动视频的转移方程中人为注入高斯噪声 $z \leftarrow f(z) + \eta \sigma^* \epsilon / \sqrt{d}$，强制模拟控制任务中的动作扰动，从而拟合出扩张率随噪声水平 $\eta$ 变化的线性定律。

```python
# 伪代码：世界模型一致性训练与几何扩张率评估

def train_step(batch, encoder, transition_model, lambda_weight, K_horizons, eta_noise=0.0):
    # 1. 获取真实观测并编码为 Teacher-forced 潜在状态
    obs = batch['observations'] # shape: [batch_size, seq_len, C, H, W]
    z_tf = encoder(obs)         # shape: [batch_size, seq_len, latent_dim]
    
    loss_recon = compute_reconstruction_loss(z_tf, obs)
    loss_1step = compute_one_step_loss(z_tf, transition_model)
    loss_cons = 0.0
    
    # 2. 计算多步潜在一致性损失
    for t in range(seq_len - max(K_horizons)):
        z_hat = z_tf[:, t]
        for k in range(1, max(K_horizons) + 1):
            # 注入随机强迫噪声（用于诊断实验）
            if eta_noise > 0:
                noise = torch.randn_like(z_hat) * (eta_noise * 0.5 / sqrt(latent_dim))
                z_hat = z_hat + noise
                
            z_hat = transition_model(z_hat, batch['actions'][:, t+k-1])
            
            if k in K_horizons:
                # 计算预测状态与真实编码状态的 L2 距离
                loss_cons += torch.mean((z_hat - z_tf[:, t+k]) ** 2)
                
    total_loss = loss_recon + loss_1step + lambda_weight * loss_cons
    total_loss.backward()
    return total_loss

def evaluate_geometry(val_pool, transition_model, num_pairs=240, k_steps=20):
    # 1. 从验证集中采样潜在状态对
    z_i, z_j = sample_pairs(val_pool, num_pairs)
    initial_dist = torch.norm(z_i - z_j, dim=-1)
    
    # 2. Rollout k 步
    z_hat_i, z_hat_j = z_i, z_j
    for _ in range(k_steps):
        z_hat_i = transition_model(z_hat_i)
        z_hat_j = transition_model(z_hat_j)
        
    final_dist = torch.norm(z_hat_i - z_hat_j, dim=-1)
    
    # 3. 计算距离比值并取 95% 分位数
    ratios = final_dist / (initial_dist + 1e-8)
    L20_q95 = torch.quantile(ratios, 0.95)
    
    return L20_q95
```

### 第二模块：核心架构解释

本论文的框架分为两个完全解耦的阶段：**离线无目标动力学学习**与**部署期可观测对齐规划**。

**1. 离线动力学学习 (Offline Goal-Agnostic Training)**
模型使用轻量级 2D ViT 作为编码器，接收被随机掩码的 PDE 观测场。编码器提取潜在状态后，一个动作条件预测器（Action-conditioned Predictor）根据历史潜在状态和输入的动作序列，自回归地预测未来的潜在状态。
训练过程完全没有奖励信号，也不解码回像素/物理场。为了防止潜在空间坍缩，使用了 VICReg（方差-协方差正则化）、时间拉直惩罚（Temporal Straightening）以及 Delta-JEPA 风格的动作解码辅助损失。

**2. 可观测对齐探针 (Observable-Aligned Probe)**
在冻结上述训练好的编码器和预测器后，作者在预测出的潜在轨迹上训练了一个极简单的线性岭回归探针（Ridge Map），用于从潜在向量中直接预测系统的动能（Kinetic Energy, KE）。

**3. 部署期控制规划 (Deployment-time MPPI Planning)**
使用模型预测路径积分（MPPI）控制器。MPPI 生成大量候选动作序列，冻结的预测器在潜在空间中展开这些动作的未来状态。此时，**不计算**预测潜在状态与目标潜在状态的 $L^2$ 距离，而是用探针将预测潜在状态转换为动能（KE），并计算其与目标动能的误差，以此作为代价函数选出最优动作。

**模型架构与实验流程**

MissionBench 并非提出一个新的网络架构，而是构建了一个针对通用 MLLM 的零样本闭环评估框架。其核心流程如下：

2. **闭环交互循环**：

   - **执行 (Execution)**：仿真器执行该离散动作，更新环境状态，并进入下一轮循环。
3. **评估指标**：通过任务成功率（Success Rate）、路径效率等指标，评估模型在没有特定领域微调情况下的任务级胜任力。

```python
        state, reward, done, info = self.env.execute(action)
        self.action_history.append(action)

def evaluate_mllm_agent(mllm, env, max_steps=50):

    done = False
    step_count = 0

    while not done and step_count < max_steps:

        # 环境步进与状态更新
        next_obs, done = env.step(action)
        obs = next_obs
        step_count += 1
```

### 第二模块：核心架构解释

TRACE-RealWorld (TRW) 的架构位于底层记录存储与顶层事务规划器之间，其核心流程分为四个阶段：

1. 状态声明与语义提升 (Semantic Elevation)：世界模型输出带有不确定性的预测状态。TRW 将这些原始信号转化为类型化的“声明”（Claims，例如“E12航道可通行”），并附带校准后的置信度、支持度（Support）和分布外（OOD）分数。
2. 自适应刷新与门控 (Adaptive Refresh & Gate)：当系统准备基于某项声明执行不可逆动作时，门控机制会检查声明的新鲜度与有效性。刷新决策基于信息价值（Value of Information, VoI）：只有当潜在的传感器观测结果足以跨越决策边界（即改变当前最优动作）且能降低预期损失时，系统才会花钱调用传感器。
3. 承诺执行与依赖追踪 (Commitment & Dependency)：一旦门控放行，动作被执行并形成“承诺”。所有的承诺构成一个有向无环图（DAG）。
4. 事务补偿与审计 (Saga Compensation & Audit)：如果后续的观测证明先前的声明已失效（超出容差），TRW 会触发失效检测。系统沿着依赖图计算受影响的集合，并调用 Saga 模式的补偿事务（如重新规划路线、释放资源），而不会全局重置。所有状态流转均写入仅追加（Append-only）的日志中。

```python
class TRACERealWorld:
    def __init__(self, world_model, record_store, planner):
        self.world_model = world_model
        self.record_store = record_store
        self.planner = planner
        self.dependency_graph = DAG()

    def evaluate_commitment(self, action, claim, channels):
        # 1. 检查声明的硬性约束 (OOD, Support, 基础新鲜度)
        if not self._check_gate_conditions(claim):
            return "HOLD"

        # 2. 自适应刷新逻辑 (基于信息价值 VoI)
        best_channel = None
        max_net_value = 0
        
        for channel in channels:
            # 计算观测结果改变当前最优决策的预期损失减少量
            expected_loss_reduction = self._calculate_voi(action, claim, channel)
            net_value = expected_loss_reduction - channel.cost
            
            if net_value > max_net_value and self._is_adequate(channel):
                max_net_value = net_value
                best_channel = channel

        # 3. 决定是否需要获取新证据
        if best_channel is not None:
            new_evidence = best_channel.observe()
            claim = self.world_model.update_belief(claim, new_evidence)
            self.record_store.append(RevisionRecord(claim, new_evidence))
            
            # 重新评估更新后的声明
            if not self._check_gate_conditions(claim):
                return "ESCALATE"

        # 4. 授权执行并记录依赖
        self.record_store.append(CommitmentRecord(action, claim))
        self.dependency_graph.add_node(action, depends_on=claim)
        return "CLEAR"

    def handle_invalidation(self, invalidated_claim):
        # 当被动监控或后续刷新发现声明失效时触发
        self.record_store.append(InvalidationRecord(invalidated_claim))
        
        # 计算依赖范围内的受影响集合
        affected_commitments = self.dependency_graph.get_descendants(invalidated_claim)
        
        # 执行局部 Saga 补偿
        for commitment in reversed(affected_commitments):
            self.planner.execute_compensation(commitment.compensating_transaction)
            self.record_store.append(CompensationRecord(commitment))
```

### 第二模块：核心架构解释

CommandLM 的架构是一个典型的“感知-压缩-推理”三阶段多模态流水线，其核心在于如何将高维的时空物理世界降维成 LLM 能理解的离散语义。

**1. 多模态传感器融合 (BEVFusion)**
模型首先冻结了一个预训练的 BEVFusion 模块。多视角相机的 RGB 图像通过 Swin Transformer 提取特征并进行深度估计投影到 BEV 空间；激光雷达点云直接体素化投影。两者在 BEV 空间拼接，形成包含几何与语义的密集特征图。

**2. 时序上下文编码 (Temporal Buffer)**
摒弃单帧输入，模型维护一个滚动缓冲区（Rolling Buffer），以 12Hz 的频率堆叠过去 2 秒的 BEV 特征图。这赋予了模型短期的物理记忆，使其能够感知车辆的运动趋势和车道变化。

**3. 时空压缩瓶颈 (Q-Former)**
这是架构的核心信息漏斗。堆叠后的庞大时空 BEV 张量被送入 Q-Former（包含 32 个可学习的 Query）。通过交叉注意力机制，这 32 个 Query 从时空特征中提取最关键的场景元素，将物理信号压缩为固定长度的视觉 Token。

**4. 视觉-语言对齐与生成 (LLM Integration)**
压缩后的 Token 通过一个线性适配器（Adapter Up）映射到 Qwen-2.5 14B 的输入维度，并被 `<vision_start>` 和 `<vision_end>` 特殊字符包裹。LLM 结合文本提示（Prompt），以自回归的方式输出行为级描述。

```python
        self.q_former = QFormer(num_queries=32, hidden_size=768)
        self.adapter_up = Linear(768, 5120)

    def forward(self, multi_cameras, lidar_point_cloud, text_prompt):
        # 1. 提取当前帧 BEV 特征
        current_bev = self.bev_fusion(multi_cameras, lidar_point_cloud)
        
        # 2. 时序堆叠 (Shape: [24, C, H, W])
        self.temporal_buffer.append(current_bev)
        stacked_bev = self.temporal_buffer.get_stacked_tensor()
        
        # 3. 降维与 Q-Former 压缩
        bev_features = self.adapter_down(stacked_bev.flatten(temporal_dim))
        vision_tokens = self.q_former(bev_features) # Shape: [32, 768]
        
        # 4. 映射到 LLM 维度
        llm_vision_inputs = self.adapter_up(vision_tokens) # Shape: [32, 5120]
        
        # 5. 拼接 Prompt 并自回归生成
        input_embeds = concat([
            self.llm.embed("<vision_start>"),
            llm_vision_inputs,
            self.llm.embed("<vision_end>"),
            self.llm.embed(text_prompt)
        ])
        
        behavior_caption = self.llm.generate(input_embeds)
        return behavior_caption
```

### 第三模块：核心学术问答

Q1: 这篇论文试图解决什么核心问题？

A1: 解决端到端自动驾驶大模型在真实世界部署时面临的“黑盒”问题。现有模型要么直接输出控制信号缺乏可解释性，要么在处理包含大量绝对坐标的视觉问答时因 LLM 的数值不敏感性而产生严重幻觉，且计算开销过大无法在车端运行。

Q2: 作者提出的核心解决方案（创新点）是什么？

A2: 核心创新在于将“时序 BEV 融合”与“去坐标化的语义提示”相结合。在架构上，利用 Q-Former 将 2 秒的连续 BEV 特征压缩为极少量的视觉 Token；在数据上，将传统的像素/坐标级标注替换为抽象的对象级标签（如 `<|object_ref_start|>obj1`），迫使模型进行纯语义的意图推理而非几何计算。

Q3: 实验设计的关键支撑（或巧妙之处/数据集特点）是什么？

A3: 关键支撑是其自建的 CommandLM-nuScenes 数据集。作者极其敏锐地察觉到 LLM 处理坐标的劣势，因此对开源的 DriveLM 数据集进行了“清洗”，去除了所有绝对坐标。随后利用 72B 的大模型作为教师，基于场景摘要反向生成了 4 万多条专注于“行为探索”的问答对，极大地丰富了模型在复杂交互场景下的语言监督信号。

Q4: 这篇论文最显著的贡献点（Top 3）是什么？

A4:
1. 提出了一种融合多模态时序 BEV 特征与量化 LLM 的轻量化架构，验证了 Q-Former 在时空特征压缩上的可行性。
2. 开发了一套去坐标化、以对象为中心的数据集增强流水线，显著降低了 LLM 产生空间幻觉的概率。
3. 确立了“行为级描述（Behavior-level captioning）”作为自动驾驶感知与底层控制之间的一种有效中间态接口。

Q5: 这项工作在哪些相关研究的基础上做了推进？（列举1-2个最核心的前置工作即可）

A5:
1. BLIP-2：直接继承了其 Q-Former 架构作为视觉-语言的对齐瓶颈。
2. DriveLM：在其提供的图结构 VQA 数据集基础上，推进了去坐标化和行为级意图的扩展。

Q6: 客观评价：这篇论文的局限性、漏洞或"未讲明的故事"是什么？对后续科研有何启发？

A6: 最大的漏洞在于其**极度糟糕的实时性与空间保真度**。1.78 秒的推理延迟在自动驾驶中是不可接受的；同时，其 SPICE 得分极低（0.1337），且真实行为匹配率不到一半。这揭示了一个未讲明的物理本质：**Q-Former 的暴力压缩（将 24 帧高维 BEV 压成 32 个 Token）丢失了极其关键的细粒度空间与动力学信息**。对后续科研的启发是：在 VLA 架构中，不能简单套用传统 VLM 的 Q-Former 来处理高动态的物理世界数据。未来的研究必须探索非对称的注意力机制、流式架构（Streaming Architecture）或基于状态空间模型（如 Mamba）的记忆库，以在不爆炸 Token 数量的前提下，无损地保留时空演变细节。
