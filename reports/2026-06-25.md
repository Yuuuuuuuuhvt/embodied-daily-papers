import torch.nn as nn

class WOLFVLA(nn.Module):
    def __init__(self, groot_backbone):
        super().__init__()
        # 冻结视觉和语言编码器
        self.vision_encoder = groot_backbone.vision_encoder.requires_grad_(False)
        self.language_encoder = groot_backbone.language_encoder.requires_grad_(False)
        # 优化动作解码器 (基于流匹配的扩散模型)
        self.action_decoder = groot_backbone.action_decoder.requires_grad_(True)
        
    def forward(self, rgb_images, language_instruction, proprioception, target_action_chunk=None):
        # 1. 多模态特征提取
        v_tokens = self.vision_encoder(rgb_images)
        l_tokens = self.language_encoder(language_instruction)
        obs_features = concat(v_tokens, l_tokens, proprioception)
        
        if self.training:
            # 2. 训练阶段：流匹配 (Flow-Matching) 目标计算
            batch_size = target_action_chunk.shape[0]
            # 采样噪声和时间步
            noise = torch.randn_like(target_action_chunk)
            tau = torch.rand(batch_size, 1, 1).to(target_action_chunk.device)
            
            # 线性插值构建带噪动作
            noisy_action = tau * target_action_chunk + (1 - tau) * noise
            
            # 预测向量场
            predicted_vector_field = self.action_decoder(noisy_action, obs_features, tau)
            
            # 计算目标流方向
            target_vector_field = target_action_chunk - noise
            
            # 计算 MSE 损失
            loss = nn.MSELoss()(predicted_vector_field, target_vector_field)
            return loss
            
        else:
            # 3. 推理阶段：欧拉积分去噪生成动作块
            K_steps = 10
            dt = 1.0 / K_steps
            current_action = torch.randn(batch_size, horizon, action_dim) # a^0 ~ N(0, I)
            
            for k in range(K_steps):
                tau = torch.full((batch_size, 1, 1), k * dt)
                vector_field = self.action_decoder(current_action, obs_features, tau)
                current_action = current_action + dt * vector_field
                
            return current_action # 返回去噪后的动作块 (关节增量)

### 第二模块：核心架构解释

Causal-rCM 的核心架构是一个三阶段的流水线，辅以高度定制化的底层并行与算子基建。

**阶段一：因果教师模型准备 (Causal Teacher Preparation)**
将预训练的双向扩散模型（Bidirectional DiT）通过 Teacher-Forcing 转换为自回归因果模型。模型在训练时接收拼接的“干净历史帧”与“加噪目标帧”，通过因果掩码（Causal Mask）确保信息单向流动。

**阶段二：Teacher-Forcing 一致性蒸馏 (TF-CM)**
在固定历史上下文的前提下，对目标帧应用连续时间一致性模型（sCM/MeanFlow）。为了解决长视频序列计算雅可比向量积（JVP）时的显存爆炸问题，作者开发了支持自定义稀疏掩码的 FlashAttention-2 JVP 算子。此阶段为后续微调提供了极强且模式完整的初始化权重。

**阶段三：Self-Forcing 分布匹配蒸馏 (SF-DMD)**
模型切换到真实的自回归推理模式。利用 KV Cache 逐块（Chunk）生成视频。生成完成后，对最终输出的视频块应用 DMD 损失（通过一个预训练的 Fake-score 网络与 Teacher 网络计算得分差）。梯度仅通过每个 Chunk 的最后一次去噪步骤进行反向传播。

整个架构建立在 FSDP2（全分片数据并行）、Ulysses CP（上下文并行）以及选择性激活重计算（SAC）之上，确保了超长视频序列在多 GPU 集群上的可训练性。

        score_teacher = teacher_model(final_video)
        score_fake = fake_score_model(final_video)
    
    # DMD loss forces student distribution to match teacher distribution
    loss_dmd = mse_loss(final_video, score_fake - score_teacher)

### 第二模块：核心架构解释

ICWM 的核心思想是将 VLA 从静态的映射函数转变为动态的推理引擎。其工作流分为训练和测试两个阶段，核心在于如何构造和利用交互上下文（Interaction Context）$\mathcal{T}$。

**1. 训练阶段 (In-Context Training)**
在训练时，模型不仅接收当前观察 $o_t$ 和语言指令 $l$，还会被强制要求关注一段前置的交互上下文 $\mathcal{T}$。$\mathcal{T}$ 由 $N$ 个随机采样的交互片段组成，每个片段包含 $(o_i^s, a_i, o_i^e)$，即起始图像、执行的动作、结束图像。模型通过标准的行为克隆（BC）损失进行训练，被迫学会从 $\mathcal{T}$ 中提取当前视角的空间映射关系，以准确预测目标动作 $a_t$。

**2. 推理阶段 (Test-Time Active Probing and Inference)**
部署到新环境时，分为两步：
- **主动探测 (Active Probing)**：机器人不急于执行任务，而是先在安全工作区内执行 $N$ 次随机的试探性动作，记录下这 $N$ 次的视觉变化，构建出当前真实环境的上下文 $\mathcal{T}$。
- **上下文执行 (In-Context Execution)**：将收集到的 $\mathcal{T}$ 作为 Prompt 拼接到当前任务的观察和指令前，输入给 VLA 模型。Transformer 的注意力机制会利用 $\mathcal{T}$ 中的动态信息来校准当前观察，输出适应新环境的动作。

import torch
import torch.nn as nn

class ICWM_VLA(nn.Module):
    def __init__(self, vision_encoder, action_tokenizer, vlm_backbone):
        super().__init__()
        self.vision_encoder = vision_encoder
        self.action_tokenizer = action_tokenizer
        self.vlm_backbone = vlm_backbone # e.g., Qwen2.5-VL
        
    def encode_interaction_clip(self, img_start, action, img_end):
        """编码单个试探性交互片段"""
        v_start = self.vision_encoder(img_start)
        a_token = self.action_tokenizer.encode(action)
        v_end = self.vision_encoder(img_end)
        # 拼接形成单个 clip 的 token 序列
        return torch.cat([v_start, a_token, v_end], dim=1)

    def forward(self, context_clips, current_obs, instruction):
        """
        context_clips: List of (img_start, action, img_end) tuples, length N
        current_obs: 当前视角的图像
        instruction: 语言指令文本
        """
        # 1. 构建 Interaction Context T
        context_tokens = []
        for img_s, act, img_e in context_clips:
            clip_token = self.encode_interaction_clip(img_s, act, img_e)
            context_tokens.append(clip_token)
        T_tokens = torch.cat(context_tokens, dim=1) # 隐式包含系统配置 \psi
        
        # 2. 构建 Task Query
        obs_tokens = self.vision_encoder(current_obs)
        text_tokens = self.vlm_backbone.tokenize_text(instruction)
        query_tokens = torch.cat([obs_tokens, text_tokens], dim=1)
        
        # 3. In-Context Inference
        # 模型通过 Attention 机制，利用 T_tokens 中的动力学信息校准 query_tokens
        full_sequence = torch.cat([T_tokens, query_tokens], dim=1)
        action_logits = self.vlm_backbone(full_sequence)
        
        # 4. 解码动作
        predicted_action = self.action_tokenizer.decode(action_logits)
        return predicted_action

*   **核心有效数据:** 在真实世界 Franka 机器人（仅占训练数据 7.6% 的长尾数据）上，引入动作先验与历史压缩后，平均成功率从 35.0%（无先验基线）跃升至 68.0%。这直接证明了动作先验能够极大缓解数据稀缺带来的欠拟合，使策略在长尾分布上表现出极强的稳定性。
*   **存疑/需辩证看待的数据:** 论文图 6 关于蒸馏步数（$N_{decay}$）的消融实验显示，当蒸馏步数拉长至 30k 步（覆盖大部分训练周期）时，模型在 LIBERO 上的成功率反而跌至 63.1%，低于无先验基线（64.6%）。这表明，强加的动作隐空间虽然在早期能稳定梯度，但在后期会限制 VLM 结合视觉语义进行端到端微调的表征上限。先验只能作为初始化的引导，不能作为全局优化的硬约束。

### 第二模块：核心架构解释

本研究的架构分为完全解耦的两个阶段：纯动作先验学习阶段（Stage 1）与动作先验引导的 VLA 训练阶段（Stage 2）。

**Stage 1: 动作先验学习 (Action Prior Learning)**
目标是让动作模块在没有任何视觉和语言条件的情况下，仅通过本体感觉状态（Proprioceptive states）和动作（Actions）学习物理运动的内在时序结构。
1.  **输入构建:** 将状态 $s$ 和动作 $a$ 交错排列，并在序列头部拼接可学习的具身数据集标识 Token ($p_k$) 和汇总 Token ($cls$)。
2.  **编码器 ($E_{\phi_{enc}}$):** 采用 Transformer 架构，将上述序列压缩，提取 $cls$ 对应的 $L_2$ 归一化特征作为动作隐变量 $z$。该变量表征了整段动作块的宏观运动语义。
3.  **解码器 ($D_{\phi_{dec}}$):** 采用基于 Flow-Matching 的 Transformer。以隐变量 $z$ 为条件，输入加噪的动作 $x_r$，预测去噪速度场 $v_r$，通过重构原始动作块来优化整个自编码器。

**Stage 2: 动作先验引导的 VLA 训练 (Action-Prior-Guided VLA Training)**
目标是将 Stage 1 学到的物理运动流形注入到 VLM 的跨模态对齐过程中。
1.  **历史上下文压缩:** 复用 Stage 1 的编码器，将过去的历史状态-动作轨迹 $\tau_{hist}$ 压缩为一个单一的隐变量 Token $z_{hist}$，以极低的计算代价为 VLM 提供时序感受野。
2.  **VLM 预测:** VLM 接收视觉观察 $o_t$、语言指令 $l_t$、数据集 Token $p_k$、历史 Token $z_{hist}$ 以及一个可学习的查询 Token $q$。VLM 输出 $q$ 对应的特征作为预测的动作隐变量 $z'$。
3.  **早期隐空间蒸馏:** 在训练早期，强制 VLM 预测的 $z'$ 逼近 Stage 1 编码器生成的真实轨迹隐变量 $z$（Stop-gradient）。该蒸馏损失的权重 $\lambda(k)$ 随训练步数 $k$ 线性衰减至 0，允许模型在后期进行自由的端到端微调。
4.  **动作生成:** 复用 Stage 1 的解码器作为 VLA 的动作头，接收 VLM 预测的 $z'$，通过 Flow-Matching 生成最终动作。

**核心伪代码实现:**

```python
import torch
import torch.nn as nn

class ActionPriorVLA(nn.Module):
    def __init__(self, vlm_backbone, action_encoder, action_decoder, decay_steps=5000):
        super().__init__()
        self.vlm = vlm_backbone
        self.encoder = action_encoder # Pre-trained in Stage 1
        self.decoder = action_decoder # Pre-trained in Stage 1
        self.decay_steps = decay_steps
        
        # Freeze encoder during Stage 2 to maintain stable latent space
        for param in self.encoder.parameters():
            param.requires_grad = False

    def get_decay_weight(self, current_step):
        # Monotonically decaying weight for distillation
        return max(0.0, 1.0 - current_step / self.decay_steps)

    def forward(self, obs, lang, state_action_history, current_trajectory, current_step):
        # 1. History Compression (using frozen Stage 1 encoder)
        # Compress long history into a single token
        z_hist = self.encoder(state_action_history) 
        
        # 2. VLM Forward Pass
        # VLM predicts the latent action embedding z_pred
        z_pred = self.vlm(obs, lang, z_hist) 
        
        # 3. Ground-truth Latent Extraction
        # Extract structured latent z_gt from current trajectory
        with torch.no_grad():
            z_gt = self.encoder(current_trajectory)
            
        # 4. Flow-Matching Action Generation
        # Decoder predicts velocity field v_pred conditioned on z_pred
        noisy_actions, r, v_gt = self.add_flow_matching_noise(current_trajectory.actions)
        v_pred = self.decoder(noisy_actions, r, z_pred)
        
        # 5. Loss Computation
        # Primary prediction loss (end-to-end VLA objective)
        L_pred = nn.functional.mse_loss(v_pred, v_gt)
        
        # Latent Alignment Distillation Loss
        L_align = nn.functional.mse_loss(z_pred, z_gt.detach())
        
        # Auxiliary Reconstruction Loss (maintains decoder geometry)
        v_recon = self.decoder(noisy_actions, r, z_gt.detach())
        L_recon = nn.functional.mse_loss(v_recon, v_gt)
        
        # Total Loss with early-stop distillation schedule
        lambda_k = self.get_decay_weight(current_step)
        L_total = L_pred + lambda_k * (L_align + L_recon)
        
        return L_total
```

**2. VLA 模型架构与训练流程**

**核心逻辑伪代码 (Python 风格):**

import torch
import torch.nn as nn

class WOLFVLA(nn.Module):
    def __init__(self, groot_backbone):
        super().__init__()
        # 冻结视觉和语言编码器
        self.vision_encoder = groot_backbone.vision_encoder.requires_grad_(False)
        self.language_encoder = groot_backbone.language_encoder.requires_grad_(False)
        # 优化动作解码器 (基于流匹配的扩散模型)
        self.action_decoder = groot_backbone.action_decoder.requires_grad_(True)
        
    def forward(self, rgb_images, language_instruction, proprioception, target_action_chunk=None):
        # 1. 多模态特征提取
        v_tokens = self.vision_encoder(rgb_images)
        l_tokens = self.language_encoder(language_instruction)
        obs_features = concat(v_tokens, l_tokens, proprioception)
        
        if self.training:
            # 2. 训练阶段：流匹配 (Flow-Matching) 目标计算
            batch_size = target_action_chunk.shape[0]
            # 采样噪声和时间步
            noise = torch.randn_like(target_action_chunk)
            tau = torch.rand(batch_size, 1, 1).to(target_action_chunk.device)
            
            # 线性插值构建带噪动作
            noisy_action = tau * target_action_chunk + (1 - tau) * noise
            
            # 预测向量场
            predicted_vector_field = self.action_decoder(noisy_action, obs_features, tau)
            
            # 计算目标流方向
            target_vector_field = target_action_chunk - noise
            
            # 计算 MSE 损失
            loss = nn.MSELoss()(predicted_vector_field, target_vector_field)
            return loss
            
        else:
            # 3. 推理阶段：欧拉积分去噪生成动作块
            K_steps = 10
            dt = 1.0 / K_steps
            current_action = torch.randn(batch_size, horizon, action_dim) # a^0 ~ N(0, I)
            
            for k in range(K_steps):
                tau = torch.full((batch_size, 1, 1), k * dt)
                vector_field = self.action_decoder(current_action, obs_features, tau)
                current_action = current_action + dt * vector_field
                
            return current_action # 返回去噪后的动作块 (关节增量)

### 第二模块：核心架构解释

Causal-rCM 的核心架构是一个三阶段的流水线，辅以高度定制化的底层并行与算子基建。

**阶段一：因果教师模型准备 (Causal Teacher Preparation)**
将预训练的双向扩散模型（Bidirectional DiT）通过 Teacher-Forcing 转换为自回归因果模型。模型在训练时接收拼接的“干净历史帧”与“加噪目标帧”，通过因果掩码（Causal Mask）确保信息单向流动。

**阶段二：Teacher-Forcing 一致性蒸馏 (TF-CM)**
在固定历史上下文的前提下，对目标帧应用连续时间一致性模型（sCM/MeanFlow）。为了解决长视频序列计算雅可比向量积（JVP）时的显存爆炸问题，作者开发了支持自定义稀疏掩码的 FlashAttention-2 JVP 算子。此阶段为后续微调提供了极强且模式完整的初始化权重。

**阶段三：Self-Forcing 分布匹配蒸馏 (SF-DMD)**
模型切换到真实的自回归推理模式。利用 KV Cache 逐块（Chunk）生成视频。生成完成后，对最终输出的视频块应用 DMD 损失（通过一个预训练的 Fake-score 网络与 Teacher 网络计算得分差）。梯度仅通过每个 Chunk 的最后一次去噪步骤进行反向传播。

**底层基建支撑**
整个架构建立在 FSDP2（全分片数据并行）、Ulysses CP（上下文并行）以及选择性激活重计算（SAC）之上，确保了超长视频序列在多 GPU 集群上的可训练性。

# Stage 2: TF-sCM (Teacher-Forcing Continuous-Time Consistency)
def train_step_tf_scm(student_model, teacher_model, clean_context, noisy_target, t):
    # 1. Pack sequence and apply causal mask
    packed_seq = concat([clean_context, noisy_target])
    causal_mask = build_custom_tf_mask()
    
    # 2. Compute teacher velocity (stop gradient)
    with torch.no_grad():
        v_teacher = teacher_model(packed_seq, t, mask=causal_mask)
    
    # 3. Compute student velocity and JVP using custom FlashAttention-2 kernel
    # JVP computes the directional derivative along the teacher's ODE trajectory
    v_student, jvp_term = custom_flash_attn_jvp(
        student_model, packed_seq, t, v_teacher, mask=causal_mask
    )
    
    # 4. Compute continuous-time consistency loss (MSE between tangent and velocity difference)
    tangent_target = v_teacher - v_student_stop_grad - t * jvp_term
    loss_scm = mse_loss(v_student, tangent_target)
    loss_scm.backward()

# Stage 3: SF-DMD (Self-Forcing Distribution Matching Distillation)
def train_step_sf_dmd(student_model, teacher_model, fake_score_model, z_noise, num_chunks):
    kv_cache = empty_cache()
    generated_chunks = []
    
    # 1. Autoregressive Rollout (Self-Forcing)
    for i in range(num_chunks):
        # Multi-step denoising for the current chunk
        x_chunk = z_noise[i]
        for step in range(denoising_steps):
            # Only keep gradient for the final denoising step to save memory
            if step == denoising_steps - 1:
                x_chunk = student_model(x_chunk, kv_cache)
            else:
                with torch.no_grad():
                    x_chunk = student_model(x_chunk, kv_cache)
        
        generated_chunks.append(x_chunk)
        # Update KV cache with the newly generated chunk (detached)
        kv_cache.append(x_chunk.detach()) 
    
    # 2. Distribution Matching Distillation
    final_video = concat(generated_chunks)
    with torch.no_grad():
        score_teacher = teacher_model(final_video)
        score_fake = fake_score_model(final_video)
    
    # DMD loss forces student distribution to match teacher distribution
    loss_dmd = mse_loss(final_video, score_fake - score_teacher)
    loss_dmd.backward()

### 第二模块：核心架构解释

ICWM 的核心思想是将 VLA 从静态的映射函数转变为动态的推理引擎。其工作流分为训练和测试两个阶段，核心在于如何构造和利用交互上下文（Interaction Context）$\mathcal{T}$。

**1. 训练阶段 (In-Context Training)**
在训练时，模型不仅接收当前观察 $o_t$ 和语言指令 $l$，还会被强制要求关注一段前置的交互上下文 $\mathcal{T}$。$\mathcal{T}$ 由 $N$ 个随机采样的交互片段组成，每个片段包含 $(o_i^s, a_i, o_i^e)$，即起始图像、执行的动作、结束图像。模型通过标准的行为克隆（BC）损失进行训练，被迫学会从 $\mathcal{T}$ 中提取当前视角的空间映射关系，以准确预测目标动作 $a_t$。

**2. 推理阶段 (Test-Time Active Probing and Inference)**
部署到新环境时，分为两步：
- **主动探测 (Active Probing)**：机器人不急于执行任务，而是先在安全工作区内执行 $N$ 次随机的试探性动作，记录下这 $N$ 次的视觉变化，构建出当前真实环境的上下文 $\mathcal{T}$。
- **上下文执行 (In-Context Execution)**：将收集到的 $\mathcal{T}$ 作为 Prompt 拼接到当前任务的观察和指令前，输入给 VLA 模型。Transformer 的注意力机制会利用 $\mathcal{T}$ 中的动态信息来校准当前观察，输出适应新环境的动作。

**核心伪代码实现逻辑**：

import torch
import torch.nn as nn

class ICWM_VLA(nn.Module):
    def __init__(self, vision_encoder, action_tokenizer, vlm_backbone):
        super().__init__()
        self.vision_encoder = vision_encoder
        self.action_tokenizer = action_tokenizer
        self.vlm_backbone = vlm_backbone # e.g., Qwen2.5-VL
        
    def encode_interaction_clip(self, img_start, action, img_end):
        """编码单个试探性交互片段"""
        v_start = self.vision_encoder(img_start)
        a_token = self.action_tokenizer.encode(action)
        v_end = self.vision_encoder(img_end)
        # 拼接形成单个 clip 的 token 序列
        return torch.cat([v_start, a_token, v_end], dim=1)

    def forward(self, context_clips, current_obs, instruction):
        """
        context_clips: List of (img_start, action, img_end) tuples, length N
        current_obs: 当前视角的图像
        instruction: 语言指令文本
        """
        # 1. 构建 Interaction Context T
        context_tokens = []
        for img_s, act, img_e in context_clips:
            clip_token = self.encode_interaction_clip(img_s, act, img_e)
            context_tokens.append(clip_token)
        T_tokens = torch.cat(context_tokens, dim=1) # 隐式包含系统配置 \psi
        
        # 2. 构建 Task Query
        obs_tokens = self.vision_encoder(current_obs)
        text_tokens = self.vlm_backbone.tokenize_text(instruction)
        query_tokens = torch.cat([obs_tokens, text_tokens], dim=1)
        
        # 3. In-Context Inference
        # 模型通过 Attention 机制，利用 T_tokens 中的动力学信息校准 query_tokens
        full_sequence = torch.cat([T_tokens, query_tokens], dim=1)
        action_logits = self.vlm_backbone(full_sequence)
        
        # 4. 解码动作
        predicted_action = self.action_tokenizer.decode(action_logits)
        return predicted_action

# 部署时的使用流程

# 2. Task Execution Phase

### 第二模块：核心架构解释

本研究的架构分为完全解耦的两个阶段：纯动作先验学习阶段（Stage 1）与动作先验引导的 VLA 训练阶段（Stage 2）。

**Stage 1: 动作先验学习 (Action Prior Learning)**
目标是让动作模块在没有任何视觉和语言条件的情况下，仅通过本体感觉状态（Proprioceptive states）和动作（Actions）学习物理运动的内在时序结构。
1.  **输入构建:** 将状态 $s$ 和动作 $a$ 交错排列，并在序列头部拼接可学习的具身数据集标识 Token ($p_k$) 和汇总 Token ($cls$)。
2.  **编码器 ($E_{\phi_{enc}}$):** 采用 Transformer 架构，将上述序列压缩，提取 $cls$ 对应的 $L_2$ 归一化特征作为动作隐变量 $z$。该变量表征了整段动作块的宏观运动语义。
3.  **解码器 ($D_{\phi_{dec}}$):** 采用基于 Flow-Matching 的 Transformer。以隐变量 $z$ 为条件，输入加噪的动作 $x_r$，预测去噪速度场 $v_r$，通过重构原始动作块来优化整个自编码器。

**Stage 2: 动作先验引导的 VLA 训练 (Action-Prior-Guided VLA Training)**
目标是将 Stage 1 学到的物理运动流形注入到 VLM 的跨模态对齐过程中。
1.  **历史上下文压缩:** 复用 Stage 1 的编码器，将过去的历史状态-动作轨迹 $\tau_{hist}$ 压缩为一个单一的隐变量 Token $z_{hist}$，以极低的计算代价为 VLM 提供时序感受野。
2.  **VLM 预测:** VLM 接收视觉观察 $o_t$、语言指令 $l_t$、数据集 Token $p_k$、历史 Token $z_{hist}$ 以及一个可学习的查询 Token $q$。VLM 输出 $q$ 对应的特征作为预测的动作隐变量 $z'$。
3.  **早期隐空间蒸馏:** 在训练早期，强制 VLM 预测的 $z'$ 逼近 Stage 1 编码器生成的真实轨迹隐变量 $z$（Stop-gradient）。该蒸馏损失的权重 $\lambda(k)$ 随训练步数 $k$ 线性衰减至 0，允许模型在后期进行自由的端到端微调。
4.  **动作生成:** 复用 Stage 1 的解码器作为 VLA 的动作头，接收 VLM 预测的 $z'$，通过 Flow-Matching 生成最终动作。

**核心伪代码实现:**

```python
import torch
import torch.nn as nn

class ActionPriorVLA(nn.Module):
    def __init__(self, vlm_backbone, action_encoder, action_decoder, decay_steps=5000):
        super().__init__()
        self.vlm = vlm_backbone
        self.encoder = action_encoder # Pre-trained in Stage 1
        self.decoder = action_decoder # Pre-trained in Stage 1
        self.decay_steps = decay_steps
        
        # Freeze encoder during Stage 2 to maintain stable latent space
        for param in self.encoder.parameters():
            param.requires_grad = False

    def get_decay_weight(self, current_step):
        # Monotonically decaying weight for distillation
        return max(0.0, 1.0 - current_step / self.decay_steps)

    def forward(self, obs, lang, state_action_history, current_trajectory, current_step):
        # 1. History Compression (using frozen Stage 1 encoder)
        # Compress long history into a single token
        z_hist = self.encoder(state_action_history) 
        
        # 2. VLM Forward Pass
        # VLM predicts the latent action embedding z_pred
        z_pred = self.vlm(obs, lang, z_hist) 
        
        # 3. Ground-truth Latent Extraction
        # Extract structured latent z_gt from current trajectory
        with torch.no_grad():
            z_gt = self.encoder(current_trajectory)
            
        # 4. Flow-Matching Action Generation
        # Decoder predicts velocity field v_pred conditioned on z_pred
        noisy_actions, r, v_gt = self.add_flow_matching_noise(current_trajectory.actions)
        v_pred = self.decoder(noisy_actions, r, z_pred)
        
        # 5. Loss Computation
        # Primary prediction loss (end-to-end VLA objective)
        L_pred = nn.functional.mse_loss(v_pred, v_gt)
        
        # Latent Alignment Distillation Loss
        L_align = nn.functional.mse_loss(z_pred, z_gt.detach())
        
        # Auxiliary Reconstruction Loss (maintains decoder geometry)
        v_recon = self.decoder(noisy_actions, r, z_gt.detach())
        L_recon = nn.functional.mse_loss(v_recon, v_gt)
        
        # Total Loss with early-stop distillation schedule
        lambda_k = self.get_decay_weight(current_step)
        L_total = L_pred + lambda_k * (L_align + L_recon)
        
        return L_total
```

### 第二模块：核心架构解释

ROAD-VLA 的核心逻辑在于将强化学习的标量反馈转化为监督学习的密集分布匹配。其运行流程分为四个关键步骤：

1.  **数据收集 (On-policy Rollout)**: 部署的 VLA 学生模型（Student）根据当前视觉观察和语言指令，在环境中自回归地生成离散动作 Token，收集轨迹并获取稀疏的环境奖励。
2.  **优势校准与门控 (Calibrated & Gated Advantage)**: 为了降低方差，算法同时计算当前在线 Critic 的内在优势值和冻结参考 Critic（预训练 PPO 的 Critic）的参考优势值。通过统计学对齐后，仅当两者对动作好坏的判断（符号）一致时才进行混合；若产生分歧，则丢弃参考信号，防止陈旧的 Critic 误导当前策略。
3.  **构建近端教师 (Proximal Teacher Construction)**: 将上述计算出的标量优势值转化为权重，直接加到学生模型对应采样动作的 Logits 上。优势值为正，该动作 Token 的 Logit 增加；优势值为负，Logit 减少。经过 Softmax 后，形成一个在动作空间上局部优化的教师分布。
4.  **自蒸馏更新 (Self-Distillation Update)**: 计算学生模型输出的动作分布与上述构建的教师分布之间的 KL 散度（或 JSD），通过反向传播更新学生模型参数。

**核心伪代码实现：**

```python
def road_vla_update(student_policy, online_critic, ref_critic, trajectories, eta=1.0, alpha=0.5, clip_c=2.0):
    """
    ROAD-VLA 核心更新逻辑
    student_policy: 当前 VLA 模型
    online_critic: 当前价值网络
    ref_critic: 冻结的参考价值网络
    trajectories: 收集到的 on-policy 轨迹数据
    """
    loss_total = 0
    
    for step in trajectories:
        obs, action_tokens, reward = step
        
        # 1. 计算内在优势与参考优势
        adv_int = online_critic.compute_advantage(obs, reward)
        adv_ref = ref_critic.compute_advantage(obs, reward)
        
        # 2. 优势校准与门控机制
        adv_ref_calibrated = calibrate(adv_ref, adv_int) # 均值方差对齐
        
        if sign(adv_int) == sign(adv_ref_calibrated):
            # 符号一致，混合优势
            adv_mix = adv_int + alpha * (adv_ref_calibrated - adv_int)
        else:
            # 符号分歧，回退到在线优势
            adv_mix = adv_int
            
        # 截断优势权重
        weight = clip(adv_mix, -clip_c, clip_c)
        
        # 3. 构建优势引导的近端教师分布
        student_logits = student_policy.get_logits(obs)
        teacher_logits = student_logits.clone().detach()
        
        # 仅对采样到的 action_tokens 的 logits 进行扰动
        for k, token in enumerate(action_tokens):
            teacher_logits[k, token] += eta * weight
            
        teacher_probs = softmax(teacher_logits)
        student_probs = softmax(student_logits)
        
        # 4. 计算 Token 级蒸馏损失 (Forward KL)
        loss_distill = compute_kl_divergence(teacher_probs, student_probs)
        loss_total += loss_distill
        
    # 梯度下降更新 student_policy
    loss_total.backward()
    optimizer.step()
```

### 第二模块：核心架构解释

Reflective VLA 的核心架构建立在双系统（VLM + 连续动作专家）之上，其运行与训练流程如下：

1. **三元组构造**：将机器人的交互历史结构化为 $(O_i, A_i, O'_i)$ 三元组。其中 $O_i$ 是动作执行前的观测，$A_i$ 是执行的动作块（Action Chunk），$O'_i$ 是该动作块执行完毕后的观测（即动作后果）。
2. **多模态提示词拼接**：在时间步 $t$，模型接收语言指令 $L$、历史三元组序列以及当前的观测 $O_t$。所有模态（图像、本体感觉、历史动作）均被投影到共享的 VLM 词表空间中。
3. **共享注意力双系统**：VLM 编码上述多模态前缀，连续动作专家（基于 Flow-matching）作为后缀附加在 VLM 之后。动作专家与 VLM 前缀共享注意力层，使其能够直接读取历史的动作及其造成的物理后果。
4. **块因果掩码训练（Block-causal training）**：为了避免序列变长导致训练成本线性增加，作者在一个前向传播中打包 $K$ 个帧。通过定制的注意力掩码，确保第 $k$ 个查询只能看到其之前的完整三元组和当前的观测，严格防止未来信息泄露。

import torch
import torch.nn as nn

class ReflectiveVLA(nn.Module):
    def __init__(self, vlm_backbone, action_expert, action_projector):
        super().__init__()
        self.vlm = vlm_backbone
        self.action_expert = action_expert
        self.action_proj = action_projector # 映射动作到 VLM token 空间
        
    def construct_prompt(self, instruction, history_triplets, current_obs):
        # history_triplets: list of (O_i, A_i, O'_i)
        prompt_tokens = []
        for (O_i, A_i, O_prime_i) in history_triplets:
            A_tokens = self.action_proj(A_i)
            # 拼接单个历史三元组: [指令, 执行前观测, 动作, 执行后观测]
            triplet_seq = [instruction, O_i, A_tokens, O_prime_i]
            prompt_tokens.extend(triplet_seq)
            
        # 拼接当前查询: [指令, 当前观测]
        prompt_tokens.extend([instruction, current_obs])
        return prompt_tokens

    def generate_block_causal_mask(self, K, seq_lengths):
        # K: 打包的帧数
        # 生成掩码矩阵，确保第 k 个 query 只能 attend 到 j < k 的三元组以及当前的 O_k
        # 屏蔽自身的 A_k, O'_k 以及所有 j > k 的未来信息
        mask = create_causal_mask(K, seq_lengths) 
        return mask

    def forward(self, instruction, packed_frames, K):
        # 训练阶段：单次前向传播处理 K 个帧
        # packed_frames 包含 K 个 (O, A, O') 序列
        
        tokens = self.encode_multimodal(instruction, packed_frames)
        mask = self.generate_block_causal_mask(K, get_lengths(packed_frames))
        
        # VLM 处理共享前缀
        vlm_features = self.vlm(tokens, attention_mask=mask)
        
        # 动作专家基于共享特征预测动作块
        # Flow-matching 目标函数计算损失
        loss = 0
        for k in range(K):
            target_A_k = packed_frames[k].A
            pred_A_k = self.action_expert(vlm_features, query_index=k)
            loss += flow_matching_loss(pred_A_k, target_A_k)
            
        return loss / K

    @torch.no_grad()
    def inference_step(self, instruction, context_buffer, current_obs):
        # 部署阶段：利用 KV-Cache 保持实时性
        prompt = self.construct_prompt(instruction, context_buffer, current_obs)
        vlm_features = self.vlm.forward_with_kv_cache(prompt)
        pred_action_chunk = self.action_expert(vlm_features)
        return pred_action_chunk

1.  **原始误差曲线构建 (Raw Curve Generation):**

2.  **单侧共形校准 (Split-Conformal Calibration):**

3.  **轨道传输 (Orbit Transport):**

### 第二模块：核心架构解释

ACNet 的核心思想是借鉴图像生成领域的 ControlNet，将“机器人在等待推理期间已经执行的动作”作为一种边界约束条件，强行注入到冻结的动作生成头中，确保新生成的动作块能够完美衔接当前正在执行的物理动作。

**核心处理流程：**
1.  **延迟动作编码 (Delay-Action Encoding):** 提取机器人在推理延迟 $d$ 期间实际执行的动作后缀。由于动作头需要固定长度 $H$ 的输入，ACNet 使用**可学习的 Padding Tokens** 补齐未观测到的未来时间步，避免了补零或随机噪声带来的语义歧义。随后通过轻量级 Transformer 编码并进行时间池化，提取出局部边界特征。
2.  **残差注入 (Residual Injection):** 视觉-语言骨干网络保持完全冻结，动作生成头（如 Flow Matching 或 DiT）也基本冻结。ACNet 将提取的边界特征通过线性投影，以残差相加的方式，**仅注入到动作生成头的最后几层**。
3.  **局部边界修正:** 越靠近输出层的网络对最终动作轨迹的物理形态影响越大。通过在末端注入残差，模型在保留原有任务语义的同时，在输出前最后一刻修正了动作轨迹的起点，实现了平滑拼接。

**Python 风格伪代码：**

```python
import torch
import torch.nn as nn

class ActionControlNet(nn.Module):
    def __init__(self, action_dim, chunk_size, hidden_dim):
        super().__init__()
        # 冻结的预训练组件
        self.vlm_backbone = FrozenVLM()
        self.action_head = FrozenActionExpert() # 例如 8层的 DiT
        
        # ACNet 可训练组件
        self.learnable_padding = nn.Parameter(torch.randn(chunk_size, action_dim))
        self.delay_encoder = LightweightTransformer()
        self.acnet_proj = nn.Linear(encoded_dim, hidden_dim) # 投影到动作头的隐藏层维度

    def forward(self, obs, lang_instruction, executed_delay_action, delay_steps):
        # 1. 视觉-语言特征提取 (完全冻结，可缓存复用)
        with torch.no_grad():
            task_latent = self.vlm_backbone(obs, lang_instruction)
            
        # 2. 延迟动作对齐与编码
        # 将已执行的动作后缀与可学习的 padding 拼接，对齐到 chunk_size
        padded_delay_action = torch.cat([
            executed_delay_action, 
            self.learnable_padding[delay_steps:]
        ], dim=0)
        
        # 提取边界条件特征
        delay_features = self.delay_encoder(padded_delay_action)
        condition_vector = temporal_pooling(delay_features) # 提取时序池化特征
        
        # 3. 动作头前向传播与残差注入
        # 假设 action_head 有 L 层，仅在最后一层 (L-1) 注入 ACNet 信号
        hidden_state = task_latent
        for i, layer in enumerate(self.action_head.layers):
            if i == self.action_head.num_layers - 1:
                # 核心创新：在末端层注入局部边界修正信号
                residual_correction = self.acnet_proj(condition_vector)
                hidden_state = hidden_state + residual_correction
                
            hidden_state = layer(hidden_state)
            
        # 4. 输出平滑且边界对齐的动作块
        smooth_action_chunk = self.action_head.output_proj(hidden_state)
        return smooth_action_chunk
```

import torch
import torch.nn as nn

class USSFramework(nn.Module):
    def __init__(self):
        super().__init__()
        self.visual_encoder = VisualEncoder()
        self.prompt_encoders = nn.ModuleDict({
            'text': TextEncoder(),
            'point': PointEncoder(),
            'bbox': BBoxEncoder(),
            'mask': MaskEncoder()
        })
        self.hybrid_attention = HybridAttention()
        self.latent_world_model = LatentDynamicsModel()
        self.action_decoder = WaypointDecoder()

    def forward(self, image_t, prompt, prompt_type, action_history):
        # 1. 提取视觉特征与提示特征
        visual_feat = self.visual_encoder(image_t)
        prompt_token = self.prompt_encoders[prompt_type](prompt)

        # 2. 混合注意力融合，生成目标条件表征
        fused_state_t = self.hybrid_attention(visual_feat, prompt_token)

        # 3. 动作解码，输出自中心航点
        waypoints = self.action_decoder(fused_state_t)

        # 4. 潜在世界模型前向预测 (用于训练阶段)
        predicted_state_t1 = self.latent_world_model(fused_state_t, action_history)

        return waypoints, predicted_state_t1

    def compute_loss(self, image_t, image_t1, prompt, prompt_type, action_history, gt_waypoints):
        # 前向传播
        waypoints, pred_state_t1 = self.forward(image_t, prompt, prompt_type, action_history)

        # 动作模仿学习损失 (Behavior Cloning)
        action_loss = nn.MSELoss()(waypoints, gt_waypoints)

        # 世界模型自监督对齐损失
        with torch.no_grad():
            visual_feat_t1 = self.visual_encoder(image_t1)
            prompt_token = self.prompt_encoders[prompt_type](prompt)
            # 获取真实的下一时刻融合表征
            true_state_t1 = self.hybrid_attention(visual_feat_t1, prompt_token)

        # 约束预测状态与真实状态在潜在空间中一致
        dynamics_loss = 1.0 - nn.CosineSimilarity(dim=-1)(pred_state_t1, true_state_t1).mean()

        return action_loss + self.lambda_dyn * dynamics_loss

### 第二模块：核心架构解释

SVP-IL 的核心架构将复杂的 VLA 任务拆解为“语义解析与空间定位”和“连续动作生成”两个解耦的阶段。

**1. 语义解析与空间定位 (Semantic Parsing & Spatial Grounding)**

**3. 连续动作生成 (Continuous Action Generation)**
将融合后的特征输入到动作生成网络中。由于特征已经包含了明确的目标空间位置，网络只需专注于学习“如何操作（How）”，从而在低数据下稳定输出连续的机器人控制动作。

import torch
import torch.nn as nn

class SVP_IL(nn.Module):
    def __init__(self, vlm_grounding_model, visual_encoder, action_generator):
        super().__init__()
        # 预训练的VLM定位模型（冻结权重），用于零样本生成 Mask
        self.vlm = vlm_grounding_model
        self.vlm.eval()
        
        # 视觉特征提取器
        self.visual_encoder = visual_encoder
        
        # 轻量级特征融合层
        self.fusion_layer = nn.Conv2d(in_channels=visual_dim + 1, out_channels=visual_dim, kernel_size=1)
        
        # 动作生成网络（如 Diffusion Policy 或 MLP）
        self.action_generator = action_generator

    def forward(self, rgb_image, language_instruction, proprioception):
        # 阶段1：解耦的语义与几何定位 (Zero-shot)
        with torch.no_grad():
            # 生成空间视觉提示 (SVP)，即目标物体的二进制 Mask
            # mask shape: [B, 1, H, W]
            spatial_visual_prompt = self.vlm.generate_mask(rgb_image, language_instruction)
        
        # 阶段2：视觉特征提取
        # visual_features shape: [B, visual_dim, H, W]
        visual_features = self.visual_encoder(rgb_image)
        
        # 阶段3：轻量级特征级融合
        # 将 Mask 作为先验直接拼接并融合
        fused_input = torch.cat([visual_features, spatial_visual_prompt], dim=1)
        grounded_features = self.fusion_layer(fused_input)
        
        # 展平或池化特征用于动作生成
        global_feature = grounded_features.mean(dim=[2, 3]) 
        
        # 阶段4：连续动作生成
        # 结合本体感觉输出最终动作
        action = self.action_generator(global_feature, proprioception)
        
        return action

### 第二模块：核心架构解释

MANGO 的架构设计摒弃了单次生成的端到端黑盒模式，而是采用了高度解耦的三阶段流水线。每个阶段内部都运行着一套标准的“生成器 (Generator) - 评估团 (Assessors) - 裁判 (Judge)”多智能体迭代优化逻辑。生成器由具备物理常识的世界模型（World Model）担任，评估团由多个专注于单一维度（如逻辑、物理接地、冗余度）的轻量级 LLM 并行担任，裁判由推理型 LLM 担任以解决冲突并输出修改指令。

整个工作流分为三个模块：
1. **原子任务库生成 (Module I):** 输入一组复杂的自然语言指令，系统将其拆解、去重、泛化，输出一个纯粹的“原子任务库”（如 `pick_up`, `open`）。
2. **原子任务预言库生成 (Module II):** 结合仿真器提供的底层 API 函数，为模块一中的每个原子任务生成具体的验证逻辑代码（如 `pick_up` 对应 `is_held(robot, obj) and is_lifted(obj)`）。
3. **细粒度预言生成 (Module III):** 当面对一条全新的复杂指令时，系统调用前两个模块生成的库，将其规划为带有时序约束（顺序或并行）的执行图，并绑定对应的验证函数，最终输出可直接在仿真器中执行的 JSON 格式测试脚本。

**核心逻辑伪代码：**

```python
class MANGO_Framework:
    def __init__(self, simulator_apis, max_iterations=10):
        # 核心多智能体设定
        self.generator = WorldModel() # 擅长物理常识和动作规划
        self.assessors = [LogicAssessor(), GroundingAssessor(), ComplianceAssessor()] # 并行轻量级LLM
        self.judge = ReasoningLLM() # 负责最终裁决和生成结构化修改意见
        self.simulator_apis = simulator_apis
        self.max_iters = max_iterations

    def multi_agent_refinement_loop(self, initial_input, task_type):
        """通用的多智能体迭代优化闭环"""
        candidate = self.generator.generate(initial_input, task_type)
        
        for _ in range(self.max_iters):
            # 1. 评估团并行审查候选方案的不同维度
            assessor_reports = [assessor.evaluate(candidate) for assessor in self.assessors]
            
            # 2. 裁判综合所有报告进行判定
            verdict, refinement_feedback = self.judge.decide(candidate, assessor_reports)
            
            if verdict == "APPROVED":
                return candidate
            else:
                # 3. 生成器根据反馈重新修改候选方案
                candidate = self.generator.refine(candidate, refinement_feedback)
                
        return candidate # 达到最大迭代次数，强制输出当前最优解

    def generate_executable_oracle(self, complex_task_set, new_instruction, env_objects):
        """MANGO 三阶段主流程"""
        # Module I: 从复杂任务集中提取可复用的原子任务库
        atomic_task_lib = self.multi_agent_refinement_loop(
            initial_input=complex_task_set, 
            task_type="extract_atomic_tasks"
        )

        # Module II: 结合仿真器API，为原子任务生成验证逻辑(Oracle)
        atomic_oracle_lib = self.multi_agent_refinement_loop(
            initial_input=(atomic_task_lib, self.simulator_apis),
            task_type="generate_atomic_oracles"
        )

        # Module III: 为新指令生成带时序的细粒度测试预言执行计划
        fine_grained_oracle = self.multi_agent_refinement_loop(
            initial_input=(new_instruction, atomic_task_lib, atomic_oracle_lib, env_objects),
            task_type="plan_fine_grained_oracle"
        )

        return fine_grained_oracle
```

# 第二模块：核心架构解释

DynaMOMA 的架构分为两个解耦但协同的模块：瞬时抓取轨迹预测器（Predictor）和全身控制策略（Policy）。

1.  **历史状态编码**：系统在运行过程中，维护一个固定窗口的历史缓冲区，提取相机位姿序列、物体位姿序列以及由 GraspNet 生成的原始抓取候选序列。
2.  **基于锚点的抓取轨迹预测**：为了解决扩散模型推理慢和多模态不稳定的问题，模型预先通过 K-Means 在训练集上聚类出 K 个“锚点轨迹”。推理时，直接在这些锚点上添加截断噪声，结合历史状态条件进行少步数（2-step DDIM）去噪，输出 K 条预测轨迹及置信度，选取最高分轨迹。
3.  **预期引导的强化学习训练**：RL 策略接收本体感觉、视觉特征（Depth Anything V2）和预测轨迹特征，输出底盘（2-DoF）、机械臂（6-DoF）和夹爪（1-DoF）的动作。在计算 Reward 时，引入距离衰减系数 $\alpha$：当机器人距离目标较远时，Reward 引导机器人对齐当前帧的抓取位姿；当进入临近抓取窗口时，Reward 强制机器人对齐预测的未来抓取位姿。

**核心逻辑伪代码 (Python-style)**：

```python
import numpy as np

def predict_grasp_trajectory(history_context, anchors, T_trunc, diffusion_decoder):
    """
    基于锚点的截断扩散预测
    """
    # 在预定义的 K 个锚点轨迹上添加截断时间步 T_trunc 的高斯噪声
    noisy_trajs = add_gaussian_noise(anchors, timestep=T_trunc)
    
    # 条件去噪，输出 K 条轨迹及其置信度得分
    denoised_trajs, confidences = diffusion_decoder(noisy_trajs, history_context)
    
    # 选取置信度最高的轨迹作为最终预测
    best_idx = np.argmax(confidences)
    return denoised_trajs[best_idx], confidences[best_idx]

def compute_anticipation_guided_target(current_grasp, predicted_traj, eef_pos, d_near, d_far):
    """
    计算预期引导的控制目标 (对应论文 Eq. 4, 5, 6)
    """
    # 1. 计算距离依赖的信任系数 alpha
    dist = np.linalg.norm(eef_pos - current_grasp.pos)
    alpha = np.clip((dist - d_near) / (d_far - d_near), 0.0, 1.0)
    
    # 2. 将预测轨迹聚合为单一的未来目标位姿
    # 使用时间倒序的 logits，随着 alpha 减小，权重向轨迹的最近未来步 (g_0) 集中
    time_reversing_logits = np.arange(len(predicted_traj), 0, -1)
    weights = softmax((1 - alpha) * time_reversing_logits)
    future_grasp = np.sum(weights[:, None] * predicted_traj, axis=0)
    
    # 3. 线性插值生成最终 Reward 目标
    target_grasp = (1 - alpha) * future_grasp + alpha * current_grasp
    return target_grasp

def control_loop_step(obs, policy_net):
    """
    环境交互单步执行逻辑
    """
    # 预测未来抓取轨迹
    pred_traj, conf = predict_grasp_trajectory(obs.history, anchors, T_trunc, decoder)
    
    # 特征编码
    s_pred = encode_prediction(pred_traj, conf)
    s_vis = depth_anything_v2_encoder(obs.rgbd)
    s_prop = obs.proprioception
    s_grasp = encode_grasps(obs.raw_grasps)
    
    # 策略推理
    state = np.concatenate([s_prop, s_grasp, s_vis, s_pred])
    action_base, action_arm, action_gripper = policy_net(state)
    
    return action_base, action_arm, action_gripper
```

1. **特征提取与初始化**：图像主干网络提取多视角特征。系统初始化包含智能体、地图和规划的联合锚点（Anchors），并向其注入当前时间步的扩散噪声，映射为共享的 Latent Queries。
2. **时间过渡模块 (TTM)**：为了利用历史帧信息，TTM 提取记忆库中的历史 Queries。由于历史帧与当前帧处于不同的去噪阶段（噪声水平不匹配），TTM 通过 MLP 预测缩放和平移参数，将历史 Queries 强制对齐到当前的噪声流形中。
3. **统一扩散解码器**：在每一个去噪层中，Queries 首先通过自适应层归一化（AdaLN）注入当前噪声条件；随后通过时间交叉注意力融合 TTM 对齐后的历史特征；接着通过统一自注意力机制（Unified Self-Attention）让感知与规划 Queries 进行全连接交互；最后通过空间可变形注意力从图像特征中采样几何信息。
4. **锚点刷新策略 (ARS)**：在推理阶段的每次迭代后，模型评估预测结果的置信度。保留高置信度的 Queries 进入下一步去噪；将低置信度的 Queries 丢弃，并从原始噪声分布中重新采样，以此消除稀疏查询带来的训练-推理分布偏移。

```python
def UniTeD_forward_step(multi_view_images, historical_memory, current_noise_level):
    # 1. 提取多视角图像特征
    image_features = backbone(multi_view_images)
    
    # 2. 初始化联合任务的噪声锚点 (Agent, Map, Planning)
    noisy_anchors = sample_gaussian_noise(anchor_priors, current_noise_level)
    queries = anchor_embedding(noisy_anchors)
    
    # 3. 时间过渡模块 (TTM)：对齐历史特征的噪声水平
    aligned_historical_queries = []
    for hist_query, hist_condition in historical_memory:
        scale, shift = MLP(hist_condition)
        # 将历史 Query 投影到当前噪声流形
        aligned_q = (1 + scale) * LayerNorm(hist_query) + shift 
        aligned_historical_queries.append(aligned_q)
        
    # 4. 统一扩散解码器：迭代去噪与跨任务交互
    for layer in decoder_layers:
        # 注入当前噪声条件
        queries = AdaLN(queries, current_noise_level)
        # 融合对齐后的时序上下文
        queries = temporal_cross_attention(queries, aligned_historical_queries)
        # 核心：感知与规划 Query 的全连接双向交互
        queries = unified_self_attention(queries)
        # 聚合空间图像特征
        queries = spatial_deformable_attention(queries, image_features)
        
    # 5. 任务特定头输出
    agent_preds, map_preds, plan_preds, confidences = task_heads(queries)
    
    return agent_preds, map_preds, plan_preds, confidences

def inference_with_ARS(multi_view_images, steps=2):
    queries = init_noisy_anchors()
    for t in range(steps, 0, -1):
        preds, confidences = UniTeD_forward_step(multi_view_images, memory, t)
        
        # 锚点刷新策略 (ARS)：缓解分布偏移
        for i in range(len(queries)):
            if confidences[i] > threshold:
                # 高置信度：正常执行 DDIM 去噪
                queries[i] = DDIM_step(queries[i], preds[i], t)
            else:
                # 低置信度：丢弃预测，从原始分布重新采样噪声
                queries[i] = resample_noise(anchor_priors[i], t-1)
                
    return preds
```

### 第二模块：核心架构解释

G2DP 的系统架构由两个并行分支和一个推理期耦合机制组成。

**1. 扩散轨迹生成分支 (Diffusion-Based Trajectory Generation)**
模型采用 DiT（Diffusion Transformer）作为主干网络。输入自车当前状态、周围环境的向量化上下文（邻接车辆、静态物体、车道线），DiT 学习将高斯噪声逐步去噪为符合人类驾驶分布的未来轨迹。

**2. 密集代价网格构建分支 (Cost Grid Map Construction)**
采用一个冻结的 U-Net 网络，输入鸟瞰图（BEV）格式的历史动态和静态地图，预测未来多个时间步的占用概率网格（Occupancy Grids）。随后，将该占用网格与基于导航路线生成的进度期望图（Route-Aligned Progress）进行加权融合，形成一个包含碰撞风险和导航意图的时空代价体积。

**3. 基于梯度的去噪引导 (Gradient-Based Denoising Steering)**
在推理阶段，对于当前去噪步生成的轨迹样本，系统会根据自车的几何尺寸（矩形轮廓）在代价体积中进行双线性采样。为了避免单像素噪声干扰，系统提取轮廓内代价最高的 K 个点计算平均值（Top-K Aggregation），作为该时间步的能量代价。将所有时间步的代价累加得到总能量泛函，对其求导获得引导梯度，并在 DPM-Solver 的最后几个去噪步中注入该梯度，修正轨迹坐标。

**核心伪代码实现**：

```python
import torch

def compute_trajectory_energy(trajectory, cost_volume, K=15):
    """
    计算轨迹在时空代价体积下的能量泛函
    trajectory: [T, 4] (x, y, cos_theta, sin_theta)
    cost_volume: [T, H, W] 融合了占用概率和路线进度的代价网格
    """
    total_energy = 0.0
    T = trajectory.shape[0]
    
    for t in range(T):
        pose = trajectory[t]
        # 根据自车位姿获取矩形轮廓内的网格坐标
        footprint_coords = get_vehicle_footprint(pose) 
        
        # 可微双线性采样获取轮廓内的代价值
        costs = grid_sample(cost_volume[t], footprint_coords) 
        
        # Top-K 聚合：选取代价最高的 K 个点求均值
        topk_costs, _ = torch.topk(costs, K)
        step_energy = torch.mean(topk_costs)
        
        total_energy += step_energy
        
    return total_energy

def guided_diffusion_inference(dit_model, cost_volume, context, steps=10, guide_scale=0.5):
    """
    带有网格引导的扩散模型推理过程
    """
    # 初始化纯噪声轨迹
    xt = torch.randn(T, 4, requires_grad=True) 
    
    for step in range(steps, 0, -1):
        # 1. DiT 预测无约束的去噪分布 (Score)
        unconditional_score = dit_model(xt, context, step)
        
        # 2. 仅在最后阶段 (如 step 8-9) 注入引导梯度
        if step in [2, 1]: # 对应论文中的倒数第 2 和第 1 步 (Steps 8-9 out of 10)
            with torch.enable_grad():
                energy = compute_trajectory_energy(xt, cost_volume)
                # 计算能量泛函对轨迹状态的解析梯度
                guidance_gradient = torch.autograd.grad(energy, xt)[0] 
            
            # 3. 修正 Score
            guided_score = unconditional_score - guide_scale * guidance_gradient
        else:
            guided_score = unconditional_score
            
        # 4. DPM-Solver 更新轨迹
        xt = dpm_solver_update(xt, guided_score, step)
        
    return xt
```

MAPL 框架的运行逻辑分为两个异步迭代的阶段：偏好打分模型训练（低频）与策略训练（高频）。

**1. 轨迹采样与预处理**

**4. 势能差分奖励计算与策略更新**

### 第二模块：核心架构解释

ASSCG 的核心思想是为现有的“快慢双系统”自动驾驶架构外挂一个智能的“大脑门卫”。该架构的运行流程如下：

1.  **状态输入与记忆处理**：系统在每一帧接收当前的环境观测数据。为了处理长时序依赖而不引起算力爆炸，作者选用了 RWKV（一种兼具 RNN 推理效率和 Transformer 长上下文能力的架构）作为门控网络的骨干。
2.  **帧级别门控决策（Gating Decision）**：RWKV 门控网络根据当前观测和历史隐状态，输出三个离散动作之一：
    *   **Query（查询）**：唤醒慢系统（LLM），生成新的高级驾驶策略。
    *   **Cache（复用）**：保持慢系统休眠，直接提取上一次 LLM 生成的策略。
    *   **Drop（丢弃）**：保持慢系统休眠，且主动屏蔽历史策略，完全交由底层快系统（如轻量级 ViT）进行本能反应式驾驶。
3.  **计算感知强化学习（Compute-aware RL）**：该门控网络并非仅靠监督学习（SFT）训练，而是进一步采用了类似 GRPO 的强化学习算法。奖励函数（Reward）被设计为同时包含“驾驶表现”和“计算延迟/资源消耗”，迫使模型学会在安全与算力之间寻找最优解。

**核心逻辑伪代码（Python 风格）**：

```python
class ASSCG_System:
    def __init__(self, fast_planner, slow_llm_planner):
        self.fast_planner = fast_planner          # 底层轻量级快速规划器 (e.g., ViT)
        self.slow_llm_planner = slow_llm_planner  # 顶层重型慢速规划器 (e.g., LLM)
        self.gate_network = RWKV_Gate()           # RWKV门控网络
        self.cached_slow_plan = None              # 慢系统策略缓存

    def step(self, observation, rnn_hidden_state):
        # 1. 门控网络进行序列决策，更新隐状态
        gate_action, next_hidden_state = self.gate_network.forward(observation, rnn_hidden_state)
        
        # 2. 执行 Query / Cache / Drop 逻辑
        if gate_action == "Query":
            # 耗时操作：调用大模型生成新策略，并更新缓存
            active_slow_plan = self.slow_llm_planner.generate(observation)
            self.cached_slow_plan = active_slow_plan
            
        elif gate_action == "Cache":
            # 零延迟：直接复用历史策略
            active_slow_plan = self.cached_slow_plan
            
        elif gate_action == "Drop":
            # 零延迟且抑制干扰：丢弃慢系统指导，依赖快系统本能
            active_slow_plan = None
            self.cached_slow_plan = None 

        # 3. 快系统结合慢系统指导（如果有），输出最终控制指令
        final_control_action = self.fast_planner.plan(observation, active_slow_plan)
        
        return final_control_action, next_hidden_state
```

- **最能证明有效性的数据**：在 i-THOR 和 RoboTHOR 上的 Zero-shot（零样本）泛化成功率和 SPL（路径长度加权成功率）达到 SOTA。这直接证明了 LLM 提供的常识推理与场景图提供的空间拓扑结合，能有效克服未知环境的分布偏移。
- **存疑的试验结果**：论文强调该架构保持了物理机器人部署所需的“低控制延迟（low control latency）”。但实验仅在仿真器中进行，缺乏 Sim-to-Real 数据。在真实物理环境中，LLM 的异步低频更新与高频底层控制之间极易产生“状态脱节”（例如突然出现的动态障碍物），仅靠底层 GAFN 的融合是否足以保证物理安全，数据支撑不足。

### 第二模块：核心架构解释

SAGE-Nav 采用典型的分层异步架构，将导航任务拆解为“慢思考（全局规划）”与“快反应（局部控制）”两个并行流：

1. **异步全局规划 (Asynchronous Global Planning)**：LLM 作为全局规划器，独立于高频控制循环运行。它接收人类的自然语言指令和当前构建的动态场景图文本描述，将其分解为一系列具体的语义路点（Waypoints）。
2. **层次化场景图编码 (HSGE, Hierarchical Scene Graph Encoder)**：将环境中的物体、区域及其空间关系构建为动态场景图。使用关系图卷积网络（R-GCN）对该图进行编码，生成既包含物体语义又保留空间拓扑结构的结构化嵌入（Embeddings）。
3. **目标感知对齐融合 (GAFN, Goal-aware Alignment-Fusion Network)**：这是高频控制循环的核心。它接收当前的实时视觉观测特征和 HSGE 提供的场景图先验特征。通过内部的自适应门控机制（Adaptive Gating），网络能够根据当前子目标，动态决定是更依赖眼前的视觉信息（如避障），还是更依赖全局的拓扑先验（如跨房间寻路）。
4. **底层策略 (Low-level Policy)**：基于 GAFN 融合后的多模态对齐特征，输出具体的离散或连续导航动作。

**核心逻辑伪代码**：

```python
class SAGENav:
    def __init__(self, llm_planner, hsge, gafn, low_level_policy):
        self.llm = llm_planner          # 异步全局规划器
        self.hsge = hsge                # 层次化场景图编码器
        self.gafn = gafn                # 目标感知对齐融合网络
        self.policy = low_level_policy  # 底层动作策略
        self.scene_graph = DynamicSceneGraph()
        self.waypoints = []

    def async_planning_thread(self, instruction):
        """低频异步线程：LLM进行长视野规划"""
        while task_is_running:
            # 基于当前场景图上下文，LLM生成/更新语义路点
            context = self.scene_graph.get_text_context()
            self.waypoints = self.llm.decompose_to_waypoints(instruction, context)
            sleep(planning_interval)

    def high_freq_control_loop(self, visual_obs):
        """高频同步线程：底层反应式控制"""
        # 1. 更新动态场景图
        self.scene_graph.update_from_vision(visual_obs)
        
        # 2. HSGE: 提取空间与语义拓扑特征
        graph_embeddings = self.hsge.encode(self.scene_graph)
        
        # 3. 获取当前需要前往的子目标
        current_goal = self.get_current_waypoint(self.waypoints)
        
        # 4. GAFN: 动态门控融合实时视觉与全局拓扑先验
        fused_features = self.gafn.fuse(
            vision=visual_obs, 
            graph_prior=graph_embeddings, 
            goal=current_goal
        )
        
        # 5. Policy: 输出导航动作
        action = self.policy.predict_action(fused_features)
        return action
```

### 第三模块：核心学术问答

Q1: 这篇论文试图解决什么核心问题？

A1: 解决目标导航（ObjNav）任务中，现有端到端模型在长视野（long-horizon）推理能力不足，以及在未见过的全新环境中泛化能力差的问题。同时试图解决引入大模型后导致的控制延迟过高的问题。

Q2: 作者提出的核心解决方案（创新点）是什么？

A2: 提出 SAGE-Nav 层次化框架，核心在于“异步解耦”与“动态融合”。将低频的 LLM 全局语义规划与高频的底层反应式控制在时间轴上解耦；并在空间特征上，设计 HSGE 提取场景图拓扑特征，结合 GAFN 的自适应门控机制，将实时视觉感知与全局图先验进行动态对齐与融合。

Q3: 实验设计的关键支撑（或巧妙之处/数据集特点）是什么？

A3: 依托 i-THOR 和 RoboTHOR 这两个高度交互且视觉复杂的室内仿真环境。实验设计的关键支撑在于重点评估模型在 Zero-shot（零样本）条件下的泛化能力，以此验证 LLM 常识与场景图拓扑结合后，对完全未知环境布局的鲁棒性。

Q4: 这篇论文最显著的贡献点（Top 3）是什么？

A4:
1. 提出了一种异步层次化导航框架，成功将 LLM 的高级推理能力与底层高频控制解耦，缓解了具身智能中的算力与延迟瓶颈。
2. 设计了 HSGE（层次化场景图编码器），利用关系图卷积有效保留了环境的语义与空间拓扑结构，为底层控制提供结构化先验。
3. 开发了 GAFN（目标感知对齐融合网络），通过显式归纳偏置的自适应门控机制，实现了高频控制流中多模态感知与结构先验的鲁棒对齐。

Q5: 这项工作在哪些相关研究的基础上做了推进？（列举1-2个最核心的前置工作即可）

A5:
1. 基于 LLM 的具身规划研究（如 SayCan 或 LLM-Planner）：将其从单次静态规划推进为与动态场景图结合的异步持续规划。
2. 基于拓扑图/场景图的导航方法（如 VTN 或 HO-Nav）：推进了图特征与实时视觉特征在底层控制中的动态门控融合机制，而非仅仅将图用于寻路算法。

Q6: 客观评价：这篇论文的局限性、漏洞或"未讲明的故事"是什么？对后续科研有何启发？

A6: 核心漏洞在于过度依赖纯静态或准静态的仿真环境，缺乏真实物理机器人（Sim-to-Real）的部署验证。论文声称的“低控制延迟”是建立在异步解耦的系统工程假设上。但在真实物理世界中，环境是高度动态的（如突然移动的人）。低频更新的 LLM 规划与高频底层控制之间存在时间差，这种“异步”极易导致全局规划与局部状态发生致命的“语义错位”。对后续科研的启发是：在异步 VLA 架构中，必须在底层控制环路中引入轻量级的动态障碍物预测机制，或设计一种基于不确定性的“规划中断/重置”触发机制，以弥补异步更新带来的时间差风险。
