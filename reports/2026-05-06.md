    def __init__(self):
        # Define unified action space (Translation & Rotation)
        self.valid_actions = 81 
        self.difficulty_levels = [1, 2, 3, 4]

    def translate_action(self, unified_action_id, target_modality):
        # Core Innovation: Map unified ID to heterogeneous inputs
        if target_modality == "text":
            return self._get_text_prompt(unified_action_id)
        elif target_modality == "one-hot":
            return self._get_keyboard_signal(unified_action_id)
        elif target_modality == "camera_params":
            return self._get_extrinsic_matrix(unified_action_id)

        current_context = initial_state
        generated_frames = []

### 第二模块：核心架构解释

GLANCE 框架的核心逻辑是构建一个“思考策略（Thinking Policy）”与“见证编码器（Witnessing Encoder）”的自监督博弈闭环。其架构包含以下关键流程：

1.  **在线 VLM 智能体 (Online VLM Agent):** 接收当前视觉观察 $o_t$，通过冻结的 LLM 主干生成包含未来状态预测的 CoT 推理序列，并输出动作 $a_t$。提取最后一个预测 Token（`</pred>`）的隐状态，通过一个轻量级投影器（Projector）将其映射到视觉潜空间，得到预测的视觉表征 $\hat{y}_{t+1}$。
2.  **动量目标网络 (Momentum Target Network):** 智能体执行动作后，环境返回真实的下一帧观察 $o_{t+1}$。目标网络（与在线视觉编码器结构相同，但参数通过指数移动平均 EMA 缓慢更新）处理 $o_{t+1}$，输出真实的视觉表征 $y_{t+1}$。此过程截断梯度（Stop-gradient）。
3.  **好奇心驱动与表征对齐 (Curiosity & Alignment):** 计算 $\hat{y}_{t+1}$ 与 $y_{t+1}$ 之间的均方误差（MSE）作为预测损失 $\mathcal{L}_{explore}$。该损失乘以系数 $\beta$ 后作为内在奖励（Intrinsic Reward）附加到环境的稀疏奖励上，用于 PPO 策略更新；同时，该损失的梯度回传，仅更新投影器和在线视觉编码器，迫使视觉编码器学习符合语言逻辑的特征。
4.  **课程重置机制 (Curriculum Rejuvenation):** 为了防止轻量级投影器过快拟合导致预测误差降为零（好奇心枯竭），系统会监控损失的变化率。当损失在 $K$ 步内低于阈值时，随机重置投影器的权重。这会人为制造误差尖峰，迫使智能体重新审视熟悉场景中的细粒度视觉差异。

### 第二模块：核心架构解释

NEO 的核心架构是一个概率神经模型，旨在最大化条件似然 $p_\theta(y|x)$。其工作流程分为编码、程序生成与执行、长度选择以及状态正则化四个主要阶段：

1. **状态编码 (State Encoding)**：使用预训练的编码器将源观测 $x$ 和目标观测 $y$ 映射到隐空间，得到初始状态 $s_1$ 和目标状态 $s_y$。
2. **理论程序员与执行器 (Theory Programmer & Executor)**：
   - **程序员**：一个目标条件策略网络 $q_\phi(z_{i_k} | s_k, s_y)$，根据当前状态和目标状态，从离散的密码本（Codebook）中输出下一个基础图元（Primitive）的分类分布。
   - **执行器**：一个共享的确定性状态转移模型 $f_\theta(s_k, z_{i_k})$，将选定的图元应用于当前状态，生成下一个隐状态 $s_{k+1}$。该过程自回归执行 $K$ 步。
3. **最小描述长度选择 (MDL Principle)**：在展开的 $K$ 步中，计算每一步状态解码后的重建误差。通过引入指数级长度惩罚项 $\lambda_{\text{MDL}}$，模型在所有能成功重建 $y$ 的中间步中，选择步数最短的 $k^*$ 作为最终解释。反向传播仅截断至第 $k^*$ 步。
4. **状态接地 (State Grounding)**：为了防止多步执行时中间隐状态漂移到无效空间，对每一步的隐状态 $s_k$ 进行“解码-再编码”操作，并计算其与原 $s_k$ 的 L2 距离作为惩罚损失，强制中间状态停留在有效的数据流形上。

**核心逻辑伪代码 (Python 风格)**:

```python
def train_neo_step(x, y, encoder, decoder, programmer, executor, K, lambda_mdl, lambda_vq, lambda_state):
    # 1. 编码观测数据
    s_1 = encoder(x)
    s_y = encoder(y)
    
    s_k = s_1
    states = [s_1]
    reconstruction_losses = []
    
    # 2. 程序生成与执行 (展开 K 步)
    for k in range(1, K + 1):
        # 理论程序员选择图元 (VQ-VAE 离散化)
        z_k, vq_loss = programmer.sample_primitive(s_k, s_y)
        
        # 执行器进行状态转移
        s_k = executor(s_k, z_k)
        states.append(s_k)
        
        # 计算当前步的重建误差
        y_hat_k = decoder(s_k)
        rec_loss = mse_loss(y_hat_k, y)
        reconstruction_losses.append(rec_loss)
        
    # 3. 最小描述长度 (MDL) 选择
    # 寻找使得 (lambda_mdl^k * rec_loss) 最小的步数 k*
    mdl_costs = [ (lambda_mdl ** k) * reconstruction_losses[k-1] for k in range(1, K + 1) ]
    k_star = argmin(mdl_costs)
    
    # 最终的重建损失仅取第 k* 步
    final_rec_loss = reconstruction_losses[k_star]
    
    # 4. 状态接地损失 (State Grounding)
    # 强制中间状态 s_k 逼近 encoder(decoder(s_k))，使用 stop_gradient 防止梯度流向解码器/编码器
    state_loss = 0
    for s in states[1:k_star+1]:
        grounded_s = stop_gradient(encoder(decoder(s)))
        state_loss += mse_loss(s, grounded_s)
        
    # 总损失
    total_loss = final_rec_loss + lambda_vq * vq_loss + lambda_state * state_loss
    
    # 反向传播与优化
    total_loss.backward()
    optimizer.step()
    
    return total_loss
```

### 第二模块：核心架构解释

RLDX-1 的系统架构和实验流程可以拆解为“感知-记忆-动作生成”三位一体的模型设计，以及“预训练-微调-部署”的工程闭环。

**1. 模型架构流程**
*   **VLM Backbone (感知与记忆):** 基于 Qwen3-VL。为了捕捉动态，在视觉编码器中插入了时空自相似性模块（STSS）；为了长时记忆，在 VLM 输出端维护了一个 Memory Queue，缓存过去时间步的 Cognition Tokens（认知特征）。
*   **Action Model (动作与物理预测):** 核心是 Multi-Stream Action Transformer (MSAT)。它采用 Flow-Matching 目标函数。输入被分为三个流：Cognition Stream (C, 包含视觉/语言/记忆)、Action Stream (A, 包含本体状态和加噪动作)、Physics Stream (P, 包含力矩/触觉信号)。各流独立进行 LayerNorm 和 QKV 映射，随后拼接进行联合自注意力计算（Joint Self-Attention），最后拆分回各自的流进行残差更新。
*   **Inference Optimization (极致推理):** 针对真实机器人高频控制需求，通过静态图转换（Static Graph Conversion）消除 CUDA Launch 开销，并手写 Triton 算子融合（Kernel Fusion），将推理延迟从 71.2ms 压榨至 43.7ms。

**2. 核心伪代码实现逻辑**

import torch
import torch.nn as nn

class MultiStreamActionTransformer(nn.Module):
    def __init__(self):
        super().__init__()
        # 独立流的投影层
        self.norm_c = nn.LayerNorm(dim)
        self.norm_a = nn.LayerNorm(dim)
        self.norm_p = nn.LayerNorm(dim)
        
        self.qkv_c = nn.Linear(dim, dim * 3)
        self.qkv_a = nn.Linear(dim, dim * 3)
        self.qkv_p = nn.Linear(dim, dim * 3)
        
        self.joint_attn = nn.MultiheadAttention(embed_dim=dim, num_heads=heads)
        
    def forward(self, cognition_feat, noisy_action, physics_feat, timestep):
        # 1. 独立流的归一化与 QKV 映射
        qkv_c = self.qkv_c(self.norm_c(cognition_feat))
        qkv_a = self.qkv_a(self.norm_a(noisy_action))
        
        # 物理流是可选的 (应对无触觉传感器的机器人)
        if physics_feat is not None:
            qkv_p = self.qkv_p(self.norm_p(physics_feat))
            # 2. 跨模态拼接
            qkv_joint = torch.cat([qkv_c, qkv_a, qkv_p], dim=1)
        else:
            qkv_joint = torch.cat([qkv_c, qkv_a], dim=1)
            
        # 3. 联合自注意力计算 (Cross-modal interaction)
        q, k, v = qkv_joint.chunk(3, dim=-1)
        attn_out, _ = self.joint_attn(q, k, v)
        
        # 4. 拆分回独立流并输出
        if physics_feat is not None:
            out_c, out_a, out_p = torch.split(attn_out, [len_c, len_a, len_p], dim=1)
            return out_a, out_p # 同时预测去噪动作和未来物理信号
        else:
            out_c, out_a = torch.split(attn_out, [len_c, len_a], dim=1)
            return out_a, None

# Flow-Matching 训练逻辑
def compute_flow_matching_loss(model, clean_action, clean_physics, cognition_feat):
    # 采样时间步和噪声
    tau = torch.rand(batch_size)
    noise_a = torch.randn_like(clean_action)
    
    # 构造加噪动作
    noisy_action = tau * clean_action + (1 - tau) * noise_a
    
    if clean_physics is not None:
        noise_p = torch.randn_like(clean_physics)
        noisy_physics = tau * clean_physics + (1 - tau) * noise_p
    else:
        noisy_physics = None

    # 预测速度场 (Velocity Field)
    pred_v_a, pred_v_p = model(cognition_feat, noisy_action, noisy_physics, tau)
    
    # 计算目标速度
    target_v_a = clean_action - noise_a
    loss = mse_loss(pred_v_a, target_v_a)
    
    # 辅助物理预测损失
    if clean_physics is not None:
        target_v_p = clean_physics - noise_p
        loss += mse_loss(pred_v_p, target_v_p)
        
    return loss

### 第二模块：核心架构解释

本文作为综述，其“架构”体现为构建高效视频世界模型的通用技术流水线。一个典型的高效视频世界模型通常包含三个核心阶段的优化：

1.  **状态压缩与表征（对应高效建模范式）**：放弃在极高维度的原始像素空间直接建模，使用高效的 3D 自动编码器（如 Causal 3D VAE）将长视频压缩至紧凑的时空隐空间（Latent Space），并在此空间内定义动力学转移。
2.  **时空动力学推演（对应高效网络架构）**：在隐空间中预测未来状态时，摒弃全局的 3D 全注意力机制。采用时空解耦（Spatial-Temporal Factorization）、窗口注意力，或直接替换为具有线性时间复杂度的状态空间模型（SSM/Mamba），以极低的显存和算力消耗捕捉长视野因果关系。
3.  **动作条件注入与快速解码（对应高效推理算法）**：将具身智能体的动作（Action）作为条件注入网络。在推理阶段，采用并行计算（如并行自回归解码）或加速采样算法（如一致性模型 Consistency Models、LCM），将原本需要数十步的生成过程压缩至 1-4 步，实现极低延迟的闭环反馈。

**Python 风格伪代码（高效视频世界模型通用逻辑）**：

```python
class EfficientVideoWorldModel(nn.Module):
    def __init__(self, encoder, dynamics_model, decoder, fast_sampler):
        super().__init__()
        # 1. 高效表征：时空压缩自编码器
        self.encoder = encoder 
        # 2. 高效架构：如基于 Mamba 或时空解耦 Transformer 的动力学模型
        self.dynamics_model = dynamics_model 
        # 3. 高效解码器
        self.decoder = decoder
        # 4. 高效推理算法：如 LCM (Latent Consistency Model) 极少步采样器
        self.fast_sampler = fast_sampler

    def forward(self, past_frames, actions):
        """
        past_frames: [B, T_past, C, H, W] 历史观测视频
        actions: [B, T_future, D_action] 未来动作序列
        """
        # 步骤 1: 将高维像素视频压缩至低维时空隐变量 (降低计算维度)
        latent_states = self.encoder.encode(past_frames) # [B, T_past, C', H', W']
        
        # 步骤 2: 在隐空间中进行高效的时空动力学推演
        # 核心优化：dynamics_model 内部采用线性复杂度算子或解耦注意力
        predicted_latents = []
        current_latent = latent_states[:, -1]
        
        for t in range(actions.size(1)):
            action_t = actions[:, t]
            # 结合当前状态与动作，预测下一帧隐状态
            # 使用 fast_sampler 减少扩散模型的去噪步数 (如从 50 步降至 2 步)
            next_latent = self.fast_sampler.sample(
                model=self.dynamics_model, 
                condition=(current_latent, action_t),
                steps=2 
            )
            predicted_latents.append(next_latent)
            current_latent = next_latent
            
        predicted_latents = torch.stack(predicted_latents, dim=1)
        
        # 步骤 3: 将预测的隐状态解码回像素空间
        future_frames = self.decoder.decode(predicted_latents)
        
        return future_frames
```
