```python
# PhysInOne 核心逻辑伪代码

import numpy as np
from scipy.fft import fftn

class PhysInOnePipeline:
    def __init__(self):
        self.physics_engines = {
            'rigid_body': 'UE5_Chaos',
            'deformable': 'Taichi_MPM',
            'fluid': 'Doriflow_SPH'
        }
        
    def generate_scene(self, phenomena_list, assets, background):
        # 1. 组合物理现象与资产
        scene_config = self.setup_environment(phenomena_list, assets, background)
        
        # 2. 路由到对应的物理求解器进行仿真
        if 'fluid' in phenomena_list:
            simulation_data = self.run_sph_solver(scene_config)
        elif 'deformable' in phenomena_list:
            simulation_data = self.run_mpm_solver(scene_config)
        else:
            simulation_data = self.run_chaos_solver(scene_config)
            
        # 3. 渲染与多模态标注提取
        video_rgb = self.render_cameras(simulation_data)
        annotations = self.extract_ground_truth(simulation_data) # Depth, Mask, Trajectory, Params
        text_desc = self.generate_llm_caption(annotations)
        
        return video_rgb, annotations, text_desc

def calculate_pmf(video_gen, video_ref):
    """
    计算物理运动保真度 (Physical Motion Fidelity)
    video_gen, video_ref shape: (T, H, W, C)
    """
    pmf_scores = []
    for c in range(3): # 遍历 RGB 通道
        # 1. 3D 离散傅里叶变换
        fft_gen = fftn(video_gen[..., c])
        fft_ref = fftn(video_ref[..., c])
        
        # 2. 计算能量谱 (振幅的平方)
        energy_gen = np.abs(fft_gen) ** 2
        energy_ref = np.abs(fft_ref) ** 2
        
        # 3. 归一化能量谱
        norm_energy_gen = energy_gen / np.sum(energy_gen)
        norm_energy_ref = energy_ref / np.sum(energy_ref)
        
        # 4. 计算总变差 (Total Variation) 距离
        tv_distance = 0.5 * np.sum(np.abs(norm_energy_gen - norm_energy_ref))
        
        # 5. 转换为 PMF 分数 (负对数)
        pmf_scores.append(-np.log(tv_distance + 1e-8))
        
    return np.mean(pmf_scores)
```

```python
class EgoTLPipeline:
    def __init__(self, asr_model, spatial_estimator):
        self.asr_model = asr_model # e.g., WhisperX
        self.spatial_estimator = spatial_estimator # e.g., MapAnything
        
    def process_episode(self, video_stream, audio_stream):
        # 1. 词级语音对齐提取思维链 (CoT)
        transcripts = self.asr_model.transcribe_with_timestamps(audio_stream)
        
        clips = []
        # 2. 基于静默期进行片段切分
        for segment in self.split_by_silence(transcripts, threshold_sec=2.0):
            clip_video = video_stream.extract(segment.start_time, segment.end_time)
            intent_text = segment.text
            
            # 3. 动作分类与度量级空间校准
            if self.is_manipulation(intent_text):
                clip_type = "Manipulation"
                metric_data = None
            elif self.is_navigation(intent_text):
                clip_type = "Navigation"
                # 计算真实的物理行走距离
                if "walk straight" in intent_text.lower():
                    trajectory = self.spatial_estimator.get_3d_trajectory(clip_video)
                    metric_data = self.calculate_distance(trajectory)
                else:
                    metric_data = self.extract_direction(intent_text)
                    
            clips.append({
                "video": clip_video,
                "intent_cot": intent_text,
                "type": clip_type,
                "metric_label": metric_data
            })
            
        return clips

    def calculate_distance(self, trajectory):
        # 依据公式 (3) 计算轨迹欧氏距离总和
        distance = 0.0
        for i in range(1, len(trajectory)):
            distance += np.linalg.norm(trajectory[i] - trajectory[i-1])
        return distance

def evaluate_distance_estimation(vlm_model, clip, ground_truth_distance):
    prompt = "Estimate the approximate distance traveled in meters. Output only a number."
    prediction = vlm_model.generate(clip.video, prompt)
    pred_dist = float(prediction)
    
    # 计算 Mean Relative Accuracy (MRA)
    mra_score = 0.0
    thresholds = np.arange(0.5, 1.0, 0.05)
    for theta in thresholds:
        relative_error = abs(pred_dist - ground_truth_distance) / ground_truth_distance
        if relative_error < (1 - theta):
            mra_score += 1.0
    
    return mra_score / len(thresholds)
```

### 第二模块：核心架构解释

本文的核心架构旨在构建一个硬件无关的四足世界模型（QWM），其工作流程与实验设计可拆解为以下几个关键步骤：

1.  **形态学特征提取 (Morphology Feature Extraction)**：提取目标机器人的精确工程规格，包括但不限于连杆长度、质量分布、惯性张量以及电机动力学参数。
2.  **物理形态编码 (Physical Morphology Encoding)**：将上述高维且物理量纲不一的静态参数，通过神经网络（如 MLP）映射为低维的形态学潜变量（Morphology Latent），为后续的动力学预测提供紧凑的条件输入。
3.  **条件化世界模型 (Conditioned World Model)**：在标准的时序状态空间模型（如 RSSM）基础上进行改造。模型在预测下一帧状态时，不仅依赖当前状态和动作，还必须严格受控于形态学潜变量。
4.  **奖励归一化 (Reward Normalization)**：由于不同体型的机器人在相同任务下的绝对物理表现不同（例如大型犬和微型犬的绝对奔跑速度上限差异巨大），引入奖励归一化机制对齐不同形态下的策略学习目标，防止策略优化过程崩溃。
5.  **零样本策略学习 (Zero-shot Policy Learning)**：在训练好的 QWM（神经模拟器）内部，通过“想象（Imagination）”训练强化学习策略。部署时，只需输入新机器人的形态参数，即可直接输出控制指令。

**核心逻辑伪代码 (Python-style):**

```python
import torch
import torch.nn as nn

class QuadrupedalWorldModel(nn.Module):
    def __init__(self, state_dim, action_dim, morph_dim, latent_dim):
        super().__init__()
        # 1. 物理形态编码器：将工程规格转化为潜变量
        self.morph_encoder = nn.Sequential(
            nn.Linear(morph_dim, 128),
            nn.ReLU(),
            nn.Linear(128, latent_dim)
        )
        
        # 2. 条件化动力学模型：基于状态、动作和形态预测未来
        self.dynamics_net = ConditionedRSSM(state_dim, action_dim, cond_dim=latent_dim)
        
        # 3. 奖励预测器：包含奖励归一化逻辑
        self.reward_net = NormalizedRewardPredictor(state_dim, cond_dim=latent_dim)

    def forward(self, state, action, morph_params):
        # 显式编码机器人的静态物理属性
        morph_latent = self.morph_encoder(morph_params)
        
        # 形态条件化的动力学状态转移预测
        next_state_prior = self.dynamics_net.predict_next(state, action, morph_latent)
        
        # 形态条件化的奖励预测
        reward = self.reward_net(next_state_prior, morph_latent)
        
        return next_state_prior, reward

def zero_shot_deployment(qwm, policy, new_robot_morph_params, current_state):
    """在新硬件上进行零样本部署"""
    with torch.no_grad():
        # 提取新机器人的形态特征
        morph_latent = qwm.morph_encoder(new_robot_morph_params)
        
        # 策略网络根据当前状态和新机器人的形态特征输出动作
        action = policy.act(current_state, morph_latent)
        
    return action
```

### 第二模块：核心架构解释

LMGenDrive 的架构设计本质上是一个“感知-推理-想象-行动”的联合计算图。其核心流程分为三个子系统：

1. **视觉编码器 (Vision Encoder):** 采用 ResNet 提取多视角图像特征，通过 Transformer 融合并结合 BEV 位置编码，生成 BEV、航点和红绿灯的视觉 Token。为了防止 LLM 显存爆炸，使用 Q-Former 将每帧约 2k 的 Token 压缩为紧凑的帧级特征。
2. **大语言模型中枢 (LLM Brain):** 接收文本指令 Token、压缩后的视觉 Token、历史 Token，以及两组可学习的 Query（Action Query 和 World Query）。LLM 负责进行跨模态对齐与因果推理。
3. **多视角世界生成器 (Multi-View World Generator):** 这是一个基于 U-Net 的扩散模型。它接收 LLM 输出的 World Embeddings 作为跨注意力条件，同时接收当前帧的 CLIP 图像特征作为外观先验，去噪生成未来多视角的驾驶视频。

```python
    def __init__(self):
        self.vision_encoder = PretrainedVisionEncoder()
        self.q_former = QFormer(num_queries=8)
        self.llm = LLaMA_Model()
        self.world_generator = DiffusionUNet()
        
        # Learnable queries
        self.action_queries = nn.Parameter(torch.randn(N, D))
        self.world_queries = nn.Parameter(torch.randn(M, D))
        self.action_head = MLP()

    def forward(self, multi_view_images, text_instruction, mode="train"):
        # 1. Perception & Compression
        raw_visual_tokens = self.vision_encoder(multi_view_images)
        compressed_visual_tokens = self.q_former(raw_visual_tokens)
        
        text_tokens = tokenize(text_instruction)
        
        # 2. LLM Reasoning
        llm_inputs = concat([
            text_tokens, 
            compressed_visual_tokens, 
            self.action_queries, 
            self.world_queries
        ])
        
        llm_outputs = self.llm(llm_inputs)
        
        # Extract specific outputs based on query positions
        action_embeddings = extract_action_features(llm_outputs)
        world_embeddings = extract_world_features(llm_outputs)
        
        # 3. Action Prediction
        predicted_waypoints = self.action_head(action_embeddings)
        
        # 4. World Generation (Imagination)
        if mode == "train" or mode == "offline_generation":
            clip_features = extract_clip_features(multi_view_images)
            # Diffusion process conditioned on LLM world understanding
            future_videos = self.world_generator(
                noise=torch.randn_like(target_video),
                condition=world_embeddings,
                appearance_prior=clip_features
            )
            return predicted_waypoints, future_videos
            
        elif mode == "online_planning":
            # Discard diffusion model for low latency
```

### 第二模块：核心架构解释

VLA-World 的核心架构将自动驾驶建模为一个联合优化问题：同时优化策略分布（决策）和环境动态分布（世界模型）。其运行流程与三阶段训练范式如下：

**运行流程：**
1.  **感知 (Perception)：** 接收多视角图像和自车状态，输出场景级结构化描述（如障碍物位置、道路边界）。
2.  **短期预测 (Short-term Prediction)：** 基于历史状态，预测未来 0.5 秒的自车轨迹和行驶方向。
3.  **条件引导生成 (Imagination)：** 将短期预测轨迹作为条件，自回归地生成 0.5 秒后的未来帧视觉 Token。
4.  **视觉反思 (Think)：** 模型对生成的未来帧进行分析，评估潜在的碰撞风险和安全裕度。
5.  **动作与规划 (Action & Planning)：** 基于反思结果，输出高层驾驶指令，并回归出修正后的未来 3 秒精确轨迹。

**训练范式：**
1.  **视觉预训练 (Visual Pretraining)：** 仅训练视觉生成能力，让模型内化物理世界的时空演变规律。
2.  **监督微调 (SFT)：** 在构建的 nuScenes-GR-20K 数据集上进行多任务混合微调，注入“生成-思考-规划”的概念知识。
3.  **强化学习 (RL via GRPO)：** 冻结参考模型，使用基于规则的奖励函数（格式、预测、视觉约束、动作、轨迹）进行无价值网络（Value-free）的群体相对策略优化，激发模型的自我纠错与逻辑一致性。

```python
    def __init__(self, vlm_backbone):
        super().__init__()
        self.vlm = vlm_backbone # e.g., Qwen2-VL

        prompt_3 = self.build_reflection_prompt(future_visual_tokens)
        reflection_thought, final_action, long_term_traj = self.vlm.generate_text(prompt_3)

def compute_grpo_rewards(rollouts, ground_truth):
    rewards = []
    for rollout in rollouts:

        r_act = compute_f1_score(rollout.final_action, ground_truth.action)
        r_traj = compute_kinematic_consistency_and_l2(rollout.long_term_traj, ground_truth)

        rewards.append(total_reward)
        
    # Group relative normalization
    mean_r = mean(rewards)
    std_r = std(rewards)
    advantages = [(r - mean_r) / (std_r + 1e-8) for r in rewards]
    return advantages
```

### 第二模块：核心架构解释

**模型架构与流程解析：**

1.  **统一表征构建 (Raxels Representation)：**
    传统的相机位姿通常表示为旋转矩阵和平移向量（$R, t$）以及内参（$K$）。该模型将每个像素对应的相机光线（起点和方向）计算出来，形成与视频帧空间分辨率一致的密集张量，称为 Raxels。这使得相机参数与图像像素在空间维度上完全对齐。
2.  **联合扩散过程 (Joint Diffusion Process)：**
    模型不再单独对视频加噪，而是将视频帧序列 $V$ 和对应的 Raxels 序列 $R_x$ 视为一个联合变量 $(V, R_x)$。在正向扩散过程中，同时向视频和 Raxels 添加高斯噪声。
3.  **解耦自-交叉注意力机制 (Decoupled Self-Cross Attention)：**
    在去噪 U-Net 或 Transformer 内部，为了处理视觉特征和几何特征的异构性，模型采用了解耦的注意力机制。视频特征和 Raxels 特征首先各自进行 Self-Attention 提取内部时空依赖，然后通过 Cross-Attention 进行信息交互，使得视频生成的纹理受相机轨迹约束，同时相机轨迹的预测受视频内容引导。
4.  **多任务统一推理 (Unified Multi-task Inference)：**
    *   **任务 A (位姿估计)：** 输入清晰视频，对 Raxels 进行去噪，输出相机轨迹。
    *   **任务 B (联合生成)：** 输入单张/多张条件图像，同时对视频和 Raxels 进行去噪，输出新视频及对应的相机轨迹。
    *   **任务 C (控制生成)：** 输入条件图像和目标 Raxels（无噪声），仅对视频进行去噪，输出受控视频。

**核心逻辑伪代码：**

```python
class DecoupledSelfCrossAttention(nn.Module):
    def __init__(self, dim):
        super().__init__()
        self.video_self_attn = SelfAttention(dim)
        self.raxel_self_attn = SelfAttention(dim)
        self.video_cross_raxel = CrossAttention(dim)
        self.raxel_cross_video = CrossAttention(dim)

        # 1. 独立提取内部时空特征

        r_self = self.raxel_self_attn(raxel_feat)
        
        # 2. 跨模态特征交互 (相互作为 Key 和 Value)
        v_out = self.video_cross_raxel(query=v_self, key=r_self, value=r_self)
        r_out = self.raxel_cross_video(query=r_self, key=v_self, value=v_self)
        
        return v_out, r_out
```

### 第二模块：核心架构解释

PhysInOne 的核心架构分为数据生成流水线与物理评估指标两部分。

数据生成流水线采用了分层构建的逻辑：首先定义 71 种基础物理现象，随后在 UE5 中组合单/双/三物理现象构建复杂场景。为了保证物理真实性，系统调用了多个底层求解器：使用 UE5 的 Chaos Physics 处理刚体动力学与破坏，使用 Taichi (MPM) 处理弹性体、塑性体与颗粒物，使用 Doriflow (SPH) 处理流体。最终通过多相机阵列渲染出包含 RGB、深度、掩码、3D 轨迹及物理参数的完备数据，并利用 LLM 生成对齐的文本描述。

在评估层面，作者提出了 PMF (Physical Motion Fidelity) 指标。传统像素级指标（如 PSNR、FVD）极易受初始位置、光照和背景干扰。PMF 将视频转换到 3D 频域，利用傅里叶变换中“能量谱对时空平移不变”的特性，直接比较生成视频与参考视频在时空频率上的能量分布差异，从而纯粹地衡量运动动力学的相似度。

```python
# PhysInOne 核心逻辑伪代码

import numpy as np
from scipy.fft import fftn

class PhysInOnePipeline:
    def __init__(self):
        self.physics_engines = {
            'rigid_body': 'UE5_Chaos',
            'deformable': 'Taichi_MPM',
            'fluid': 'Doriflow_SPH'
        }
        
    def generate_scene(self, phenomena_list, assets, background):
        # 1. 组合物理现象与资产
        scene_config = self.setup_environment(phenomena_list, assets, background)
        
        # 2. 路由到对应的物理求解器进行仿真
        if 'fluid' in phenomena_list:
            simulation_data = self.run_sph_solver(scene_config)
        elif 'deformable' in phenomena_list:
            simulation_data = self.run_mpm_solver(scene_config)
        else:
            simulation_data = self.run_chaos_solver(scene_config)
            
        # 3. 渲染与多模态标注提取
        video_rgb = self.render_cameras(simulation_data)
        annotations = self.extract_ground_truth(simulation_data) # Depth, Mask, Trajectory, Params
        text_desc = self.generate_llm_caption(annotations)
        
        return video_rgb, annotations, text_desc

def calculate_pmf(video_gen, video_ref):
    """
    计算物理运动保真度 (Physical Motion Fidelity)
    video_gen, video_ref shape: (T, H, W, C)
    """
    pmf_scores = []
    for c in range(3): # 遍历 RGB 通道
        # 1. 3D 离散傅里叶变换
        fft_gen = fftn(video_gen[..., c])
        fft_ref = fftn(video_ref[..., c])
        
        # 2. 计算能量谱 (振幅的平方)
        energy_gen = np.abs(fft_gen) ** 2
        energy_ref = np.abs(fft_ref) ** 2
        
        # 3. 归一化能量谱
        norm_energy_gen = energy_gen / np.sum(energy_gen)
        norm_energy_ref = energy_ref / np.sum(energy_ref)
        
        # 4. 计算总变差 (Total Variation) 距离
        tv_distance = 0.5 * np.sum(np.abs(norm_energy_gen - norm_energy_ref))
        
        # 5. 转换为 PMF 分数 (负对数)
        pmf_scores.append(-np.log(tv_distance + 1e-8))
        
    return np.mean(pmf_scores)
```

```python
class EgoTLPipeline:
    def __init__(self, asr_model, spatial_estimator):
        self.asr_model = asr_model # e.g., WhisperX
        self.spatial_estimator = spatial_estimator # e.g., MapAnything
        
    def process_episode(self, video_stream, audio_stream):
        # 1. 词级语音对齐提取思维链 (CoT)
        transcripts = self.asr_model.transcribe_with_timestamps(audio_stream)
        
        clips = []
        # 2. 基于静默期进行片段切分
        for segment in self.split_by_silence(transcripts, threshold_sec=2.0):
            clip_video = video_stream.extract(segment.start_time, segment.end_time)
            intent_text = segment.text
            
            # 3. 动作分类与度量级空间校准
            if self.is_manipulation(intent_text):
                clip_type = "Manipulation"
                metric_data = None
            elif self.is_navigation(intent_text):
                clip_type = "Navigation"
                # 计算真实的物理行走距离
                if "walk straight" in intent_text.lower():
                    trajectory = self.spatial_estimator.get_3d_trajectory(clip_video)
                    metric_data = self.calculate_distance(trajectory)
                else:
                    metric_data = self.extract_direction(intent_text)
                    
            clips.append({
                "video": clip_video,
                "intent_cot": intent_text,
                "type": clip_type,
                "metric_label": metric_data
            })
            
        return clips

    def calculate_distance(self, trajectory):
        # 依据公式 (3) 计算轨迹欧氏距离总和
        distance = 0.0
        for i in range(1, len(trajectory)):
            distance += np.linalg.norm(trajectory[i] - trajectory[i-1])
        return distance

def evaluate_distance_estimation(vlm_model, clip, ground_truth_distance):
    prompt = "Estimate the approximate distance traveled in meters. Output only a number."
    prediction = vlm_model.generate(clip.video, prompt)
    pred_dist = float(prediction)
    
    # 计算 Mean Relative Accuracy (MRA)
    mra_score = 0.0
    thresholds = np.arange(0.5, 1.0, 0.05)
    for theta in thresholds:
        relative_error = abs(pred_dist - ground_truth_distance) / ground_truth_distance
        if relative_error < (1 - theta):
            mra_score += 1.0
    
    return mra_score / len(thresholds)
```

### 第二模块：核心架构解释

本文的核心架构旨在构建一个硬件无关的四足世界模型（QWM），其工作流程与实验设计可拆解为以下几个关键步骤：

1.  **形态学特征提取 (Morphology Feature Extraction)**：提取目标机器人的精确工程规格，包括但不限于连杆长度、质量分布、惯性张量以及电机动力学参数。
2.  **物理形态编码 (Physical Morphology Encoding)**：将上述高维且物理量纲不一的静态参数，通过神经网络（如 MLP）映射为低维的形态学潜变量（Morphology Latent），为后续的动力学预测提供紧凑的条件输入。

5.  **零样本策略学习 (Zero-shot Policy Learning)**：在训练好的 QWM（神经模拟器）内部，通过“想象（Imagination）”训练强化学习策略。部署时，只需输入新机器人的形态参数，即可直接输出控制指令。

```python
            nn.Linear(morph_dim, 128),
            nn.ReLU(),
            nn.Linear(128, latent_dim)
        )
        
        # 2. 条件化动力学模型：基于状态、动作和形态预测未来
        self.dynamics_net = ConditionedRSSM(state_dim, action_dim, cond_dim=latent_dim)
        
        # 3. 奖励预测器：包含奖励归一化逻辑
        self.reward_net = NormalizedRewardPredictor(state_dim, cond_dim=latent_dim)

    def forward(self, state, action, morph_params):
        # 显式编码机器人的静态物理属性
        morph_latent = self.morph_encoder(morph_params)
        
        # 形态条件化的动力学状态转移预测
        next_state_prior = self.dynamics_net.predict_next(state, action, morph_latent)
        
        # 形态条件化的奖励预测
        reward = self.reward_net(next_state_prior, morph_latent)
        
        return next_state_prior, reward

def zero_shot_deployment(qwm, policy, new_robot_morph_params, current_state):
    """在新硬件上进行零样本部署"""
    with torch.no_grad():
        # 提取新机器人的形态特征
        morph_latent = qwm.morph_encoder(new_robot_morph_params)
        
        # 策略网络根据当前状态和新机器人的形态特征输出动作
        action = policy.act(current_state, morph_latent)
        
    return action
```

### 第二模块：核心架构解释

```python
        self.q_former = QFormer(num_queries=8)
        self.llm = LLaMA_Model()
        self.world_generator = DiffusionUNet()
        
        # Learnable queries
        self.action_queries = nn.Parameter(torch.randn(N, D))
        self.world_queries = nn.Parameter(torch.randn(M, D))
        self.action_head = MLP()

    def forward(self, multi_view_images, text_instruction, mode="train"):
        # 1. Perception & Compression
        raw_visual_tokens = self.vision_encoder(multi_view_images)
        compressed_visual_tokens = self.q_former(raw_visual_tokens)
        
        text_tokens = tokenize(text_instruction)
        
        # 2. LLM Reasoning
        llm_inputs = concat([
            text_tokens, 
            compressed_visual_tokens, 
            self.action_queries, 
            self.world_queries
        ])
        
        llm_outputs = self.llm(llm_inputs)
        
        # Extract specific outputs based on query positions
        action_embeddings = extract_action_features(llm_outputs)
        world_embeddings = extract_world_features(llm_outputs)
        
        # 3. Action Prediction
        predicted_waypoints = self.action_head(action_embeddings)
        
        # 4. World Generation (Imagination)
        if mode == "train" or mode == "offline_generation":
            clip_features = extract_clip_features(multi_view_images)
            # Diffusion process conditioned on LLM world understanding
            future_videos = self.world_generator(
                noise=torch.randn_like(target_video),
                condition=world_embeddings,
                appearance_prior=clip_features
            )
            return predicted_waypoints, future_videos
            
        elif mode == "online_planning":
            # Discard diffusion model for low latency
            return predicted_waypoints
```

VLA-World 的核心架构将自动驾驶建模为一个联合优化问题：同时优化策略分布（决策）和环境动态分布（世界模型）。其运行流程与三阶段训练范式如下：

**运行流程：**
1.  **感知 (Perception)：** 接收多视角图像和自车状态，输出场景级结构化描述（如障碍物位置、道路边界）。
2.  **短期预测 (Short-term Prediction)：** 基于历史状态，预测未来 0.5 秒的自车轨迹和行驶方向。
3.  **条件引导生成 (Imagination)：** 将短期预测轨迹作为条件，自回归地生成 0.5 秒后的未来帧视觉 Token。
4.  **视觉反思 (Think)：** 模型对生成的未来帧进行分析，评估潜在的碰撞风险和安全裕度。
5.  **动作与规划 (Action & Planning)：** 基于反思结果，输出高层驾驶指令，并回归出修正后的未来 3 秒精确轨迹。

**训练范式：**
1.  **视觉预训练 (Visual Pretraining)：** 仅训练视觉生成能力，让模型内化物理世界的时空演变规律。
2.  **监督微调 (SFT)：** 在构建的 nuScenes-GR-20K 数据集上进行多任务混合微调，注入“生成-思考-规划”的概念知识。
3.  **强化学习 (RL via GRPO)：** 冻结参考模型，使用基于规则的奖励函数（格式、预测、视觉约束、动作、轨迹）进行无价值网络（Value-free）的群体相对策略优化，激发模型的自我纠错与逻辑一致性。

### 第二模块：核心架构解释

**模型架构与流程解析：**

1.  **统一表征构建 (Raxels Representation)：**
    传统的相机位姿通常表示为旋转矩阵和平移向量（$R, t$）以及内参（$K$）。该模型将每个像素对应的相机光线（起点和方向）计算出来，形成与视频帧空间分辨率一致的密集张量，称为 Raxels。这使得相机参数与图像像素在空间维度上完全对齐。
2.  **联合扩散过程 (Joint Diffusion Process)：**
    模型不再单独对视频加噪，而是将视频帧序列 $V$ 和对应的 Raxels 序列 $R_x$ 视为一个联合变量 $(V, R_x)$。在正向扩散过程中，同时向视频和 Raxels 添加高斯噪声。
3.  **解耦自-交叉注意力机制 (Decoupled Self-Cross Attention)：**
    在去噪 U-Net 或 Transformer 内部，为了处理视觉特征和几何特征的异构性，模型采用了解耦的注意力机制。视频特征和 Raxels 特征首先各自进行 Self-Attention 提取内部时空依赖，然后通过 Cross-Attention 进行信息交互，使得视频生成的纹理受相机轨迹约束，同时相机轨迹的预测受视频内容引导。
4.  **多任务统一推理 (Unified Multi-task Inference)：**
    *   **任务 A (位姿估计)：** 输入清晰视频，对 Raxels 进行去噪，输出相机轨迹。
    *   **任务 B (联合生成)：** 输入单张/多张条件图像，同时对视频和 Raxels 进行去噪，输出新视频及对应的相机轨迹。
    *   **任务 C (控制生成)：** 输入条件图像和目标 Raxels（无噪声），仅对视频进行去噪，输出受控视频。

```python
        self.video_cross_raxel = CrossAttention(dim)
        self.raxel_cross_video = CrossAttention(dim)
        
    def forward(self, video_feat, raxel_feat):
        # 1. 独立提取内部时空特征
        v_self = self.video_self_attn(video_feat)
        r_self = self.raxel_self_attn(raxel_feat)
        
        # 2. 跨模态特征交互 (相互作为 Key 和 Value)
        v_out = self.video_cross_raxel(query=v_self, key=r_self, value=r_self)
        r_out = self.raxel_cross_video(query=r_self, key=v_self, value=v_self)
        
        return v_out, r_out
```

### 第二模块：核心架构解释

VAG 的核心架构是一个基于流匹配（Flow Matching）的双流（Dual-Stream）网络。它包含一个视频分支（基于 DiT）和一个动作分支（基于 1D U-Net），两者在相同的训练时间步下同步运行。

1. **条件输入**：模型接收初始图像帧和经过 T5-XXL 编码的语言指令作为全局前缀条件。
2. **视频分支去噪**：视频分支继承自 Cosmos-Predict2。它将高斯噪声与初始帧在潜空间拼接，输入 Diffusion Transformer (DiT) 中。DiT 结合文本条件，预测出当前步的干净视频潜变量（Clean Video Latent）。
3. **跨模态桥接（核心创新）**：提取视频分支刚刚预测出的干净视频潜变量，使用自适应 3D 池化（Adaptive 3D Pooling）将其在时空维度上压缩，生成一个紧凑的全局视频嵌入向量。
4. **动作分支去噪**：动作分支采用 1D U-Net。它将上述生成的全局视频嵌入向量作为条件输入，结合当前的时间步，对动作噪声进行去噪，预测出对应的动作轨迹。
5. **同步优化**：在训练时，两个分支分别计算预测速度（Velocity）与真实速度之间的均方误差（MSE Loss），联合优化整个网络。

```python
def vag_denoise_step(noisy_video, noisy_action, image_cond, text_cond, t):
    # 1. 视频分支：基于 DiT 预测视频的 velocity
    v_video = DiT(noisy_video, image_cond, text_cond, t)
    
    # 根据 Flow Matching 公式，推导当前步预测的干净视频潜变量 z_0
    # 公式: x_t = (1-t)*x + t*epsilon -> v_t = epsilon - x
    pred_clean_video_latent = noisy_video - t * v_video 
    
    # 2. 跨模态桥接：自适应 3D 池化
    # 将高维的时空视频潜变量压缩为紧凑的全局特征向量
    global_video_embed = Adaptive3DPooling(pred_clean_video_latent)
    
    # 3. 动作分支：基于 1D U-Net 预测动作的 velocity
    # 将全局视频特征作为条件注入动作分支
    v_action = UNet1D(noisy_action, global_video_embed, t)
    
    return v_video, v_action

def training_step(video_gt, action_gt, image_cond, text_cond):
    # 随机采样时间步 t 和高斯噪声
    t = sample_timestep()
    noise_v, noise_a = sample_gaussian_noise()
    
    # Flow Matching 线性插值构造带噪输入
    noisy_video = (1 - t) * video_gt + t * noise_v
    noisy_action = (1 - t) * action_gt + t * noise_a
    
    # 执行同步去噪前向传播
    v_video_pred, v_action_pred = vag_denoise_step(noisy_video, noisy_action, image_cond, text_cond, t)
    
    # 计算 Ground Truth Velocity
    gt_v_video = noise_v - video_gt
    gt_v_action = noise_a - action_gt
    
    # 计算联合 MSE Loss
    loss_video = MSE(v_video_pred, gt_v_video)
    loss_action = MSE(v_action_pred, gt_v_action)
    total_loss = loss_video + loss_action
    
    return total_loss
```

### 第二模块：核心架构解释

AGD-MBRL 采用经典的 Dyna 风格架构，将扩散世界模型与强化学习策略进行交替训练。其核心流程分为三个阶段：

1. **数据收集与模型训练**：使用当前策略与真实环境交互收集轨迹数据，并用这些数据训练一个标准的条件扩散模型，使其能够拟合当前策略下的轨迹分布。
2. **优势引导轨迹生成（核心创新）**：在利用扩散模型“想象”未来轨迹时，采用分类器引导（Classifier-Guided）技术。在每一个逆向去噪步中，计算当前生成轨迹的优势函数梯度。论文提供了两种引导能量函数：
   - **EAG（指数优势引导）**：直接对优势函数求和并取指数，引导力度大，偏向激进探索。
   - **SAG（Sigmoid优势引导）**：将优势函数通过 Sigmoid 映射为概率，引导力度有界，偏向保守稳健。
   将计算出的梯度乘以缩放系数，注入到去噪均值中，迫使扩散模型生成比当前策略预期回报更高的轨迹。
3. **策略提升**：强化学习智能体（如 Actor-Critic）在这些经过优势引导生成的高质量“想象”轨迹上进行更新，实现策略迭代。

```python
def advantage_guided_sampling(diffusion_model, policy, value_net, state_init, N_steps, guide_type="EAG", alpha=1.0):
    # 初始化纯噪声轨迹 (包含状态和动作)
    traj = sample_gaussian_noise()
    
    for i in reversed(range(N_steps)):
        # Inpainting: 强制轨迹的初始状态为真实环境的当前状态
        traj.states[0] = state_init
        
        # 1. 扩散模型预测当前步的去噪均值和方差
        mean, var = diffusion_model.predict_denoising_step(traj, i)
        
        # 2. 评估当前去噪轨迹的优势 (Advantage = Q - V)
        # 注意：动作由策略网络条件生成，此处主要引导状态的生成
        actions = policy(traj.states)
        advantages = compute_advantage(traj.states, actions, value_net)
        
        # 3. 计算引导梯度
        if guide_type == "EAG":
            # 指数优势引导 (激进)
            energy = sum(advantages)
            gradient = compute_gradient(exp(energy), traj.states)
        elif guide_type == "SAG":
            # Sigmoid 优势引导 (保守)
            prob = prod(sigmoid(advantages))
            gradient = compute_gradient(log(prob), traj.states)
            
        # 4. 引导去噪步：将优势梯度注入均值中
        guided_mean = mean + alpha * var * gradient
        
        # 5. 采样下一步轨迹
        traj = sample_normal(guided_mean, var)
        
    return traj
```

```python
import torch
import torch.nn as nn

class AssemLM(nn.Module):
    def __init__(self, llm, point_cloud_encoder, projector):
        super().__init__()
        self.llm = llm
        self.pc_encoder = point_cloud_encoder
        self.projector = projector # 模态对齐投影层

    def forward(self, point_cloud, instruction_text, manual_image):
        # 1. 提取 3D 点云的细粒度几何与旋转特征
        pc_features = self.pc_encoder(point_cloud)
        
        # 2. 将 3D 特征投影到 LLM 的 Token 空间
        pc_tokens = self.projector(pc_features)
        
        # 3. 处理文本指令和装配说明书 (假设已有相应的 tokenizer/encoder)
        text_tokens = self.llm.tokenize(instruction_text)
        manual_tokens = self.llm.encode_image(manual_image)
        
        # 4. 拼接多模态 Prompt
        # 顺序通常为：系统指令 + 说明书 + 3D场景 + 任务指令
        multimodal_prompt = torch.cat([manual_tokens, pc_tokens, text_tokens], dim=-1)
        
        # 5. LLM 进行空间推理，自回归输出 6D 位姿 (平移 x,y,z 与 旋转四元数/欧拉角)
        output_pose_6d = self.llm.generate(multimodal_prompt)
        
        return output_pose_6d
```

```python
    t = sample_random_timestep()

    # 梯度直接作用于latents，进而反向传播到3D表示
    grad = weight(t) * (noise_pred - noise)
    loss_sds = torch.sum(grad * latents, dim=1) 
    return loss_sds
```

```python
        self.vlm = vlm
        self.simulator = simulator
        self.policy = policy_model

        # 1. Scenario Specification
        human_desc, env_desc, task_desc = self.llm.decompose_prompt(text_prompt)
        
        for _ in range(num_demos):
            # 2. Human & Scene Instantiation
            smplx_params = self.llm.get_smplx_params(human_desc)
            soft_human = self.simulator.create_hybrid_human(smplx_params)
            
            scene_layout = generate_scene_with_architect(env_desc)
            self.vlm.place_human_and_robot(scene_layout, soft_human)
            
            # 3. Motion Generation via LLM Code
            motion_code = self.llm.generate_motion_code(task_desc, available_apis=['RRT', 'Cartesian', 'get_joint_3d'])
            trajectory, target_point = execute_code_in_sim(motion_code, self.simulator)
            
            if is_valid_and_safe(trajectory):
                point_clouds = self.simulator.render_segmented_point_clouds(soft_human.target_part)
                dataset.append((point_clouds, trajectory, target_point))
                
        return dataset

    def train_hierarchical_policy(self, dataset):
        for obs_pc, trajectory, target_point in dataset:
            current_gripper_pts = trajectory.current_pose
            next_subgoal_pts = trajectory.next_waypoint
            delta_action = trajectory.get_delta_action()
            
            # High-level: Predict next subgoal based on PC, current gripper, and target point
            predicted_subgoal = self.policy.high_level_forward(
                obs_pc, current_gripper_pts, target_point
            )
            
            # Low-level: 3D Diffusion predicts delta action conditioned on subgoal
            predicted_action = self.policy.low_level_diffusion(
                obs_pc, current_gripper_pts, predicted_subgoal
            )
            
            self.policy.update_weights(predicted_subgoal, next_subgoal_pts, predicted_action, delta_action)
```

**核心机制:** 摒弃纯代码生成，引入“图像修复引导的场景构建（IGSC）”将大语言模型的空间语义转化为无碰撞的物理布局；同时，将视觉语言模型（VLM）降维用作“视觉裁判”，对每一步物理交互进行闭环拒绝采样，直接斩断误差传播链。

**4. 关键数据:**
*   **最能证明有效性的核心数据:** 在 Sim2Real（仿真到现实）迁移实验中，仅使用 10 条真实数据训练的基线成功率为 20%，而加入 250 条 V-CAGE 生成的仿真数据联合训练后，真实机械臂的成功率跃升至 55%（绝对提升 35%）。
*   **理由:** 这一数据直接证明了 V-CAGE 生成的仿真数据不仅在虚拟环境中有效，且具备极高的物理保真度和跨域泛化能力，切实缓解了 VLA 模型对真实世界数据的“数据饥荒”。
*   **存疑的试验结果:** 在数据压缩消融实验中，SortToCabinet 任务在使用压缩数据训练后的成功率（28%）反而略高于使用原始未压缩数据（25%）。

### 第二模块：核心架构解释

V-CAGE 的核心架构是一个端到端的自动化数据合成流水线，包含四个关键阶段：

1.  **图像修复引导的场景构建 (IGSC):** 
    首先，LLM 根据任务指令从资产库中选择物体，并生成自然语言布局规划。接着，将初始随机摆放的顶视图与布局文本输入图像修复模型（Inpainting Model），生成具有正确空间语义的目标图像。随后，利用 Grounding DINO 和 DINOv2 提取视觉特征，进行源图像与目标图像的密集对应点匹配，计算出目标世界坐标。最后，通过 L-BFGS-B 优化算法解决坐标转换带来的边界干涉，输出无碰撞且语义合理的物理场景。
2.  **基于模板的子任务搜索:** 
    利用资产库中预标注的功能元数据（如抓取点、放置点），系统遍历场景中的物体对。LLM 评估物体间的常识兼容性（例如“面包”可以放在“盘子”上），并将其填入预定义的机器人操作代码模板中，生成可执行的原子子任务。
3.  **VLM 闭环验证:** 
    在仿真器执行每一个子任务后，系统捕获当前视角的图像，并连同子任务描述输入给 VLM（如 Gemini 3）。VLM 仅输出二元判定（成功/失败）。一旦判定失败，立即终止当前长程任务的生成并重置场景。这本质上是一个基于视觉反馈的严格拒绝采样过程。
4.  **动作感知感知压缩:** 
    通过分析机械臂轨迹的运动学数据（如夹爪开合点、末端执行器速度峰值），提取交互最剧烈的关键帧。随后使用人类视觉感知指标（ColorVideoVDP）进行迭代优化，在保证视觉差异小于 0.1 JOD（视觉无损）的前提下，最大化视频压缩率。

```python
def v_cage_data_generation(task_instruction, asset_library, max_attempts):
    dataset = []
    
    for attempt in range(max_attempts):
        # Stage 1: Inpainting-Guided Scene Construction (IGSC)
        selected_assets = llm_agent.select_assets(task_instruction, asset_library)
        initial_scene, source_image = simulator.place_randomly_without_collision(selected_assets)
        
        layout_plan = llm_agent.plan_layout(task_instruction, initial_scene.metadata)
        target_image = inpainting_model.generate(source_image, layout_plan)
        
        target_coords = vision_model.match_features_and_recover_coords(source_image, target_image)
        optimized_scene = optimizer.resolve_collisions(target_coords)
        simulator.load_scene(optimized_scene)
        
        # Stage 2: Sub-task Search
        sub_tasks = template_engine.generate_executable_tasks(optimized_scene, task_instruction)
        
        trajectory = []
        task_success = True
        
        # Stage 3: VLM-Based Closed-Loop Verification
        for sub_task in sub_tasks:
            simulator.execute(sub_task.code)
            post_action_image = simulator.render_camera_view()
            
            # VLM acts as a strict visual critic
            is_valid = vlm_critic.verify(post_action_image, sub_task.description)
            
            if not is_valid:
                task_success = False
                break # Sever the error propagation chain
                
            trajectory.append(simulator.get_state_action_data())
            
        if task_success:
            # Stage 4: Action-Aware Compression
            compressed_trajectory = perceptual_compressor.compress(trajectory)
            dataset.append(compressed_trajectory)
            
    return dataset
```

### 第二模块：核心架构解释

WOMBET 的核心架构分为两个阶段：**离线数据生成阶段（源任务）**与**在线自适应微调阶段（目标任务）**。

1. **世界模型驱动的数据生成**：在源任务中训练一个概率集成世界模型（Probabilistic Ensemble World Model）。利用该模型进行模型预测控制（MPC），在规划时引入认知不确定性惩罚（Epistemic Uncertainty Penalty），从而生成保守且安全的合成轨迹。
2. **双标准数据过滤（Dual-Criterion Filtering）**：合成轨迹不可避免地带有模型偏差。WOMBET 引入双重阈值：只保留回报大于 $J_{th}$ 且不确定性低于 $u_{th}$ 的轨迹。这些高质量轨迹被重新标记（Relabel）为目标任务的奖励，存入离线经验池 $\mathcal{D}_S$。
3. **隐式正则化与鲁棒 Critic**：在目标任务中训练 Actor-Critic 策略。为了防止外推误差（Extrapolation Error）导致 Q 值爆炸，Critic 网络使用 LayerNorm 限制激活值大小，并采用集成最小值（Ensemble Minimum）计算 Bellman 目标。
4. **自适应采样（Adaptive Sampling）**：在在线微调时，每个 Batch 的数据由离线数据 $\mathcal{D}_S$ 和在线数据 $\mathcal{D}_T$ 混合而成。混合比例 $\alpha_k$ 根据当前 Critic 在在线数据上的平均绝对 TD 误差动态调整：TD 误差大（欠拟合/不稳定）时增加离线数据比例以稳定训练；TD 误差小（趋于稳定）时增加在线数据比例以加速适应。

**核心逻辑伪代码 (Python-style)**:

```python
def WOMBET_pipeline(source_env, target_env, world_model, policy, critic_ensemble):
    # ================= 阶段一：离线数据生成 =================
    offline_buffer = ReplayBuffer()
    
    # 1. 使用源任务训练世界模型
    world_model.train(source_env.get_data())
    
    # 2. 不确定性惩罚 MPC 规划与双标准过滤
    for _ in range(num_generation_steps):
        # 通过 MPC 寻找最大化惩罚回报的动作序列
        trajectory = MPC_plan_with_uncertainty_penalty(world_model)
        
        # 计算轨迹的总回报与平均不确定性
        return_S = compute_return(trajectory)
        uncertainty = world_model.compute_epistemic_uncertainty(trajectory)
        
        # 双标准过滤：高回报 + 低不确定性
        if return_S >= J_th and uncertainty <= u_th:
            # 重新标记为目标任务的 Reward
            trajectory.reward = target_env.compute_reward(trajectory.states, trajectory.actions)
            offline_buffer.add(trajectory)
            
    # ================= 阶段二：在线自适应微调 =================
    online_buffer = ReplayBuffer()
    alpha = 0.5 # 初始混合比例
    
    for step in range(online_finetune_steps):
        # 1. 与目标环境交互
        state = target_env.get_state()
        action = policy.select_action(state)
        next_state, reward, done = target_env.step(action)
        online_buffer.add(state, action, reward, next_state, done)
        
        # 2. 自适应采样混合 Batch
        batch_offline = offline_buffer.sample(batch_size * alpha)
        batch_online = online_buffer.sample(batch_size * (1 - alpha))
        mixed_batch = combine(batch_offline, batch_online)
        
        # 3. 隐式正则化更新 Critic (LayerNorm + Ensemble Min)
        td_errors = critic_ensemble.update(mixed_batch, policy)
        
        # 4. 更新 Policy
        policy.update(critic_ensemble, mixed_batch)
        
        # 5. 基于 TD 误差动态调整采样比例 alpha
        mean_td_error_online = compute_mean_td_error(critic_ensemble, online_buffer.get_recent())
        alpha = update_mixing_ratio(alpha, mean_td_error_online)
        
        # 6. 持续迭代更新世界模型 (可选)
        world_model.update(mixed_batch)
```

### 第二模块：核心架构解释

本文提出的三阶段Token剪枝框架（Tri-Stage Token Pruning Framework）是一个无需重新训练（Training-free）的推理期优化方法。其运行流程与实验流程如下：

**1. 阶段一：基于数据预处理的模态候选确定**
计算每个视觉Patch的2D和3D特征在最终层输出的 $L_1$ 范数，量化模型对3D特征的依赖比例。通过设定双阈值（$\tau_{2D}, \tau_{3D}$），将连续的比例映射为离散的保留策略：仅保留2D、仅保留3D或两者均保留，形成模态候选集。

**2. 阶段二：基于语义合成的候选确定**
提取模型的注意力分数矩阵，根据模态拆分后进行1D K-Means聚类（K=3），将所有Patch硬性划分为三个语义集合：目标对象、机械臂、背景。
- 背景区域：执行90%的随机剪枝。
- 机械臂区域：若3D依赖度高于全局基线，保留2D+3D；否则仅保留2D。
- 目标对象区域：默认保留2D+3D，仅在极端依赖2D时退化为仅保留2D。
由此形成语义候选集。

**3. 阶段三：基于动作迭代的时间平滑**
由于机器人操控是连续动态过程，单帧的显著性计算会导致掩码（Mask）在帧间闪烁。引入基于滑动窗口的指数移动平均（EMA）机制，利用历史帧的显著性预测值对当前帧的计算结果进行时间维度的平滑更新。

**4. 全局候选融合**
将阶段二输出的语义候选集与阶段一输出的模态候选集取逻辑交集。若交集为空（发生冲突），则强制回退到语义候选集的决策。最终输出2D和3D分支的二值化保留掩码（Mask），值为1的Token进入后续计算，值为0的被丢弃。

**核心逻辑伪代码**：

```python
        # 冷启动，不剪枝
        return full_masks_2d, full_masks_3d

    # 阶段一：模态候选确定 (基于特征范数)
    modality_candidate_set = {}
    for patch in patches:
        m_3d_ratio = compute_3d_norm_ratio(smoothed_metrics, patch)
        if m_3d_ratio < tau_2d:
            modality_candidate_set[patch] = {'2D'}
        elif m_3d_ratio > tau_3d:
            modality_candidate_set[patch] = {'3D'}
        else:
            modality_candidate_set[patch] = {'2D', '3D'}

    # 阶段二：语义候选确定 (基于注意力聚类)
    semantic_clusters = kmeans_clustering(smoothed_metrics.attention, k=3) # 返回 bg, rob, obj
    semantic_candidate_set = {}
    for patch in patches:
        if patch in semantic_clusters['bg']:
            semantic_candidate_set[patch] = set() if random() < 0.9 else {'2D', '3D'}
        elif patch in semantic_clusters['rob']:
            semantic_candidate_set[patch] = evaluate_rob_rules(smoothed_metrics, patch)
        elif patch in semantic_clusters['obj']:
            semantic_candidate_set[patch] = evaluate_obj_rules(smoothed_metrics, patch)

    # 阶段四：全局候选融合
    mask_2d, mask_3d = initialize_empty_masks()
    for patch in patches:
        if semantic_candidate_set[patch] is not empty:
            # 取交集
            final_candidate = semantic_candidate_set[patch].intersection(modality_candidate_set[patch])
            # 冲突解决
            if final_candidate is empty:
                final_candidate = semantic_candidate_set[patch]
            
            # 生成最终Mask
            if '2D' in final_candidate: mask_2d[patch] = 1
            if '3D' in final_candidate: mask_3d[patch] = 1

    return mask_2d, mask_3d
```

### 第二模块：核心架构解释

3D-VCD 的核心逻辑是**双上下文推理与逻辑值（Logits）对比融合**。整个流程分为三个客观步骤：

1.  **场景图构建与扰动 (Graph Distortion):** 接收原始的 3D 场景图（包含物体类别、3D 中心点坐标、空间范围）。通过注入高斯噪声改变坐标和尺寸，或随机替换类别标签，生成一个“失真”的负面场景图。
2.  **双路前向传播 (Dual Forward Pass):** 将用户的文本查询分别与“原始场景图”和“失真场景图”拼接，同时输入到冻结权重的 3D-MLLM 中。利用 KV Cache 技术缓存注意力状态，避免重复计算。
3.  **对比解码 (Contrastive Decoding):** 在生成每一个 Token 时，获取模型在原始场景下的预测概率分布（Logits）和失真场景下的预测概率分布。通过公式放大原始分布的权重，并减去失真分布的权重。最终通过 Softmax 输出下一个 Token。

**核心伪代码实现逻辑：**

```python
import torch
import torch.nn.functional as F

def generate_distorted_scene_graph(original_graph, noise_std=0.05):
    """对3D场景图施加语义和几何扰动"""
    distorted_graph = original_graph.copy()
    for obj in distorted_graph.objects:
        # 几何扰动：向3D质心和边界框范围添加零均值高斯噪声
        obj.centroid += torch.normal(mean=0.0, std=noise_std, size=obj.centroid.shape)
        obj.extent += torch.normal(mean=0.0, std=noise_std, size=obj.extent.shape)
        # 语义扰动：一定概率替换类别标签
        if torch.rand(1) < 0.1: 
            obj.category = get_random_category()
    return distorted_graph

def vcd_decoding_step(model, query, original_graph, distorted_graph, alpha=1.0):
    """单步视觉对比解码"""
    # 1. 分别获取原始上下文和失真上下文的 Logits
    # 实际工程中通过 Batch=2 一次性前向传播并复用 KV Cache
    logits_original = model.forward(query, original_graph) 
    logits_distorted = model.forward(query, distorted_graph)
    
    # 2. 核心公式：对比融合 Logits (Eq. 3)
    # 放大原始证据的权重，惩罚在失真环境下依然高概率的 Token
    logits_vcd = (1 + alpha) * logits_original - alpha * logits_distorted
    
    # 3. 计算最终的概率分布并采样下一个 Token
    probs = F.softmax(logits_vcd, dim=-1)
    next_token = torch.argmax(probs) # 论文默认使用贪婪解码 (Greedy Search)
    
    return next_token
```

### 第二模块：核心架构解释

PinpointQA 的构建逻辑摒弃了传统依赖人工标注的众包模式，转而采用“3D 物理引擎提取 -> 中间态表示 -> 模板化 QA 生成”的自动化流水线。这种方法保证了 Ground Truth 的绝对物理正确性。

其核心流程分为三步：
1.  **场景清洗与目标锚定**：从 ScanNet++ 和 ScanNet200 中提取 3D 网格和实例分割掩码，过滤掉背景，锁定预定义词表中的“小物体”（如钥匙、鼠标）。
2.  **中间态空间表示计算**：以目标小物体为中心，划定 1.0 米的局部邻域，计算目标与支撑面（Supporting Surface）及周围参考物（Reference Objects）的精确三维欧氏距离和垂直拓扑关系。
3.  **四阶任务实例化**：将上述同一套物理关系，分别降维映射为 TPV（二分类）、NRI（多选题）、FSD（自由文本生成）和 SSP（JSON 结构化输出）四种 QA 格式。

以下是揭示其底层数据生成逻辑的 Python 风格伪代码：

```python
def build_pinpoint_qa(scene_3d_data, small_object_vocab):
    qa_dataset = []
    
    # 1. 场景清洗与目标锚定
    instances = extract_instances(scene_3d_data)
    target_objects = [obj for obj in instances if obj.label in small_object_vocab]
    
    for target in target_objects:
        # 2. 构建中间态空间表示 (Intermediate Spatial Representation)
        local_neighborhood = get_objects_within_radius(target, instances, radius=1.0)
        supporting_surface = find_supporting_surface(target, instances)
        
        spatial_relations = []
        for ref_obj in local_neighborhood:
            if ref_obj == supporting_surface:
                continue
            distance_cm = compute_3d_euclidean_distance(target, ref_obj)
            relation_type = compute_vertical_topology(target, ref_obj) # e.g., 'side_above', 'near'
            spatial_relations.append({
                "object": ref_obj.label,
                "relation": relation_type,
                "distance_cm": distance_cm
            })
            
        # 按距离排序，提取最近的参考物
        spatial_relations.sort(key=lambda x: x["distance_cm"])
        nearest_ref = spatial_relations[0] if spatial_relations else None
        
        # 3. 四阶任务实例化
        # Task 1: TPV (Target Presence Verification)
        qa_dataset.append(generate_tpv_qa(target.label, is_present=True))
        
        # Task 2: NRI (Nearest Reference Identification)
        if nearest_ref:
            qa_dataset.append(generate_nri_qa(target.label, nearest_ref, local_neighborhood))
            
        # Task 3: FSD (Fine-Grained Spatial Description)
        qa_dataset.append(generate_fsd_qa(target.label, supporting_surface, spatial_relations[:2]))
        
        # Task 4: SSP (Structured Spatial Prediction)
        qa_dataset.append(generate_ssp_qa(target.label, supporting_surface, spatial_relations[:2]))
        
    return qa_dataset
```

### 第二模块：核心架构解释

PilotBench 的核心架构并非提出一个新的网络模型，而是构建了一套严密的物理-语义交叉验证流水线与评估范式。其运行流程分为数据构建、模型推理与复合评估三个阶段：

数据构建阶段：通过双频 RTK-GNSS、大气数据计算机（ADC）和惯性基准单元（IRU）采集 34 通道、10Hz 的真实飞行数据。通过 8 阶段清洗流水线，将连续轨迹严格切分为 9 个标准飞行阶段（5 个直线段 P1-P5，4 个过渡段 T1-T4），并注入对应的自然语言指令与安全包线约束。

模型推理阶段：模型接收结构化的历史航路点数据与自然语言指令，输出单步的轨迹与姿态预测。在此过程中，引入了两种提示工程策略：上下文学习（+ICL）用于稳定输出语法格式，物理思维链（+Phy-CoT）用于注入飞行约束语义。

复合评估阶段：计算 Pilot-Score。该指标由回归误差（MAE, RMSE）和指令遵循度（字段完整性、有效性、格式）线性组合而成，直接对齐航空业的所需导航性能（RNP）标准。

以下是 PilotBench 核心评估逻辑的 Python 风格伪代码：

```python
import numpy as np

class PilotBenchEvaluator:
    def __init__(self, safety_envelopes):
        self.safety_envelopes = safety_envelopes
        self.weight_regression = 0.6
        self.weight_instruction = 0.4

    def compute_regression_score(self, pred_traj, true_traj):
        # 计算 MAE 和 RMSE
        mae = np.mean(np.abs(pred_traj - true_traj))
        rmse = np.sqrt(np.mean((pred_traj - true_traj)**2))
        # 通过分段线性映射将误差转换为 0-100 的分数
        R_score = 0.6 * self.map_to_score(mae) + 0.4 * self.map_to_score(rmse)
        return R_score

    def compute_instruction_score(self, pred_format, pred_values, phase):
        # c1: 字段完整性, c2: 字段有效性 (是否在安全包线内), c3: 格式合规性
        c1 = self.check_completeness(pred_format)
        c2 = self.check_validity(pred_values, self.safety_envelopes[phase])
        c3 = self.check_format(pred_format)
        
        I_score = (0.5 * c1 + 0.3 * c2 + 0.2 * c3) * 100
        return max(I_score, 5.0) # 设置最低分 5 分防止完全失效分类

    def calculate_pilot_score(self, model_output, ground_truth, phase):
        pred_traj, pred_format, pred_values = self.parse_output(model_output)
        
        R = self.compute_regression_score(pred_traj, ground_truth)
        I = self.compute_instruction_score(pred_format, pred_values, phase)
        
        # 核心公式: F(MAE, RMSE, c) = 0.6 * R + 0.4 * I
        pilot_score = self.weight_regression * R + self.weight_instruction * I
        return pilot_score

# 评估执行示例
evaluator = PilotBenchEvaluator(safety_envelopes=FAA_STANDARDS)
for phase, data in flight_dataset:
    # 传统模型: 只有轨迹预测，没有格式和指令理解

    trad_score = evaluator.calculate_pilot_score(trad_output, data.truth, phase)

    llm_output = llm.generate(prompt=data.instruction + data.history, strategy="Phy-CoT")
    llm_score = evaluator.calculate_pilot_score(llm_output, data.truth, phase)
```
