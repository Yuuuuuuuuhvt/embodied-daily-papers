        super().__init__()
        # Causal Transformer backbone
        self.transformer = CausalTransformerBackbone()
        
        # Joint prediction heads
        self.action_head = nn.Linear(hidden_dim, action_dim)
        self.state_head = nn.Linear(hidden_dim, state_dim)
        self.command_dist_head = CommandDistributionHead(hidden_dim)
        
    def forward(self, state, terrain_geometry, current_command):
        # Extract features using causal attention
        features = self.transformer(state, terrain_geometry, current_command)
        
        # Predict next action, state, and feasible command distribution
        next_action = self.action_head(features)
        next_state_pred = self.state_head(features)
        next_command_dist = self.command_dist_head(features)
        
        return next_action, next_state_pred, next_command_dist

def deploy_step(bwm_model, state, terrain, upstream_command, threshold=0.1):
    # Forward pass to get the distribution of feasible commands
    next_action, _, command_dist = bwm_model(state, terrain, upstream_command)
    
    # Online implausible command detection
    command_probability = command_dist.log_prob(upstream_command).exp()

        # Retract command: sample a safe, feasible behavior from the learned distribution
        safe_command = command_dist.sample()

        next_action, _, _ = bwm_model(state, terrain, safe_command)
        
    return next_action

### 第二模块：核心架构解释

DA-LeWM 的核心逻辑在于“训练时增加约束，推理时保持轻量”。其架构与流程分为以下三个阶段：

1. **状态编码与前向预测**：使用视觉编码器（ViT）将当前观测图像映射为隐变量。随后使用自回归预测器，根据当前隐变量和动作序列，预测未来的隐变量。
2. **动作条件辅助监督（训练阶段）**：在计算基础的预测误差和防止特征坍塌的正则化损失（SIGReg）之外，引入两个多层感知机（MLP）：
   - 逆动力学头：输入当前隐变量和预测的下一帧隐变量，输出对应的动作。
   - 目标动作头：输入当前隐变量和目标隐变量，输出到达目标所需的动作。
3. **基于隐空间距离的 MPC 规划（推理阶段）**：在实际控制机器人时，完全丢弃上述两个辅助头。使用交叉熵方法（CEM）生成大量候选动作序列，通过世界模型预测这些序列的终点隐变量。直接计算终点隐变量与目标隐变量之间的欧氏距离作为代价，选出距离最小的动作执行。

### 第二模块：核心架构解释

RP1 的核心架构分为两个独立的神经网络组件：Critic（评价网络）和 Planner（规划网络）。整个训练过程是完全离线的，且底层世界模型（World Model）的权重保持冻结。

1. 评价网络（Critic）的训练：
传统方法通常计算当前状态与目标状态在潜在空间（Latent Space）的欧氏距离来评估动作好坏。RP1 摒弃了这一点，训练了一个目标条件价值函数 $V_\psi$。该函数通过时序差分（TD）学习，直接预测从当前状态到达目标状态所需的真实物理时间步数（时间可达性）。

2. 规划网络（Planner）的训练：
Planner 是一个残差神经网络。它不直接输出动作，而是输出对现有动作序列的“修改量”。
在训练时，系统初始化一个全零的动作序列。
世界模型根据该动作序列推演出最终的预测状态。
Critic 评估该最终状态距离目标的代价，并计算代价对动作序列的梯度。
Planner 接收当前动作序列、代价、以及梯度，输出一个动作残差，加到原动作序列上。
该过程迭代 K 次。最终的优化目标是最小化最后一次迭代产生的动作序列所对应的 Critic 代价。

3. 推理阶段：
在实际测试中，给定初始状态和目标，Planner 仅需执行极少次数（如 8 次）的前向传播，即可输出高质量的动作序列，直接绕过了传统方法成千上万次的随机采样。

planner = NeuralPlanner()         # 待训练的残差规划网络

# 训练超参数
K = 8  # 规划器的迭代细化次数
learning_rate = 3e-4

def train_planner_step(start_latent, goal_latent):
    # 初始化一个空的动作序列 (例如 5 个时间步)
    action_plan = initialize_zero_plan(horizon=5)
    
    # 记录所有迭代步骤的代价，用于计算损失
    trajectory_costs = []
    
    for k in range(K):
        # 1. Rollout: 使用冻结的世界模型推演当前动作序列的最终状态
        final_latent = world_model.rollout(start_latent, action_plan)
        
        # 2. Evaluate: 使用 Critic 评估最终状态距离目标的代价
        # 开启梯度追踪以获取代价对动作的导数
        cost = critic.evaluate(final_latent, goal_latent)
        cost_gradient = compute_gradient(cost, action_plan)
        
        # 3. Improve: Planner 接收当前计划、代价和梯度，输出动作残差
        # 注意：梯度仅作为 Planner 的输入特征，不直接用于更新动作
        action_residual = planner(action_plan, cost.detach(), cost_gradient.detach())
        
        # 更新动作序列并裁剪至合法范围
        action_plan = clip(action_plan + action_residual, min_action, max_action)
        
        trajectory_costs.append(cost)
    
    # 4. Optimize: 最小化最后一步的代价，并对中间步骤的代价进行正则化
    loss = trajectory_costs[-1] + lambda_mean * mean(trajectory_costs)
    
    # 反向传播更新 Planner 的权重
    loss.backward()
    update_planner_weights(learning_rate)

import torch.nn as nn
from gsplat import rasterization

class GS_VLA_Canonicalizer(nn.Module):
    def __init__(self):
        super().__init__()
        # 4M parameter U-Net
        self.unet = UNet(in_channels=4, out_channels=14) 
        self.pose_mlp = MLP(in_dim=14, out_dim=128)
        
    def forward(self, I_s, D_s, C_s, C_star, K):
        # 1. Compute pose difference for FiLM modulation
        pose_diff = compute_pose_diff(C_s, C_star)
        z_pose = self.pose_mlp(pose_diff)
        
        # 2. Predict 3D Gaussian attributes per pixel
        x = torch.cat([I_s, D_s], dim=1)
        gaussian_attrs = self.unet(x, z_pose) 
        
        delta_mu, log_s, q, alpha, delta_c = split_attrs(gaussian_attrs)
        
        # 3. Geometric back-projection to world space
        X_w = backproject(D_s, C_s, K) 
        
        # 4. Apply depth-bounded residual to prevent geometric drift
        mu = X_w + 0.1 * D_s * torch.tanh(delta_mu)
        color = I_s + delta_c
        
        # 5. Differentiable rasterization at canonical pose
        I_star_hat = rasterization(
            means=mu, 
            scales=torch.exp(log_s), 
            quats=q, 
            opacities=torch.sigmoid(alpha), 
            colors=color, 
            viewmat=C_star, 
            projmat=K
        )
        
        return I_star_hat

# Deployment Pipeline

class EmbodimentEvaluator:
    def __init__(self, source_model, target_robot):
        self.model = source_model
        self.robot = target_robot
        self.report_card = {
            "modified_components": [],
            "target_data_used": 0,
            "calibration_effort": 0,
            "human_interventions": 0,
            "failure_causes": {"perception": 0, "interface": 0, "contact": 0}
        }
        self.eac_data = [] # Embodiment Adaptation Curve data

    def adapt_to_target(self, adaptation_budget):
        # Record the engineering work required BEFORE evaluation
        target_data = self.robot.collect_demonstrations(num=adaptation_budget)
        self.report_card["target_data_used"] += adaptation_budget
        
        # E.g., LoRA fine-tuning, action decoder alignment
        self.model.fine_tune(target_data)
        self.report_card["modified_components"].append("Action Decoder")
        self.report_card["calibration_effort"] += self.robot.align_control_rate()

    def evaluate_deployment(self, num_trials):
        success_count = 0
        for trial in range(num_trials):
            state = self.robot.reset()
            done = False
            while not done:
                action = self.model.predict(state)
                next_state, status = self.robot.step(action)
                
                # Track failures that success rate hides
                if status == "unsafe_contact" or status == "kinematic_violation":
                    self.report_card["human_interventions"] += 1
                    self.report_card["failure_causes"]["contact"] += 1
                    break # Manual reset required
                
                if status == "success":
                    success_count += 1
                    done = True
                    
        success_rate = success_count / num_trials
        return success_rate

    def generate_eac_curve(self, budget_list):
        # Generate data for Embodiment Adaptation Curve (EAC)
        for budget in budget_list:
            self.adapt_to_target(budget)
            sr = self.evaluate_deployment(num_trials=50)
            self.eac_data.append({"adaptation_effort": budget, "success_rate": sr})
        return self.eac_data

### 第二模块：核心架构解释

RoleSub 的核心架构分为四个连续的客观逻辑步骤：

1. **Token 级初步筛选 (Token-Level Reduction)**：首先使用现有的重要性评分机制（如基于 QK 的注意力分数）对视觉 token 进行初步筛选，保留设定比例的 token，丢弃其余 token。
2. **正交值空间分解 (Orthogonal Value-Space Decomposition)**：对于保留下来的 token，其 Query 和 Key 保持不变。将其 Value 向量通过一个固定的正交矩阵映射到正交路由空间，并将其划分为 S 个互不重叠的特征分组。正交变换保证了基底转换的无损性。
3. **基于角色的动态路由 (Role-Conditioned Value Routing)**：构建一个轻量级路由网络，输入包含三个信号：当前 token 的隐藏状态、通过 Softmax 提取的低维隐式角色表征（Latent Role Representation）、以及全局语言指令的池化上下文。三者线性组合生成对 S 个 Value 分组的保留偏好得分。
4. **自适应预算分配与掩码重构 (Adaptive Budget Allocation & Masking)**：根据预设的模态预算（如视觉或语言的平均保留组数），利用隐式角色表征作为优先级信号，在同类型 token 之间动态分配保留名额。选出 Top-K 的分组后，未选中的分组置零，最后通过逆正交矩阵将稀疏化后的 Value 向量映射回原始空间。

**核心逻辑伪代码 (Python/PyTorch 风格)**:

```python
import torch
import torch.nn as nn
import torch.nn.functional as F

class RoleSubRouter(nn.Module):
    def __init__(self, hidden_dim, value_dim, num_groups=16, num_roles=6):
        super().__init__()
        self.S = num_groups
        self.group_dim = value_dim // num_groups
        
        # 固定的正交矩阵 (Frozen Orthogonal Matrix)
        orthogonal_matrix = torch.nn.init.orthogonal_(torch.empty(value_dim, value_dim))
        self.register_buffer('R', orthogonal_matrix)
        
        # 路由评分网络
        self.f_v = nn.Linear(hidden_dim, num_groups)
        self.f_role = nn.Linear(hidden_dim, num_roles)
        self.W_role = nn.Linear(num_roles, num_groups, bias=False)
        self.f_lang = nn.Linear(hidden_dim, num_groups)
        
        self.alpha_role = 0.1
        self.alpha_lang = 0.1

    def forward(self, hidden_states, values, lang_context, target_budget_K):
        # hidden_states: [N, hidden_dim], values: [N, value_dim]
        
        # 1. 正交值空间分解
        # z: [N, value_dim] -> reshape to [N, S, group_dim]
        z = torch.matmul(values, self.R) 
        
        # 2. 计算隐式角色表征
        # a_role: [N, num_roles]
        a_role = F.softmax(self.f_role(hidden_states), dim=-1)
        
        # 3. 计算路由得分
        score_token = self.f_v(hidden_states)
        score_role = self.W_role(a_role)
        score_lang = self.f_lang(lang_context) # lang_context is pooled language hidden states
        
        # 综合得分: [N, S]
        routing_scores = score_token + self.alpha_role * score_role + self.alpha_lang * score_lang
        
        # 4. 组选择与掩码生成 (此处简化了跨 token 的动态预算分配，展示核心 Top-K 逻辑)
        # 实际实现中 target_budget_K 会根据 a_role 在 token 间动态调整
        _, topk_indices = torch.topk(routing_scores, k=target_budget_K, dim=-1)
        mask = torch.zeros_like(routing_scores).scatter_(-1, topk_indices, 1.0)
        
        # 扩展掩码到完整的 value 维度: [N, S, 1]
        mask_expanded = mask.unsqueeze(-1).expand(-1, -1, self.group_dim).reshape(values.shape[0], -1)
        
        # 5. 应用掩码并逆映射回原空间
        z_masked = z * mask_expanded
        routed_values = torch.matmul(z_masked, self.R.T)
        
        return routed_values
```

### 第二模块：核心架构解释

本文的方法论不涉及神经网络架构的设计与权重更新，而是构建了一个由 LLM 驱动的自动化软件工程闭环（Agentic Self-Learning）。其核心流程分为外部的“策略生成循环”与内部的“策略执行循环”。

1. **外部策略生成循环 (Agentic Loop)**：
   * LLM 接收自然语言任务描述，搜索并读取仿真环境的源代码。
   * LLM 编写初始的感知与控制 Python 代码。
   * 在仿真器中运行代码，LLM 读取执行日志、报错信息和成功率指标。
   * LLM 自主诊断运动学 Bug，将控制逻辑从早期的启发式搜索（Heuristics）重构为模型预测控制（MPC），并针对困难几何形状（如字母 Q、B）生成课程学习（Curriculum Learning）进行重点突破。

2. **内部策略执行循环 (Generated Code Pipeline)**：
   * **闭环感知 (Perception)**：处理 RGB 图像，通过颜色分割、掩码腐蚀剔除机械臂本体，使用粗到精的 IoU 拟合提取目标物体的位姿（Pose）。
   * **参数自适应 (Parameter Adaptation)**：在每次推动完成后，对比数字孪生环境中的预测位姿与实际位姿，动态修正摩擦系数增益。
   * **MPC 规划与控制 (Planning & Control)**：基于状态机（规划、接近、推动、撤退），在离散的接触点库中，利用准静态接触模型预测不同动作的位姿误差，贪心选择能最大程度降低误差的动作指令下发给底层 PD 控制器。

### 第二模块：核心架构解释

SparsePR 的核心架构旨在构建一个可执行的、低误差的块稀疏注意力算子。其流程主要由三个阶段构成：

1. **响应耦合划分 (Response-Coupled Partitioning)**：传统的空间或时间划分无法保证同一块内的 Query 关注相似的 Key。SparsePR 首先对少量 Query 进行采样，计算它们对所有 Key 的响应。基于这些响应特征，将 Key 和 Value 聚类成对齐的 K/V 组，并提取质心。随后，所有的 Query 根据与质心的相似度被路由到对应的 K/V 组，确保共享路由的 Query 具有高度重叠的注意力支持。
2. **块稀疏注意力计算 (Block-Sparse Attention)**：根据上述路由关系，仅计算被选中的 Query-Key 块的注意力权重和输出，大幅降低 FLOPs。
3. **探针拟合残差重建 (Probe-Fitted Residual Reconstruction)**：从 Query 中抽取极小比例（探针预算）计算精确的稠密注意力输出。计算这些探针的精确输出与稀疏输出之间的残差。利用线性回归，拟合出一个从稀疏输出特征空间到残差空间的仿射变换矩阵。最后，将该仿射变换应用到所有 Query 的稀疏输出上，补偿被丢弃的注意力交互。

**核心伪代码 (Python 风格)**:

```python
def sparse_pr_attention(Q, K, V, probe_budget_ratio=0.05, sparsity_ratio=0.25):
    N, D = Q.shape
    
    # === 阶段 1: 响应耦合划分 ===
    # 1. 采样少量 Query 获取对 Key 的全局响应
    Q_sample = random_sample(Q, num_samples=int(N * 0.01))
    key_responses = Q_sample @ K.T  
    
    # 2. 基于响应特征对 K/V 进行聚类分组，并获取质心
    KV_groups, centroids = cluster_keys_by_response(key_responses, K, V)
    
    # 3. 将所有 Query 路由到最匹配的 K/V 组
    Q_routes = assign_queries_to_centroids(Q, centroids, sparsity_ratio)
    
    # === 阶段 2: 块稀疏注意力计算 ===
    # 仅计算路由匹配的块，获得稀疏输出
    sparse_output = compute_block_sparse_attention(Q, K, V, Q_routes)
    
    # === 阶段 3: 探针拟合残差重建 ===
    # 1. 选取探针 Query 计算精确注意力
    num_probes = int(N * probe_budget_ratio)
    probe_indices = select_probe_indices(N, num_probes)
    Q_probe = Q[probe_indices]
    
    exact_probe_output = exact_attention(Q_probe, K, V)
    sparse_probe_output = sparse_output[probe_indices]
    
    # 2. 计算探针的真实残差
    probe_residual = exact_probe_output - sparse_probe_output
    
    # 3. 拟合仿射变换 (W, b): sparse_probe_output -> probe_residual
    W, b = fit_affine_transformation(sparse_probe_output, probe_residual)
    
    # 4. 全局应用残差补偿
    estimated_residual = sparse_output @ W + b
    final_output = sparse_output + estimated_residual
    
    return final_output
```

### 第二模块：核心架构解释

DA-WAM 的核心架构分为三个主要阶段：观察编码、动作条件化未来预测、轨迹评分。其训练过程巧妙结合了离线数据的专家匹配与硬负样本挖掘。

1.  **观察编码 (Observation Encoding)**：使用预训练的 V-JEPA 2.1 作为视觉主干。在线编码器（Online Encoder）通过 LoRA 微调提取当前帧的场景 Token；目标编码器（Target Encoder）通过指数移动平均（EMA）更新，提取真实未来帧的 Token 作为预测目标。
2.  **动作条件化未来预测 (Action-Conditioned Future Prediction)**：将生成的 $N$ 个候选轨迹编码为动作表征。预测器以动作表征为 Query，当前场景 Token 为 Key 和 Value 进行交叉注意力计算，为每个候选轨迹输出一个专属的未来潜变量。
3.  **轨迹评分 (Trajectory Scoring)**：评分器接收当前场景 Token、动作表征和专属未来潜变量，输出分解的规划因子（如碰撞、合规性、进度等）以及最终的标量效用得分。
4.  **训练对齐策略 (Training Alignment)**：由于离线数据集仅包含专家轨迹的真实未来，模型仅对“与专家最匹配的候选轨迹”施加未来潜变量的回归损失（Dense JEPA Loss）。对于其他反事实轨迹，通过下游的评分损失间接优化。同时引入几何相似但不安全的“硬负样本”轨迹，增强模型对安全边界的判别力。

**核心逻辑伪代码 (Python-style)**：

import torch
import torch.nn as nn

class DAWAM(nn.Module):
    def __init__(self, v_jepa_backbone, predictor, scorer):
        super().__init__()
        # 1. 观察编码器 (LoRA微调) 与 EMA 目标编码器
        self.online_encoder = apply_lora(v_jepa_backbone)
        self.target_encoder = copy_and_freeze(self.online_encoder)
        
        self.traj_encoder = TrajectoryEncoder()
        self.predictor = predictor # Action-conditioned Predictor
        self.scorer = scorer       # Future-latent-conditioned Scorer
        
    def forward(self, current_obs, future_obs, candidate_trajs, expert_traj, hard_neg_trajs):
        # 提取当前场景特征
        Z_t = self.online_encoder(current_obs)
        
        # 训练阶段：合并生成的候选轨迹与硬负样本轨迹
        all_trajs = torch.cat([candidate_trajs, hard_neg_trajs], dim=1)
        
        # 编码所有轨迹动作
        actions = self.traj_encoder(all_trajs) # [B, N, D]
        
        # 2. 动作条件化未来预测 (一对一映射)
        # Query: actions, Key/Value: Z_t
        Z_hat_future = self.predictor(query=actions, key=Z_t, value=Z_t) 
        
        # 3. 轨迹评分
        # 结合当前场景、动作、专属预测未来进行打分
        factors, utility_scores = self.scorer(Z_t, actions, Z_hat_future)
        
        if self.training:
            # 提取真实未来特征 (Stop Gradient)
            with torch.no_grad():
                Z_future_target = self.target_encoder(future_obs)
            
            # 专家匹配：找到与真实专家轨迹最接近的候选轨迹索引
            expert_idx = match_expert(candidate_trajs, expert_traj)
            
            # 仅对专家匹配的轨迹计算未来潜变量预测损失 (JEPA Loss)
            Z_hat_expert = Z_hat_future[batch_indices, expert_idx]
            loss_pred = mse_loss(Z_hat_expert, Z_future_target)
            
            # 计算所有轨迹的评分损失 (因子回归 + 效用排序)
            loss_factor = compute_factor_loss(factors, ground_truth_factors)
            loss_rank = compute_ranking_loss(utility_scores, ground_truth_utilities)
            
            total_loss = loss_pred + loss_factor + loss_rank
            return total_loss
        else:
            # 推理阶段：选择效用得分最高的轨迹
            best_traj_idx = torch.argmax(utility_scores, dim=1)
            return candidate_trajs[batch_indices, best_traj_idx]

    def update_target_encoder(self, momentum=0.99):
        # EMA 更新目标编码器
        for online_param, target_param in zip(self.online_encoder.parameters(), self.target_encoder.parameters()):
            target_param.data = momentum * target_param.data + (1 - momentum) * online_param.data

### 第二模块：核心架构解释

本文作为一篇权威综述，并未提出单一的神经网络权重，而是提炼并定义了通向具身智能的**图增强第一视角视觉语言模型（Graph-enhanced EgoVLM）**的理想架构与处理流程。该架构将感知、结构化与推理严格分层。

**1. 基础感知层 (Base Perception Layer)**
摒弃全局密集特征提取，专注于高信息密度区域。利用 100DOH 或 HaMeR 等检测器与网格重建模型，在严重的自我运动和遮挡下，精准定位佩戴者的双手以及处于激活状态（被操作）的物体。

**2. 关系图构建层 (Relational Graph Construction)**
将孤立的检测框转化为动态手物交互图（HOI Graph）或第一视角动作场景图（EASG）。节点代表手和物体，边代表接触状态（无接触、自我接触、便携物体接触等）。图结构随时间在“前置条件 (Pre-condition) -> 接触 (Contact) -> 不可逆点 (PNR) -> 后置条件 (Post-condition)”的生命周期中动态演变。

**3. 图引导的时空推理层 (Graph-Guided Spatiotemporal Reasoning)**
针对长达数小时的第一视角视频，拒绝使用计算代价极高的均匀帧采样。利用构建的交互图作为控制器，执行图引导的帧采样（Graph-guided Frame Sampling），仅将发生关键状态改变的帧送入 VLM 的注意力机制中，或将图结构压缩为工作记忆（Active Memory）供长时序查询。

**4. 交互感知对齐层 (Interaction-Aware Semantic Alignment)**
在视觉与语言的对齐阶段，放弃传统的 InfoNCE 粗粒度对比学习。采用 EgoNCE++ 等非对称对比目标，通过大语言模型生成细粒度的手物动力学描述，并挖掘改变核心动词的硬负样本，强制模型在连续向量空间中对齐精细的交互动作。

**底层逻辑伪代码实现：**

```python
class GraphEnhancedEgoVLM:
    def __init__(self, detector, graph_builder, vlm_backbone, alignment_head):
        self.detector = detector              # 手物检测与位姿估计
        self.graph_builder = graph_builder    # 动态图构建器
        self.vlm_backbone = vlm_backbone      # 视觉语言大模型骨干
        self.alignment_head = alignment_head  # 交互感知对比学习头

    def forward(self, video_stream, text_query):
        # 1. 基础感知：提取手与激活对象
        hands, active_objects = self.detector.extract_entities(video_stream)
        
        # 2. 关系构建：生成随时间演变的手物交互图 (HOI Graph)
        # 包含 Pre-condition, Contact, PNR, Post-condition 关键节点
        hoi_graph_sequence = self.graph_builder.build_dynamic_graph(hands, active_objects)
        
        # 3. 图引导采样：过滤冗余背景与运动模糊，仅保留交互关键帧
        key_frames, structured_prompts = self.graph_builder.sample_informative_frames(
            video_stream, hoi_graph_sequence, text_query
        )
        
        # 4. 交互感知特征提取：将图结构作为先验注入视觉编码器
        visual_features = self.vlm_backbone.encode_vision(key_frames, condition=structured_prompts)
        text_features = self.vlm_backbone.encode_text(text_query)
        
        # 5. 语义对齐：计算细粒度动词/名词感知的相似度
        # 惩罚那些“认对物体但认错动作”的硬负样本
        similarity_score = self.alignment_head.compute_interaction_aware_loss(
            visual_features, text_features
        )
        
        return similarity_score
```
