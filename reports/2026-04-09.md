class NeuralComputer:
    def __init__(self, video_model):
        # 核心是一个预训练的视频生成模型（充当World Model）
        self.model = video_model
        # 隐式内存：通过上下文历史来维持运行时状态
        self.context_memory = [] 

    def execute_step(self, instruction, current_frame, user_action):
        # 将指令、当前屏幕像素和用户动作拼接为模型输入
        prompt = self._format_input(instruction, current_frame, user_action)
        
        # 将当前输入加入上下文内存
        self.context_memory.append(prompt)
        
        # 核心计算：模型自回归地预测下一帧屏幕（即计算结果的视觉呈现）
        next_frame_pixels = self.model.generate_next_frame(self.context_memory)
        
        return next_frame_pixels

    def run_program(self, instruction, initial_frame, action_sequence):
        # 模拟计算机运行一段程序或响应一系列操作
        current_frame = initial_frame
        output_frames = []
        
        for action in action_sequence:
            current_frame = self.execute_step(instruction, current_frame, action)
            output_frames.append(current_frame)
            
        return output_frames

### 第二模块：核心架构解释

PhyEdit 的整体架构分为三个明确的客观执行阶段：3D 变换预处理、主干网络前向传播、以及 2D-3D 联合监督训练。

1. 3D 变换模块 (3D Transformation Module): 接收原始图像、物体掩码和目标移动向量。调用 3D 基础模型预测当前视角的深度图和相机位姿。利用这些参数，将 2D 像素反投影为 3D 空间坐标，在 3D 空间中对物体施加位移，最后将其重新投影回 2D 平面，生成一张具备正确透视关系的预览图。
2. 主干网络前向传播 (Backbone Training Forward): 采用 DiT 作为基础生成器。将文本指令特征、加噪的图像潜变量（Latent）、以及上一步生成的预览图特征在通道维度进行拼接，输入到 DiT 模块中预测噪声。
3. 联合监督机制 (Joint Supervision): 在训练计算损失时，不仅计算常规的潜空间去噪损失（MSE），还将预测出的干净潜变量解码回像素空间，再次通过 3D 基础模型提取深度图，与真实目标图像的深度图计算尺度不变对数损失（SILog Loss）。通过这种像素级深度对齐，强制模型输出符合 3D 物理逻辑的结果。

```python
def phyedit_forward_and_loss(image_src, image_tgt, mask, text_prompt, delta_p):
    # 阶段 1: 3D 变换模块生成物理预览图
    # 预测源图像深度与相机参数
    depth_src, camera_pose = depth_foundation_model(image_src)
    
    # 2D 到 3D 反投影
    points_3d = unproject(image_src, mask, depth_src, camera_pose)
    
    # 在 3D 空间中施加物理位移
    points_3d_moved = points_3d + delta_p
    
    # 3D 到 2D 重新投影，生成预览图
    image_preview = project(points_3d_moved, camera_pose)
    
    # 阶段 2: 主干网络前向传播
    # 编码输入条件
    latent_noisy = add_noise(encode(image_src))
    latent_preview = encode(image_preview)
    text_embeds = text_encoder(text_prompt)
    
    # 拼接条件并输入 DiT 预测速度/噪声
    condition = concatenate([text_embeds, latent_noisy, latent_preview])
    velocity_pred = DiT_backbone(condition)
    
    # 阶段 3: 2D-3D 联合监督计算损失
    # 1. 潜空间去噪损失 (2D 监督)
    velocity_gt = compute_ground_truth_velocity(image_src, image_tgt)
    loss_2d = MSELoss(velocity_pred, velocity_gt)
    
    # 2. 像素级深度损失 (3D 监督)
    latent_pred_clean = compute_clean_latent(latent_noisy, velocity_pred)
    image_pred = decode(latent_pred_clean)
    
    depth_pred, _ = depth_foundation_model(image_pred)
    depth_gt, _ = depth_foundation_model(image_tgt)
    loss_3d = SILogLoss(depth_pred, depth_gt)
    
    # 总损失
    total_loss = loss_2d + lambda_d * loss_3d
    return total_loss
```

1. **关键帧选择 (Keyframe Selection)**：计算连续帧之间的密集光流，以平均光流幅度作为运动显著性得分。通过时间非极大值抑制提取局部峰值作为关键帧。若数量不足，则均匀采样补充。

4. **上下文序列化 (Context Serialization)**：将机器人硬件配置、任务计划、带有时间戳的关键帧标签、接触状态和全局场景图拼接成结构化的文本提示，与 RGB 关键帧和伪鸟瞰图一起输入给 VLM。

### 第二模块：核心架构解释

T-STAR 的核心架构将传统的线性轨迹优化重构为树状拓扑优化，整个流程分为三个关键阶段：

**1. 认知树构建 (Cognitive Tree Construction)**
模型首先采样多条完整的 ReAct 轨迹。随后，通过评估节点间的“功能等价性”（基于动作分布的 KL 散度估计）和“历史兼容性”（改变状态的动作序列是否一致），将不同轨迹中实质相同的中间状态合并。这一步将独立的链条压缩成一棵包含共享前缀和分歧节点的认知树。

**2. 内省估值与思想嫁接 (Introspective Valuation & Thought Grafting)**
在认知树上执行 Bellman 回溯，将稀疏的轨迹级奖励转化为节点级的 Q 值，从而计算出方差更小的节点级优势。系统会扫描树中的内部节点，寻找子节点 Q 值差异巨大的“关键分歧点”。在这些点上，模型提取成功的子分支和失败的子分支，强制 Agent 在相同的上下文中对比两者，生成一条将成功逻辑转移到失败上下文中的“纠正性思维（Rectified Thought）”。

**3. 外科手术式策略优化 (Surgical Policy Optimization)**
最终的损失函数是双轨制的。全局层面保留 GRPO 损失，利用树状结构带来的低方差优势信号更新所有步骤；局部层面引入 Bradley-Terry 偏好损失（DPO 变体），仅在关键分歧点上，将生成的“纠正性思维”作为正样本，原始失败思维作为负样本，进行精准的局部梯度更新。

```python
def train_tstar(policy, tasks, M, delta, epsilon_kl):
    for task in tasks:
        # 1. 采样独立轨迹链条
        trajectories = sample_trajectories(policy, task, num_samples=M)
        
        # 2. 构建认知树：合并功能等价的节点
        tree = build_cognitive_tree(trajectories, epsilon_kl)
        
        # 3. 内省估值：通过 Bellman 回溯计算树节点 Q 值与优势
        q_values = compute_q_tree_values(tree)
        node_advantages = compute_tree_advantages(q_values)
        
        grafting_dataset = []
        
        # 4. 寻找关键分歧点并进行思想嫁接
        for node in tree.internal_nodes():
            v_plus, v_minus, value_spread = get_children_spread(node, q_values)
            
            # 如果子节点价值差异超过阈值，认定为关键决策失误点
            if value_spread > delta:
                # 对比成功与失败分支，生成纠正性思维
                z_rect = generate_rectified_thought(policy, node.context, v_plus, v_minus)
                grafting_dataset.append({
                    'context': node.context,
                    'z_rect': z_rect,       # 正样本
                    'z_fail': v_minus.z,    # 负样本
                    'timestep': node.timestep
                })
                
        # 5. 外科手术式策略优化
        # 全局轨迹级更新 (低方差)
        loss_grpo = compute_grpo_loss(policy, trajectories, node_advantages)
        # 局部关键步骤偏好更新 (精准纠正)
        loss_surgical = compute_bradley_terry_loss(policy, grafting_dataset) 
        
        total_loss = loss_grpo + lambda_weight * loss_surgical
        update_policy_weights(policy, total_loss)
```

### 第二模块：核心架构解释

Geo-EVS 的核心架构分为数据构建、鲁棒性训练和条件扩散生成三个阶段。首先，利用预训练的几何基础模型（VGGT）从源图像提取点云，并重投影至目标位姿，生成稀疏的几何条件图。其次，在训练时，从预先计算的伪影库中采样掩码并叠加到条件图上，模拟外推视角的几何缺失。最后，将处理后的条件图与噪声潜变量拼接，输入 U-Net 进行标准的去噪训练。

以下是核心逻辑的 Python 风格伪代码：

import torch
import torch.nn as nn

class GeoEVS(nn.Module):
    def __init__(self, vggt_model, unet, vae, artifact_library, p_drop=0.1, p_artifact=0.4):
        super().__init__()
        self.vggt = vggt_model # Frozen geometric feature extractor
        self.unet = unet       # Latent diffusion denoiser
        self.vae = vae
        self.artifact_library = artifact_library # Precomputed mask library A
        self.p_drop = p_drop
        self.p_artifact = p_artifact

    def geometry_aware_reprojection(self, source_img, source_pose, target_pose):
        # Extract dense point map and convert to colored point cloud P
        point_cloud = self.vggt.extract_geometry(source_img, source_pose)
        
        # Reproject P to target pose using z-buffer rasterization
        # Returns sparse geometric condition map X^v
        condition_map = z_buffer_rasterization(point_cloud, target_pose)
        return condition_map

    def inject_artifact_mask(self, condition_map):
        # Bernoulli sampling for artifact injection
        if torch.rand(1).item() < self.p_artifact:
            # Uniformly sample a mask from the library
            mask = self.artifact_library.sample()
            # Apply mask (element-wise multiplication)
            condition_map = condition_map * mask
        return condition_map

    def forward_train(self, source_img, target_img, source_pose, target_pose, t):
        # 1. Construct geometric condition map (GAR)
        # In training, target_pose == source_pose (observed views)
        condition_map = self.geometry_aware_reprojection(source_img, source_pose, target_pose)
        
        # 2. Inject artifacts to simulate extrapolation defects
        condition_map = self.inject_artifact_mask(condition_map)
        
        # 3. Classifier-free guidance dropout
        if torch.rand(1).item() < self.p_drop:
            condition_map = torch.zeros_like(condition_map) # Null condition
            
        # 4. Latent Diffusion
        z_0 = self.vae.encode(target_img)
        noise = torch.randn_like(z_0)
        z_t = add_noise(z_0, noise, t)
        
        # Concatenate condition map features with noisy latent
        cond_feat = encode_condition(condition_map)
        unet_input = torch.cat([z_t, cond_feat], dim=1)
        
        # Predict noise
        noise_pred = self.unet(unet_input, t)
        
        # Compute MSE loss
        loss = nn.MSELoss()(noise_pred, noise)
        return loss

    def forward_inference(self, source_img, source_pose, extrapolated_pose, steps=30):
        # Unified reprojection interface for out-of-trajectory pose
        condition_map = self.geometry_aware_reprojection(source_img, source_pose, extrapolated_pose)
        
        # Start from pure noise
        z_t = torch.randn((1, latent_channels, H, W))
        cond_feat = encode_condition(condition_map)
        
        # Denoising loop with classifier-free guidance
        for t in reversed(range(steps)):
            unet_input = torch.cat([z_t, cond_feat], dim=1)
            noise_pred_cond = self.unet(unet_input, t)
            
            # Null condition for CFG
            null_cond_feat = encode_condition(torch.zeros_like(condition_map))
            unet_input_uncond = torch.cat([z_t, null_cond_feat], dim=1)
            noise_pred_uncond = self.unet(unet_input_uncond, t)
            
            # Apply guidance scale
            noise_pred = noise_pred_uncond + 1.5 * (noise_pred_cond - noise_pred_uncond)
            z_t = step_denoise(z_t, noise_pred, t)
            
        # Decode to image space
        extrapolated_img = self.vae.decode(z_t)
        return extrapolated_img
